rite_compatibility_modifier = {
	modifier = {
		add = $SCALE$
		exists = $TARGET$
		$TARGET$ ?= { this = $FAITH$ }
	}
	modifier = {
		add = $SCALE$
		exists = $TARGET$
		$TARGET$ ?= { rite_hostility_level = { target = $RITE$ value > 3 } }
	}
	modifier = {
		add = $SCALE$
		exists = $TARGET$
		$TARGET$ ?= { rite_hostility_level = { target = $RITE$ value > 2 } }
	}
	modifier = {
		add = $SCALE$
		exists = $TARGET$
		$TARGET$ ?= { rite_hostility_level = { target = $RITE$ value > 1 } }
	}
}
average_sub_realm_rite_compatibility_modifier = {
	modifier = {
		add = {
			value = 0
			$TARGET$ = {
				every_sub_realm_county = {
					title_province = {
						rite = {
							if = {
								limit = { rite_hostility_level = { target = $RITE$ value > 3 } }
								add = $SCALE$
							}
							if = {
								limit = { rite_hostility_level = { target = $RITE$ value > 2 } }
								add = $SCALE$
							}
							if = {
								limit = { rite_hostility_level = { target = $RITE$ value > 1 } }
								add = $SCALE$
							}
							if = {
								limit = { this = $RITE$ }
								add = $SCALE$
							}
						}
					}
				}
			}
			divide = {
				value = 0
				$TARGET$ = { every_sub_realm_county = { add = 1 } }
				min = 1
			}
		}
	}
}
