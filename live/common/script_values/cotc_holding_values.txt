# A rough measurement of the economic needs of this system's population
county_demand = {
	value = 0
	county = { every_county_province = { add = holding_demand } }
	min = 0
}
holding_demand = {
	value = holding_population
}
holding_population = {
	value = 0
	if = {
		limit = { has_building = temple_01 }
		add = low_demand_tier_1
	}
	else_if = {
		limit = { has_building = temple_02 }
		add = low_demand_tier_2
	}
	else_if = {
		limit = { has_building = temple_03 }
		add = low_demand_tier_3
	}
	else_if = {
		limit = { has_building = temple_04 }
		add = low_demand_tier_4
	}
	else_if = {
		limit = { has_building = temple_05 }
		add = low_demand_tier_5
	}
	else_if = {
		limit = { has_building = castle_01 }
		add = mid_demand_tier_1
	}
	else_if = {
		limit = { has_building = castle_02 }
		add = mid_demand_tier_2
	}
	else_if = {
		limit = { has_building = castle_03 }
		add = mid_demand_tier_3
	}
	else_if = {
		limit = { has_building = castle_04 }
		add = mid_demand_tier_4
	}
	else_if = {
		limit = { has_building = castle_05 }
		add = mid_demand_tier_5
	}
	else_if = {
		limit = { has_building = city_01 }
		add = high_demand_tier_1
	}
	else_if = {
		limit = { has_building = city_02 }
		add = high_demand_tier_2
	}
	else_if = {
		limit = { has_building = city_03 }
		add = high_demand_tier_3
	}
	else_if = {
		limit = { has_building = city_04 }
		add = high_demand_tier_4
	}
	else_if = {
		limit = { has_building = city_05 }
		add = high_demand_tier_5
	}
}
county_development_deficit = { # Returns the difference between the county's Demand and Development. Useful for AI consideration.
	value = 0
	add = county_demand
	subtract = development
}
county_economic_anxiety = { # How concerned is the population of this county over the economy?
	value = 0
	add = county_development_deficit
	scope:relevant_title ?= {
		holder = {
			if = {
				limit = { debt_level >= 0 }
				add = {
					value = debt_level
					add = 1
					multiply = 5
				}
			}
		}
	}
}
county_security_anxiety = { # How concerned is the population of this county over their security?
	value = 0
	# Security anxiety due to low control
	add = {
		value = 100
		subtract = county_control
	}
}
