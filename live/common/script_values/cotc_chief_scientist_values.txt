}

chief_scientist_spread_innovations_contextual_bonuses = {
	if = {
		limit = { councillor_liege_has_erudition_legacy_5_perk = yes }
		add = {
			value = scope:councillor.chief_scientist_spread_innovations_monthly_increase_dynasty_perk_bonus
			desc = SPREAD_INNOVATION_DYNASTY_PERK_BONUS_VALUE
		}
	}
	if = {
		limit = { councillor_liege_has_family_business_bonus = yes }
		add = {
			value = scope:councillor.chief_scientist_spread_innovations_monthly_increase_family_business_bonus
			desc = FAMILY_BUSINESS_BONUS_VALUE
		}
	}
	if = {
		limit = { councillor_liege_has_consulted_house_bonus = yes }
		add = {
			value = scope:councillor.chief_scientist_spread_innovations_monthly_increase_consulted_house_bonus
			desc = CONSULTED_HOUSE_BONUS_VALUE
		}
	}
	if = {
		limit = { scope:county.faith = { has_doctrine_parameter = faster_innovation_spread } }
		add = {
			value = scope:county.faith.faster_innovation_spread_bonus
			desc = COUNTY_FASTER_INNOVATION_SPREAD
		}
	}
	# Development bonus.
	if = {
		limit = {
			exists = scope:county
			scope:county = { development_level > 0 }
		}
		add = {
			value = scope:county.county_development_innovation_spread_bonus
			desc = CHIEF_SCIENTIST_SPREAD_INNOVATIONS_DEVELOPMENT_BONUS
		}
	}
	# Population penalty
	if = {
		limit = { exists = scope:county }
		subtract = {
			value = scope:county.county_demand_innovation_spread_penalty
			desc = CHIEF_SCIENTIST_SPREAD_INNOVATIONS_DEMAND_PENALTY
		}
	}
}
