# Technology Preferences
social_research_project_score = {
	add = diplomacy
	add = intrigue
}
economic_research_project_score = {
	add = stewardship
}
military_research_project_score = {
	add = ai_hawkishness
	if = { # Favor military projects when threatened
		limit = { is_threatened_trigger = yes }
		add = {
			value = 100
			add = ai_rationality
			divide = 20
			min = 0
		}
	}
	if = { # Heavily favor military projects in wartime
		limit = { is_at_war = yes }
		add = {
			value = 100
			add = ai_rationality
			divide = 10
			min = 0
		}
	}
}
learning_research_project_score = {

medical_research_project_score = {

cloning_research_project_score = {

# Research Project Rewards
research_project_prestige_reward = {
	value = 50 # Base reward
	if = { # Our culture doesn't have this innovation
		limit = { culture = { NOT = { has_innovation = scope:discovered_innovation } } }
		add = 100
	}
	if = { # Nobody has this innovation
		limit = { scope:first_ever_discovery ?= flag:yes }
		add = 100
	}
	if = { # Era bonus
		limit = {
			scope:discovered_innovation = {
				OR = {
					has_innovation_parameter = colonial_civic
					has_innovation_parameter = colonial_military
				}
			}
			culture = { NOT = { has_cultural_era_or_later = culture_era_early_medieval } }
		}
		multiply = 1.25
	}
	else_if = {
		limit = {
			scope:discovered_innovation = {
				OR = {
					has_innovation_parameter = confederate_civic
					has_innovation_parameter = confederate_military
				}
			}
		}
		culture = {
			if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_early_medieval } }
				multiply = 1.5
			}
			else_if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_high_medieval } }
				multiply = 1.25
			}
		}
	}
	else_if = {
		limit = {
			scope:discovered_innovation = {
				OR = {
					has_innovation_parameter = trimillennial_civic
					has_innovation_parameter = trimillennial_military
				}
			}
		}
		culture = {
			if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_early_medieval } }
				multiply = 1.75
			}
			else_if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_high_medieval } }
				multiply = 1.5
			}
			else_if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_late_medieval } }
				multiply = 1.25
			}
		}
	}
	else_if = {
		limit = {
			scope:discovered_innovation = {
				OR = {
					has_innovation_parameter = neorenaissance_civic
					has_innovation_parameter = neorenaissance_military
				}
			}
		}
		culture = {
			if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_early_medieval } }
				multiply = 2
			}
			else_if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_high_medieval } }
				multiply = 1.75
			}
			else_if = {
				limit = { NOT = { has_cultural_era_or_later = culture_era_late_medieval } }
				multiply = 1.5
			}
			else = { multiply = 1.25 }
		}
	}
}
research_project_piety_reward = {
	value = 0
	faith = {
		if = {
			limit = { exists = scope:military }
			add = {
				value = ideology_hawkishness
				multiply = 50
			}
		}
		if = {
			limit = { exists = scope:cybernetics }
			if = {
				limit = { has_doctrine_parameter = hates_cybernetics }
				subtract = 100
			}
			else_if = {
				limit = { has_doctrine_parameter = supports_cybernetics }
				add = 100
			}
		}
		if = {
			limit = { exists = scope:gene_modding }
			if = {
				limit = { has_doctrine_parameter = hates_gene_modding }
				subtract = 100
			}
			else_if = {
				limit = { has_doctrine_parameter = supports_gene_modding }
				add = 100
			}
		}
		if = {
			limit = { exists = scope:cloning }
			if = {
				limit = { has_doctrine_parameter = hates_cloning }
				subtract = 100
			}
			else_if = {
				limit = { has_doctrine_parameter = supports_cloning }
				add = 100
			}
		}
	}
}
