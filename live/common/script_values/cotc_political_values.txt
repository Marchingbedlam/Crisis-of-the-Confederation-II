# Constants
miniscule_campaign_strength_change = 0.01
minor_campaign_strength_change = 0.02
moderate_campaign_strength_change = 0.05
major_campaign_strength_change = 0.1
massive_campaign_strength_change = 0.2

title_political_support = {
	value = 0
	save_temporary_scope_as = temp_candidate
	scope:title = {
		every_de_jure_county = {
			every_county_province = {
				limit = {
					faith = scope:temp_candidate.faith
					province_owner ?= { target_is_same_character_or_above = scope:title.holder }
				}
				add = holding_population
			}
		}
	}
	# Multiply by level of devotion
	multiply = {
		value = 1
		add = piety_level
	}
	# Multiply by fervor
	faith = { multiply = fervor }
	# Divide by 10 to keep a reasonable total value
	divide = 10
	round = yes
}
political_campaign_cost_modifier = {
	value = 0
	if = {
		limit = { type = legislative_election_campaign }
		var:legislature ?= { var:legislature_title ?= { add = title_basic_cost_multiplier } }
	}
	else = { scheme_target_title ?= { add = title_leadership_cost_multiplier } }
}
county_election_score = { # Returns scope:temp_candidate's popularity in the scoped county
	value = 0
	title_province = {
		faith = { save_temporary_scope_as = temp_faith }
		culture = { save_temporary_scope_as = temp_culture }
	}
	# Opinion of incumbent
	if = {
		limit = { reverse_is_incumbent_for_title = { CHARACTER = scope:temp_candidate } }
		add = county_opinion
	}
	scope:temp_candidate = {
		# Opinion of candidate's ideology and credibility
		add = {
			value = 0
			if = {
				limit = { scope:temp_faith = { faith_hostility_level = { target = scope:temp_candidate.faith value = 3 } } }
				subtract = 30
			}
			else_if = {
				limit = { scope:temp_faith = { faith_hostility_level = { target = scope:temp_candidate.faith value = 2 } } }
				subtract = 20
			}
			else = {
				add = {
					value = piety_level
					multiply = 5
				}
			}
			multiply = scope:temp_faith.fervor
			divide = 100
		}
		# Opinion of candidate's personality
		if = {
			limit = { num_virtuous_traits = { value >= 5 target = scope:temp_faith } }
			add = 25
		}
		else_if = {
			limit = { num_virtuous_traits = { value >= 4 target = scope:temp_faith } }
			add = 20
		}
		else_if = {
			limit = { num_virtuous_traits = { value >= 3 target = scope:temp_faith } }
			add = 15
		}
		else_if = {
			limit = { num_virtuous_traits = { value >= 2 target = scope:temp_faith } }
			add = 10
		}
		else_if = {
			limit = { num_virtuous_traits = { value >= 1 target = scope:temp_faith } }
			add = 5
		}
		if = {
			limit = { num_sinful_traits = { value >= 5 target = scope:temp_faith } }
			subtract = 25
		}
		else_if = {
			limit = { num_sinful_traits = { value >= 4 target = scope:temp_faith } }
			subtract = 20
		}
		else_if = {
			limit = { num_sinful_traits = { value >= 3 target = scope:temp_faith } }
			subtract = 15
		}
		else_if = {
			limit = { num_sinful_traits = { value >= 2 target = scope:temp_faith } }
			subtract = 10
		}
		else_if = {
			limit = { num_sinful_traits = { value >= 1 target = scope:temp_faith } }
			subtract = 5
		}
		# Skill effects
		if = {
			limit = { scope:temp_faith = { has_doctrine = doctrine_military_power } }
			add = {
				value = martial
				subtract = 10
				multiply = 3
			}
		}
		if = {
			limit = { scope:temp_faith = { has_doctrine = doctrine_commercial_power } }
			add = {
				value = stewardship
				subtract = 10
				multiply = 3
			}
		}
		if = {
			limit = {
				scope:temp_faith = {
					OR = {
						has_doctrine = doctrine_technocratic_power
						has_doctrine = doctrine_theocratic_power
					}
				}
			}
			add = {
				value = learning
				subtract = 10
				multiply = 3
			}
		}
		# Cultural bias
		if = {
			limit = { NOT = { culture = scope:temp_culture } }
			divide = {
				value = 2
				if = {
					limit = { scope:temp_faith = { has_doctrine = doctrine_particularism } }
					multiply = 5
				}
			}
		}
		# Sexism
		if = {
			limit = {
				OR = {
					AND = {
						is_male_cotc = no
						scope:temp_faith = { has_doctrine = doctrine_gender_male_dominated }
					}
					AND = {
						is_female_cotc = no
						scope:temp_faith = { has_doctrine = doctrine_gender_female_dominated }
					}
				}
			}
			divide = 10
		}
	}
}
candidate_election_score = { # General effects of the candidate as a person
	value = 0
	# Candidate's Prestige
	add = {
		value = prestige_level
		multiply = 3
	}
	# Skill Effects
	add = {
		value = diplomacy
		subtract = 10
		multiply = 3
	}
}

legislator_endorsement_score = { # How valuable is your endorsement of this legislator?
	value = candidate_election_score
	save_temporary_scope_as = temp_candidate
	scope:temp_riding = {
		if = {
			limit = { tier = tier_county }
			add = county_election_score
		}
		else = {
			every_de_jure_county = {
				limit = { target_is_de_facto_liege_or_above = scope:temp_legislature.var:legislature_title }
				add = county_election_score
			}
			divide = {
				value = 0
				every_de_jure_county = {
					limit = { target_is_de_facto_liege_or_above = scope:temp_legislature.var:legislature_title }
					add = 1
				}
				min = 1
			}
		}
	}
}
legislator_base_score = { # Base score for a candidate in a legislative election
	value = candidate_election_score
	save_temporary_scope_as = temp_candidate
	# Population opinion effects
	scope:temp_riding = {
		if = {
			limit = { tier = tier_county }
			add = county_election_score
		}
		else = {
			every_de_jure_county = {
				limit = { target_is_de_facto_liege_or_above = scope:temp_legislature.var:legislature_title }
				add = county_election_score
			}
			divide = {
				value = 0
				every_de_jure_county = {
					limit = { target_is_de_facto_liege_or_above = scope:temp_legislature.var:legislature_title }
					add = 1
				}
				min = 1
			}
		}
	}
	# Campaign effects
	every_scheme = {
		limit = {
			scheme_type = legislative_election_campaign
			var:legislature ?= { this = scope:temp_legislature }
			has_variable = public_campaign_strength
		}
		multiply = { value = var:public_campaign_strength }
	}
}
legislator_total_score = { # This allows for undisplayed "hidden" variables that affect election outcomes
	value = legislator_base_score
	# Campaign effects
	every_scheme = {
		limit = {
			scheme_type = legislative_election_campaign
			var:legislature ?= { this = scope:temp_legislature }
			has_variable = hidden_campaign_strength
		}
		multiply = var:hidden_campaign_strength
	}
	# Random fudge factor
	multiply = {
		value = {
			fixed_range = {
				min = 0.9
				max = 1.1
			}
		}
	}
}

legislative_seniority = { # Score calculation for how early a legislator gets to speak
	value = 0
	add = {
		value = influence_level
		multiply = 5
	}
	add = {
		value = prestige_level
		multiply = 3
	}
	add = piety_level
	if = {
		limit = { has_variable = years_in_legislature }
		add = var:years_in_legislature
	}
}
months_until_next_election = { # How long until the next election?
	value = 0
	var:legislature_title.holder = {
		switch = {
			trigger = has_realm_law
			two_year_terms_law = { add = 21 }
			four_year_terms_law = { add = 45 }
			six_year_terms_law = { add = 69 }
			eight_year_terms_law = { add = 93 }
			ten_year_terms_law = { add = 117 }
		}
	}
	if = {
		limit = { exists = var:months_in_session }
		subtract = var:months_in_session
	}
}
ideology_represented_in_legislature = { # How many legislators follow this ideology?
	value = 0
	save_temporary_scope_as = temp_faith
	scope:party_legislature = {
		every_in_list = {
			variable = legislators
			limit = { faith = scope:temp_faith }
			add = 1
		}
	}
}
