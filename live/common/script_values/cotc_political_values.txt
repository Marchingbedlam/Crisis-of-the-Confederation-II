# This calculates a basic level of political support for a character among the general
# population of a particular title (both de jure and de facto). This can return a fairly
# variable number so it's recommended that any modifications to this be multiplicative
# rather than additive.
# scope:title: The title being evaluated
title_political_support = {
	value = 0
	save_temporary_scope_as = temp_candidate
	scope:title = {
		every_de_jure_county = {
			every_county_province = {
				limit = {
					faith = scope:temp_candidate.faith
					province_owner = { target_is_same_character_or_above = scope:title.holder }
				}
				add = holding_population
			}
		}
	}
	# Multiply by level of devotion
	multiply = {
		value = 1
		add = piety_level
	}
	# Multiply by fervor
	faith = { multiply = fervor }
	# Divide by 10 to keep a reasonable total value
	divide = 10
	round = yes
}
# Score calculation for legislative elections
legislator_score = {
	value = 0
	save_temporary_scope_as = temp_candidate
	# Opinion of incumbent
	if = {
		limit = { is_in_list = incumbents }
		add = scope:county.county_opinion
	}
	# Opinion of candidate's ideology and credibility
	add = {
		value = 0
		if = {
			limit = { scope:temp_faith = { faith_hostility_level = { target = scope:temp_candidate.faith value = 3 } } }
			subtract = 30
		}
		else_if = {
			limit = { scope:temp_faith = { faith_hostility_level = { target = scope:temp_candidate.faith value = 2 } } }
			subtract = 20
		}
		else = {
			add = {
				value = piety_level
				multiply = 5
			}
		}
		multiply = scope:temp_faith.fervor
		divide = 100
	}
	# Opinion of candidate's prestige
	add = {
		value = prestige_level
		multiply = 3
	}
	# Opinion of candidate's personality
	if = {
		limit = { num_virtuous_traits = { value >= 5 target = scope:temp_faith } }
		add = 25
	}
	else_if = {
		limit = { num_virtuous_traits = { value >= 4 target = scope:temp_faith } }
		add = 20
	}
	else_if = {
		limit = { num_virtuous_traits = { value >= 3 target = scope:temp_faith } }
		add = 15
	}
	else_if = {
		limit = { num_virtuous_traits = { value >= 2 target = scope:temp_faith } }
		add = 10
	}
	else_if = {
		limit = { num_virtuous_traits = { value >= 1 target = scope:temp_faith } }
		add = 5
	}
	if = {
		limit = { num_sinful_traits = { value >= 5 target = scope:temp_faith } }
		subtract = 25
	}
	else_if = {
		limit = { num_sinful_traits = { value >= 4 target = scope:temp_faith } }
		subtract = 20
	}
	else_if = {
		limit = { num_sinful_traits = { value >= 3 target = scope:temp_faith } }
		subtract = 15
	}
	else_if = {
		limit = { num_sinful_traits = { value >= 2 target = scope:temp_faith } }
		subtract = 10
	}
	else_if = {
		limit = { num_sinful_traits = { value >= 1 target = scope:temp_faith } }
		subtract = 5
	}
	# Skill effects
	add = {
		value = diplomacy
		subtract = 10
		multiply = 3
	}
	if = {
		limit = { scope:temp_faith = { has_doctrine = doctrine_military_power } }
		add = {
			value = martial
			subtract = 10
			multiply = 3
		}
	}
	if = {
		limit = { scope:temp_faith = { has_doctrine = doctrine_commercial_power } }
		add = {
			value = stewardship
			subtract = 10
			multiply = 3
		}
	}
	if = {
		limit = { scope:temp_faith = { has_doctrine = doctrine_administrative_power } }
		add = {
			value = learning
			subtract = 10
			multiply = 3
		}
	}
	# Hook so characters can manipulate elections
	if = {
		limit = { exists = var:legislative_external_value }
		add = var:legislative_external_value
	}
	# Cultural bias
	if = {
		limit = { NOT = { culture = scope:temp_culture } }
		divide = {
			value = 2
			if = {
				limit = { scope:temp_faith = { has_doctrine = doctrine_particularism } }
				multiply = 5
			}
		}
	}
	# Sexism
	if = {
		limit = {
			OR = {
				AND = {
					is_female_cotc = yes
					scope:temp_faith = { has_doctrine = doctrine_gender_male_dominated }
				}
				AND = {
					is_male_cotc = yes
					scope:temp_faith = { has_doctrine = doctrine_gender_female_dominated }
				}
			}
		}
		divide = 10
	}
}

# How much do you approve of the bill being debated? (used for characters who aren't part of the debate directly)
opinion_of_bill = {
	value = 0
	# BUDGET OPINIONS
	# Total Budget
	if = {
		limit = {
			scope:activity.proposed_total_budget > 100
			scope:activity.proposed_total_budget > scope:activity.var:ruler.current_total_budget
			OR = {
				ai_interventionism < 0
				ai_rationality > 0
			}
		}
		subtract = {
			value = scope:activity.proposed_budget_deficit
			if = {
				limit = { scope:activity.var:ruler.debt_level >= 0 }
				multiply = {
					value = scope:activity.var:ruler.debt_level
					add = 2
				}
			}
		}
	}
	if = {
		limit = {
			scope:activity.var:ruler.current_total_budget > 100
			scope:activity.proposed_total_budget < scope:activity.var:ruler.current_total_budget
			OR = {
				ai_interventionism < 0
				ai_rationality > 0
			}
		}
		subtract = {
			value = activity.proposed_budget_deficit
			if = {
				limit = { scope:activity.var:ruler.debt_level >= 0 }
				multiply = {
					value = scope:activity.var:ruler.debt_level
					add = 2
				}
			}
		}
	}
	# Specific Budgets - Personality
	if = {
		limit = {
			scope:activity = {
				has_variable = proposed_military_budget
				var:proposed_military_budget < var:ruler.current_military_budget
			}
		}
		subtract = ai_hawkishness
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_military_budget
				var:proposed_military_budget > var:ruler.current_military_budget
			}
		}
		add = ai_hawkishness
	}
	if = {
		limit = {
			scope:activity = {
				has_variable = proposed_salary_budget
				var:proposed_salary_budget < var:ruler.current_salary_budget
			}
		}
		subtract = ai_intellectualism
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_salary_budget
				var:proposed_salary_budget > var:ruler.current_salary_budget
			}
		}
		add = ai_intellectualism
	}
	# Specific Budgets - Self-Interest
	if = {
		limit = {
			scope:activity = {
				has_variable = proposed_military_budget
				var:proposed_military_budget < var:ruler.current_military_budget
			}
			court_is_military_trigger = yes
		}
		subtract = {
			value = ai_greed
			min = 0
		}
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_military_budget
				var:proposed_military_budget > var:ruler.current_military_budget
			}
			court_is_military_trigger = yes
		}
		add = {
			value = ai_greed
			min = 0
		}
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_ministry_budget
				var:proposed_ministry_budget < var:ruler.current_ministry_budget
			}
			court_is_commercial_trigger = yes
		}
		subtract = {
			value = ai_greed
			min = 0
		}
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_ministry_budget
				var:proposed_ministry_budget > var:ruler.current_ministry_budget
			}
			court_is_commercial_trigger = yes
		}
		add = {
			value = ai_greed
			min = 0
		}
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_salary_budget
				var:proposed_salary_budget < var:ruler.current_salary_budget
			}
			court_is_administrative_trigger = yes
		}
		subtract = {
			value = ai_greed
			min = 0
		}
	}
	else_if = {
		limit = {
			scope:activity = {
				has_variable = proposed_salary_budget
				var:proposed_salary_budget > var:ruler.current_salary_budget
			}
			court_is_administrative_trigger = yes
		}
		add = {
			value = ai_greed
			min = 0
		}
	}
}
# How hard a "sell" is this bill? (for diplomacy challenge purposes)
bill_sell_difficulty = {
	value = 20
	add = bill_unreasonableness
	subtract = bill_necessity
	min = 10
}
# How hard a "target" is this bill?
bill_target_difficulty = {
	value = 20
	add = bill_necessity
	subtract = bill_unreasonableness
	min = 10
}
bill_necessity = {
	value = 0
	if = {
		limit = { has_activity_type = budget_bill_activity }
		if = {
			limit = {
				proposed_budget_deficit < 0
				scope:ruler.current_total_budget > 100
			}
			add = {
				value = scope:ruler.current_total_budget
				subtract = proposed_total_budget
				divide = 5
			}
		}
		if = {
			limit = {
				scope:ruler = { is_at_war = yes }
				var:proposed_military_budget > scope:ruler.current_military_budget
			}
			add = {
				value = scope:ruler.current_military_budget
				subtract = var:proposed_military_budget
				divide = 5
			}
		}
		if = {
			limit = {
				proposed_total_budget < scope:ruler.current_total_budget
				proposed_budget_deficit < 0
				scope:ruler.debt_level >= 0
			}
			add = {
				value = scope:ruler.debt_level
				add = 1
				multiply = 3
			}
		}
	}
}
bill_unreasonableness = {
	value = 0
	if = {
		limit = { has_activity_type = budget_bill_activity }
		if = {
			limit = { proposed_budget_deficit > 0 }
			add = {
				value = proposed_budget_deficit
				divide = 2
				ceiling = yes
			}
		}
		if = {
			limit = {
				proposed_total_budget >= scope:ruler.current_total_budget
				scope:ruler.debt_level >= 0
			}
			add = {
				value = scope:ruler.debt_level
				add = 1
				multiply = 3
			}
		}
		if = {
			limit = { var:proposed_military_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_military_budget
				if = {
					limit = { scope:ruler = { is_at_war = yes } }
					multiply = 2
				}
			}
		}
		if = {
			limit = { var:proposed_ministry_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_military_budget
				divide = 3
				ceiling = yes
			}
		}
		if = {
			limit = { var:proposed_salary_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_salary_budget
				divide = 2
				ceiling = yes
			}
		}
	}
}

# These are used to evaluate a legislator's initial stance on a budget
military_spending_score = {
	value = 0
	if = {
		limit = { current_military_budget = scope:proposed_budget }
		add = 200
	}
	else_if = {
		limit = { current_military_budget > scope:proposed_budget }
		add = lower_military_spending_score
	}
	else_if = {
		limit = { current_military_budget < scope:proposed_budget }
		add = raise_military_spending_score
	}
}
lower_military_spending_score = {
	value = 0
	# Personality
	add = ai_greed
	subtract = ai_hawkishness
	# Military spending is urgently needed
	if = {
		limit = { scope:budget_ruler = { is_at_war = yes } }
		subtract = 20
		subtract = {
			value = ai_rationality
			min = 0
		}
	}
	# We are in debt!
	if = {
		limit = { scope:budget_ruler.debt_level >= 0 }
		add = {
			value = ai_rationality
			min = 0
