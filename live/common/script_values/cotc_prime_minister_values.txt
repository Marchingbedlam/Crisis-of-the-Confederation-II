# This is used to calculate the "default" popularity of a prime minister
prime_minister_score = {
	value = legislative_seniority
	multiply = {
		value = 0
		liege = {
			every_held_title = {
				limit = { has_variable = legislature }
				var:legislature ?= {
					every_in_list = {
						variable = legislators
						limit = { faith = { this = ROOT.faith } }
						add = 1
					}
				}
			}
		}
	}
}
total_supporters_that_care_about_qualifications = {
	value = 0
	if = {
		limit = { NOT = { exists = scope:activity } }
		save_temporary_scope_as = activity
	}
	every_guest_subset = {
		name = bill_supporters
		limit = { legislator_cares_about_qualifications_trigger = yes }
		add = 1
	}
}
