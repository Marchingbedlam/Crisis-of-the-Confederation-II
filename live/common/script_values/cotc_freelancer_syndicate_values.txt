cotc_syndicate_target_status_value = {
	value = 0
	if = { limit = { cotc_syndicate_status_independent_trigger = yes } add = 1 }
	else_if = { limit = { cotc_syndicate_status_protected_trigger = yes } add = 2 }
	else_if = { limit = { cotc_syndicate_status_frontier_trigger = yes } add = 3 }
}

cotc_syndicate_grip_value = {
	value = 0
	if = { limit = { has_variable = cotc_syndicate_grip } add = var:cotc_syndicate_grip }
}

cotc_syndicate_crowding_value = {
	value = 1
	if = {
		limit = { cotc_has_syndicate_trigger = yes }
		value = 0
		var:cotc_syndicate_turf = {
			every_in_list = { variable = cotc_syndicate_operators  add = 1 }
		}
		min = 1
	}
}

cotc_syndicate_grip_gain_value = {
	value = 6

	if = {
		limit = { var:cotc_syndicate_turf ?= { variable_list_size = { name = cotc_syndicate_operators  value >= 2 } } }
		multiply = 0.25
	}
	if = {
		limit = { var:cotc_syndicate_turf ?= { variable_list_size = { name = cotc_syndicate_operators  value >= 3 } } }
		multiply = 0.50
	}
	if = {
		limit = { var:cotc_syndicate_turf ?= { variable_list_size = { name = cotc_syndicate_operators  value >= 5 } } }
		multiply = 0.75
	}

	if = { limit = { cotc_syndicate_status_independent_trigger = yes } multiply = 1.25 }
	if = { limit = { cotc_syndicate_status_frontier_trigger = yes } multiply = 0.8 }
	if = { limit = { var:cotc_syndicate_turf ?= { development_level >= 10 } } multiply = 1.2 }

	min = 1
}

cotc_syndicate_grip_suborn_value = 60
cotc_syndicate_grip_takeover_value = 80
cotc_syndicate_grip_secession_value = 90
cotc_syndicate_takeover_cb_cost = 150

# HEAT is notoriety, held per boss rather than per System, so two syndicates sharing
# a System draw separate attention and the loudest one takes the raid.
cotc_syndicate_heat_raw_value = {
	value = 0
	if = { limit = { has_variable = cotc_syndicate_heat } add = var:cotc_syndicate_heat }
}

# The number the bands are read from.
cotc_syndicate_heat_value = {
	value = cotc_syndicate_heat_raw_value

	# A big operation is harder to miss than a small one.
	add = {
		value = cotc_syndicate_grip_value
		divide = 2
	}

	# Already a known criminal to the System's holder.
	if = {
		limit = {
			var:cotc_syndicate_turf.holder ?= {
				has_opinion_modifier = {
					target = root
					modifier = gallowsbait_scum_opinion
				}
			}
		}
		add = 25
	}

	if = { limit = { cotc_syndicate_status_independent_trigger = yes } multiply = 0.75 }
	if = { limit = { cotc_syndicate_status_frontier_trigger = yes } multiply = 1.5 }

	# A Sector already drowning in crime has less attention to spare
	if = {
		limit = {
			any_character_situation = {
				any_situation_sub_region = {
					has_sub_region_phase_parameter = cotc_underworld_rich
					any_situation_sub_region_participant_group = {
						participant_group_type = underworld_bosses
						participant_group_has_character = root
					}
				}
			}
		}
		multiply = 0.8
	}

	floor = yes
	min = 0
}

# Heat added by one criminal job completed on your own turf.
cotc_crackdown_heat_per_job_value = 8
# Heat shed each year when nothing new happens.
cotc_crackdown_heat_yearly_decay_value = 4
# The pulse adds heat rather than subtracting it, so it needs the negative form.
cotc_crackdown_heat_yearly_decay_negative_value = {
	value = cotc_crackdown_heat_yearly_decay_value
	multiply = -1
}

cotc_interdiction_cb_cost = 100
