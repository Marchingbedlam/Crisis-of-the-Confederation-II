counter_synergy_ai_weight_ew_cruisers = {
    # EW Cruisers counter destroyers, which counter corvettes & missiles
	value = 0
	if = {
		limit = {
			OR = {
				number_maa_regiments_of_base_type = {
					type = missile_boats
					value >= 3
				}
				number_maa_regiments_of_base_type = {
					type = corvettes
					value >= 3
				}
				number_maa_regiments_of_base_type = {
					type = missile_frigates
					value >= 2  # Frigates are more expensive after all
				}
			}
			NOT = { #no need to add more if i.e. regional or inherited maa of that type already exists
				number_maa_regiments_of_base_type = {
					type = ew_cruisers
					value >= 1
				}
			}
		}
		add = 60
	}
}
