opinion_of_rite = {
	value = scope:target_rite.fervor
	# Current rite's opinion of this one
	rite = {
		if = {
			limit = { rite_hostility_level = { target = scope:target_rite value >= 3 } }
			subtract = 200
		}
		else_if = {
			limit = { rite_hostility_level = { target = scope:target_rite value >= 2 } }
			subtract = 100
		}
		else_if = {
			limit = { rite_hostility_level = { target = scope:target_rite value = 0 } }
			add = 100
		}
	}
	if = {
		limit = { rite = scope:target_rite }
		add = ai_zeal
	}
	else = { subtract = ai_zeal }
	# Practical considerations
	if = { # Ideology is aristocratic
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_aristocracy } }
		if = {
			limit = {
				is_ruler = yes
				government_has_flag = government_is_aristocratic
			}
			add = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { is_ruler = yes }
			subtract = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { holder = { government_has_flag = government_is_aristocratic } } }
			add = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { exists = yes } }
			subtract = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology is democratic
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_democracy } }
		if = {
			limit = {
				is_ruler = yes
				government_has_flag = government_is_democratic
			}
			add = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { is_ruler = yes }
			subtract = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { holder = { government_has_flag = government_is_democratic } } }
			add = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { exists = yes } }
			subtract = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology is meritocratic
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_meritocracy } }
		if = {
			limit = {
				is_ruler = yes
				government_has_flag = government_is_meritocratic
			}
			add = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { is_ruler = yes }
			subtract = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { holder = { government_has_flag = government_is_meritocratic } } }
			add = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { exists = yes } }
			subtract = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology considers all rulers illegitimate
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_no_legitimacy } }
		if = {
			limit = { is_ruler = yes }
			subtract = {
				value = 50
				add = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = { any_heir_title = { exists = yes } }
			subtract = {
				value = 25
				add = ai_greed
				min = 0
			}
		}
	}
	if = { # Ideology supports military rule
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_military_power } }
		if = {
			limit = { character_is_military_trigger = yes }
			add = {
				value = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = {
				OR = {
					character_is_commercial_trigger = yes
					character_is_technocratic_trigger = yes
					character_is_theocratic_trigger = yes
				}
			}
			subtract = {
				value = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology supports commercial rule
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_commercial_power } }
		if = {
			limit = { character_is_commercial_trigger = yes }
			add = {
				value = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = {
				OR = {
					character_is_military_trigger = yes
					character_is_technocratic_trigger = yes
					character_is_theocratic_trigger = yes
				}
			}
			subtract = {
				value = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology supports technocracy
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_technocratic_power } }
		if = {
			limit = { character_is_technocratic_trigger = yes }
			add = {
				value = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = {
				OR = {
					character_is_military_trigger = yes
					character_is_commercial_trigger = yes
					character_is_theocratic_trigger = yes
				}
			}
			subtract = {
				value = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology supports theocracy
		limit = { scope:target_rite = { rite_has_doctrine = doctrine_theocratic_power } }
		if = {
			limit = { character_is_theocratic_trigger = yes }
			add = {
				value = ai_greed
				min = 0
			}
		}
		else_if = {
			limit = {
				OR = {
					character_is_military_trigger = yes
					character_is_commercial_trigger = yes
					character_is_technocratic_trigger = yes
				}
			}
			subtract = {
				value = ai_greed
				min = 0
			}
		}
	}
	if = { # Ideology restricts ruler's authority
		limit = {
			scope:target_rite = {
				OR = {
					rite_has_doctrine = doctrine_separation_of_powers
					rite_has_doctrine = doctrine_parliamentarianism
					rite_has_doctrine = doctrine_anarchism
				}
			}
		}
		if = {
			limit = { is_ruler = yes }
			subtract = {
				value = ai_greed
				min = 0
			}
		}
		if = {
			limit = { has_variable = legislature }
			add = {
				value = ai_greed
				min = 0
			}
		}
	}
	else_if = { # Ideology does not restrict ruler's authority
		limit = {
			scope:target_rite = {
				OR = {
					rite_has_doctrine = doctrine_despotism
					rite_has_doctrine = doctrine_dictatorship
				}
			}
		}
		if = {
			limit = { is_ruler = yes }
			add = {
				value = ai_greed
				min = 0
			}
		}
		else = {
			subtract = {
				value = ai_greed
				min = 0
			}
		}
	}
	# Personality Effects
	if = {
		limit = { scope:target_rite = { rite_authoritarianism > 0 } }
		add = ai_authoritarianism
	}
	else_if = {
		limit = { scope:target_rite = { rite_authoritarianism < 0 } }
		subtract = ai_authoritarianism
	}
	if = {
		limit = { scope:target_rite = { rite_hawkishness > 0 } }
		add = ai_hawkishness
	}
	else_if = {
		limit = { scope:target_rite = { rite_hawkishness < 0 } }
		subtract = ai_hawkishness
	}
	if = {
		limit = { scope:target_rite = { rite_intellectualism > 0 } }
		add = ai_intellectualism
	}
	else_if = {
		limit = { scope:target_rite = { rite_intellectualism < 0 } }
		subtract = ai_intellectualism
	}
	if = {
		limit = { scope:target_rite = { rite_interventionism > 0 } }
		add = ai_interventionism
	}
	else_if = {
		limit = { scope:target_rite = { rite_interventionism < 0 } }
		subtract = ai_interventionism
	}
	if = {
		limit = { scope:target_rite = { rite_lawfulness > 0 } }
		add = ai_lawfulness
	}
	else_if = {
		limit = { scope:target_rite = { rite_lawfulness < 0 } }
		subtract = ai_lawfulness
	}
	if = {
		limit = { scope:target_rite = { rite_mysticism > 0 } }
		add = ai_mysticism
	}
	else_if = {
		limit = { scope:target_rite = { rite_mysticism < 0 } }
		subtract = ai_mysticism
	}
	if = {
		limit = { scope:target_rite = { rite_parochialism > 0 } }
		add = ai_parochialism
	}
	else_if = {
		limit = { scope:target_rite = { rite_parochialism < 0 } }
		subtract = ai_parochialism
	}
}
