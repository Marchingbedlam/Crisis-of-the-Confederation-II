### IMPLANT INSTALLATION DESIRABILITY
install_target_implant_score = {
	# Do we have an existing implant to compare it to?
	if = {
		limit = { scope:replace = yes }
		every_equipped_character_artifact = {
			limit = { artifact_slot_type = scope:target_implant.artifact_slot_type }
			save_temporary_scope_as = existing_implant
			min = install_target_implant_by_type_score
		}
	}
	else = { add = install_target_implant_by_type_score }
}
install_target_implant_by_type_score = {
	# Type-specific scores
	if = {
		limit = { scope:target_implant.artifact_type = neural_implant }
		add = install_neural_implant_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = dermal_implant }
		add = install_dermal_implant_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = internal_implant }
		add = install_internal_implant_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = monocular_implant }
		add = install_monocular_implant_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = binocular_implant }
		add = install_binocular_implant_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = artificial_arm }
		add = install_artificial_arm_score
	}
	else_if = {
		limit = { scope:target_implant.artifact_type = artificial_leg }
		add = install_artificial_leg_score
	}
}

### IMPLANT DESIRIBILITY
neural_implant_self_score = {
	if = {
		limit = { has_variable = target_diplomacy_gain }
		min = var:target_diplomacy_gain
		min = var:target_martial_gain
		min = var:target_stewardship_gain
		min = var:target_intrigue_gain
		min = var:target_learning_gain
		min = var:target_personal_prowess_gain
		min = var:target_piloting_gain
	}
	if = {
		limit = { has_personal_tenet_flag = wants_implants }
		multiply = 2
	}
	else_if = {
		limit = { has_personal_tenet_flag = abstains_from_cybernetics }
		divide = 2
	}
}
neural_implant_other_score = {
	value = 0
	if = {
		limit = { ai_rationality > ai_greed }
		every_councillor = {
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
	}
	if = {
		limit = { ai_sociability > ai_greed }
		every_close_family_member = {
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
		every_relation = {
			type = friend
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
	}
}
dermal_implant_self_score = {
	# Reasons to want a dermal implant
	if = { # Our face is pretty screwed up
		limit = { has_trait = disfigured }
		add = 5
	}
	if = { # We have a skin disease and the implant would help
		limit = { any_character_trait = { has_trait_flag = skin_disease } }
		add = {
			value = 10
			subtract = health
			min = 0
		}
	}
	if = { # We just think implants are cool
		limit = { has_personal_tenet_flag = wants_implants }
		add = 5
	}
	# Ideological factors
	if = {
		limit = { ai_zeal >= 0 }
		switch = {
			trigger = rite.rite_has_parameter
			supports_cybernetics = { multiply = 2 }
			hates_cybernetics = { divide = 2 }
		}
	}
	if = {
		limit = { has_personal_tenet_flag = abstains_from_cybernetics }
		multiply = 0
	}
}
