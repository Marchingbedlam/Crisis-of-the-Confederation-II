neural_implant_self_score = {
	if = {
		limit = { has_variable = target_diplomacy_gain }
		min = var:target_diplomacy_gain
		min = var:target_martial_gain
		min = var:target_stewardship_gain
		min = var:target_intrigue_gain
		min = var:target_learning_gain
		min = var:target_personal_prowess_gain
		min = var:target_piloting_gain
	}
	if = {
		limit = { has_personal_tenet_flag = wants_implants }
		multiply = 2
	}
	else_if = {
		limit = { has_personal_tenet_flag = abstains_from_cybernetics }
		divide = 2
	}
}
neural_implant_other_score = {
	value = 0
	if = {
		limit = { ai_rationality > ai_greed }
		every_councillor = {
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
	}
	if = {
		limit = { ai_sociability > ai_greed }
		every_close_family_member = {
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
		every_relation = {
			type = friend
			limit = { might_want_implant_gift = { TYPE = neural_implant } }
			min = neural_implant_self_score
		}
	}
}
dermal_implant_self_score = {
	if = {
		limit = { has_trait = disfigured }
		add = 5
	}
	if = {
		limit = { any_character_trait = { has_trait_flag = skin_disease } }
		add = {
			value = 10
			subtract = health
			min = 0
		}
	}
	if = {
		limit = { has_personal_tenet_flag = wants_implants }
		add = 5
	}
	if = {
		limit = { ai_zeal >= 0 }
		switch = {
			trigger = rite.rite_has_parameter
			supports_cybernetics = { multiply = 2 }
			hates_cybernetics = { divide = 2 }
		}
	}
	if = {
		limit = { has_personal_tenet_flag = abstains_from_cybernetics }
		multiply = 0
	}
}
