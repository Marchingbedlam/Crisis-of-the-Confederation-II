	every_courtier = { add = 4 }

		limit = { domicile ?= { num_domicile_buildings >= 4 } }
		add = 30
	}
	if = {
		limit = { domicile ?= { num_domicile_buildings >= 8 } }
		add = 30
	}

		add = 30
