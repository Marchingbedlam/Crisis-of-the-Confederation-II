# How scared is this character of another character, in a personal way?
ai_personal_fear_of = {
	value = 0
	save_temporary_scope_as = temp_fearer
	# We know they're currently trying to kill us
	if = {
		limit = {
			scope:fearee = {
				any_scheme = {
					type = murder
					target_character = scope:temp_fearer
					is_scheme_exposed = yes
				}
			}
			ai_boldness <= ai_rationality
		}
		add = {
			value = ai_rationality
			multiply = 5
			min = 0
		}
		subtract = {
			value = ai_boldness
			multiply = 5
			max = 0
		}
	}
	# We know they have previously attempted to kill us
	if = {
		limit = {
			any_known_secret = {
				secret_type = secret_murder_attempt
				secret_owner = scope:fearee
				secret_target = scope:temp_fearer
			}
			ai_boldness <= ai_rationality
		}
		add = {
			value = ai_rationality
			multiply = 3
			min = 0
		}
		subtract = {
			value = ai_boldness
			multiply = 3
			max = 0
		}
	}
	# They're a known tyrant
	if = {
		limit = { scope:fearee.tyranny >= ai_boldness }
		add = scope:fearee.tyranny
	}
	# We were bullied by them
	if = {
		limit = {
			has_relation_bully = scope:fearee
			ai_boldness < 50
		}
		add = 10
	}
	# We have good reason to believe they're no threat to us
	if = {
		limit = {
			OR = {
				# They love us!
				reverse_opinion = { target = scope:fearee value >= 50 }
			}
			# We think about this stuff rationally
			ai_rationality >= ai_vengefulness
			# Sanity check, we should absolutely be afraid of anyone we KNOW is out to get us
			NOT = {
				scope:fearee = {
					any_scheme = {
						is_hostile = yes
						is_scheme_exposed = yes
						target_character = scope:temp_fearer
					}
				}
			}
		}
		multiply = 0
	}
	# They're harmless
	if = {
		limit = {
			scope:fearee = {
				OR = {
					is_alive = no
					is_incapable = yes
					is_imprisoned = yes
				}
			}
		}
		multiply = 0
	}
}
