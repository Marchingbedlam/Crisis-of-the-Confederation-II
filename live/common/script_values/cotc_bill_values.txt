# How much do you approve of the bill being debated?
opinion_of_bill = {
	value = 0
	
	add = support_bill_score
	subtract = block_bill_score
	
	if = {
		limit = { scope:activity = { has_activity_type = budget_bill_activity } }
		if = {
			limit = { scope:activity.proposed_budget_deficit > 0 }
			subtract = balance_budget_score
		}
		else = { add = balance_budget_score }
		if = {
			limit = { scope:activity.var:proposed_military_budget >= scope:activity.ruler.current_military_budget }
			add = military_spending_score
		}
		else = { subtract = military_spending_score }
		if = {
			limit = { scope:activity.var:proposed_ministry_budget >= scope:activity.ruler.current_ministry_budget }
			add = ministry_spending_score
		}
		else = { subtract = ministry_spending_score }
		if = {
			limit = { scope:activity.var:proposed_salary_budget >= scope:activity.ruler.current_salary_budget }
			add = salary_spending_score
		}
		else = { subtract = salary_spending_score }
	}
}
support_bill_score = {
	value = 0
	add = "opinion(scope:host)"
	add = {
		value = ai_authoritarianism
		if = {
			limit = { opinion = { target = scope:host value > 0 } }
			min = 0
		}
		else = { max = 0 }
	}
	subtract = {
		value = ai_rationality
		min = 0
	}
}
block_bill_score = {
	value = 0
	subtract = "opinion(scope:host)"
	if = {
		limit = { opinion = { target = scope:host value < 0 } }
		add = ai_vengefulness
	}
	subtract = {
		value = ai_rationality
		min = 0
	}
}
# How hard a "sell" is this bill? (for diplomacy challenge purposes)
bill_sell_difficulty = {
	value = 10
	add = bill_unreasonableness
	subtract = bill_necessity
	min = 10
}
# How hard a "target" is this bill?
bill_target_difficulty = {
	value = 10
	add = bill_necessity
	subtract = bill_unreasonableness
	min = 10
}
bill_necessity = {
	value = 0
	if = {
		limit = { has_activity_type = budget_bill_activity }
		if = {
			limit = {
				proposed_budget_deficit < 0
				scope:ruler.current_total_budget > 100
			}
			add = {
				value = scope:ruler.current_total_budget
				subtract = proposed_total_budget
				divide = 5
			}
		}
		if = {
			limit = {
				scope:ruler = { is_at_war = yes }
				var:proposed_military_budget > scope:ruler.current_military_budget
			}
			add = {
				value = scope:ruler.current_military_budget
				subtract = var:proposed_military_budget
				divide = 5
			}
		}
		if = {
			limit = {
				proposed_total_budget < scope:ruler.current_total_budget
				proposed_budget_deficit < 0
				scope:ruler.debt_level >= 0
			}
			add = {
				value = scope:ruler.debt_level
				add = 1
				multiply = 3
			}
		}
	}
}
bill_unreasonableness = {
	value = 0
	if = {
		limit = { has_activity_type = budget_bill_activity }
		if = {
			limit = { proposed_budget_deficit > 0 }
			add = {
				value = proposed_budget_deficit
				divide = 2
				ceiling = yes
			}
		}
		if = {
			limit = {
				proposed_total_budget >= scope:ruler.current_total_budget
				scope:ruler.debt_level >= 0
			}
			add = {
				value = scope:ruler.debt_level
				add = 1
				multiply = 3
			}
		}
		if = {
			limit = { var:proposed_military_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_military_budget
				if = {
					limit = { scope:ruler = { is_at_war = yes } }
					multiply = 2
				}
			}
		}
		if = {
			limit = { var:proposed_ministry_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_military_budget
				divide = 3
				ceiling = yes
			}
		}
		if = {
			limit = { var:proposed_salary_budget <= 10 }
			add = {
				value = 15
				subtract = var:proposed_salary_budget
				divide = 2
				ceiling = yes
			}
		}
	}
}
