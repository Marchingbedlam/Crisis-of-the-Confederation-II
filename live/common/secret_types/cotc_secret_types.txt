secret_official_bribe = {
	category = civil

	is_valid = {
		secret_official_bribe_is_valid_trigger = { OWNER = scope:secret_owner TARGET = scope:secret_target }
	}

	is_shunned = {
		NOT = { secret_official_bribe_is_criminal_trigger = { OWNER = scope:secret_owner } }
	}

	is_criminal = {
		secret_official_bribe_is_criminal_trigger = { OWNER = scope:secret_owner }
	}

	on_expose = {
		save_scope_as = secret
	}
	
	on_owner_death = { remove_secret = yes }
}

secret_researched_innovation = {
	category = technology
	
	is_valid = { always = yes }
	is_shunned = { secret_researched_innovation_is_shunned_trigger = { OWNER = scope:secret_owner } }
	is_criminal = { always = no }
	on_expose = { save_scope_as = secret }
	on_owner_death = { remove_secret = yes }
}
