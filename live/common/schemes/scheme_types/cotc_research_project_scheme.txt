research_project = {
	skill = learning
	category = contract # So that it doesn't interfere with other schemes
	target_type = nothing
	is_secret = no
	
	allow = {
		trigger_if = {
			limit = { NOT = { has_character_flag = inheriting_research_project } }
			highest_held_title_tier >= tier_county
			exists = capital_county
		}
		trigger_else = { always = yes }
	}
	valid = {
		scope:owner = {
			trigger_if = {
				limit = { NOT = { has_character_flag = inheriting_research_project } }
				highest_held_title_tier >= tier_county
				exists = capital_county
			}
			trigger_else = { always = yes }
		}
		trigger_if = {
			limit = { scope:scheme = { has_variable = research_subject } }
			scope:owner = { capital_county ?= { NOT = { county_has_tech = { TECH = scope:scheme.var:research_subject } } } }
		}
	}
	
	base_progress_goal = t1_base_phase_length_value # Research projects take a long time!
	speed_per_skill_point = -0.01 # Landed rulers are very busy, it should be impractical to progress the project without assistance
	spymaster_speed_per_skill_point = 0
	target_spymaster_speed_per_skill_point = 0
	speed_per_target_skill_point = 0
	base_maximum_success = 100
	minimum_success = 0
	base_success_chance = { base = 0 }
	
	odds_prediction = { value = learning multiply = 4 }
	
	agent_leave_threshold = -24
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_personal_general_suite_modifier = yes
		# Opinion of this particular project
		research_subject_approval_suite_modifier = yes
	}
	valid_agent = {
		is_valid_agent_standard_trigger = yes
		trigger_if = { # Must be in the same realm
			limit = { NOT = { top_liege = scope:owner.top_liege } }
			has_domicile = yes
			domicile ?= { domicile_location ?= { county.holder ?= { top_liege = scope:owner.top_liege } } }
		}
	}
	agent_groups_owner_perspective = { guests courtiers vassals nearby_domicile_owners }
	
	on_monthly = { trigger_event = { on_action = research_project_on_monthly } }
	on_hud_click = { trigger_event = { on_action = research_project_on_hud_click } }
	on_semiyearly = { trigger_event = { on_action = research_project_on_semiyearly } }
	on_phase_completed = { trigger_event = { on_action = research_project_on_phase_completed } }
	on_invalidated = { trigger_event = { on_action = research_project_on_invalidated } }
}
