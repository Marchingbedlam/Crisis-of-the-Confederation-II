cotc_politician_template = {
	age = { 20 60 }
	gender_female_chance = root_faith_dominant_gender_adjusted_female_chance #because chancellor gender is doctrine dependent
	random_traits = yes
	culture = root.culture
	faith = root.faith
	diplomacy = {
		min_guest_template_skill max_guest_template_skill
	}
	random_traits_list = {
		education_diplomacy_1 = { weight = { base = 35 } }
		education_diplomacy_2 = { weight = { base = 35 } }
		education_diplomacy_3 = { weight = { base = 20 } }
		education_diplomacy_4 = { weight = { base = 10 } }
	}
	# Motive for going into politics
	random_traits_list = {
		ambitious = {}
		greedy = {}
		just = {}
		zealous = {}
	}
	# Electoral strategy
	random_traits_list = {
		count = { 1 2 }
		honest = {}
		deceitful = {}
		brave = {}
		wrathful = {}
		diligent = {}
		gregarious = {}
	}
	dynasty = generate
	after_creation = {
		add_prestige_level = 1 # Politicians should always be at least a little prestigous
		if = {
			limit = { culture = culture:terran }
			initialize_character_effect = yes # Terrans should get a randomized Terran culture, everyone else is fine as-is
		}
		random_list = {
			200 = {
				# Character is of average weight, nothing happens	
			}
			25 = {
				change_current_weight = -25
			}
			25 = {
				change_current_weight = -75
			}
			25 = {
				change_current_weight = 25
			}
			25 = {
				change_current_weight = 75
			}
			5 = {
				change_current_weight = 100
			}
			5 = {
				change_current_weight = 200
			}
		}
		random = {
			chance = 1
			
			add_trait = lifestyle_poet
			
			modifier = {
				add = 15
				culture = { has_cultural_parameter = poet_trait_more_common }
			}
		}
		set_interesting_traits_and_modifiers_effect = yes
		# Generated politicians should never be content, why are you in politics in the first place?
		if = {
			limit = { has_trait = content }
			remove_trait = content
			add_trait = ambitious
		}
		# Similarly, generated politicians should never be lazy, this is a hard job
		if = {
			limit = { has_trait = lazy }
			remove_trait = lazy
			add_trait = diligent
		}
		# And it involves working with people!
		if = {
			limit = { has_trait = shy }
			remove_trait = shy
			add_trait = gregarious
		}
		if = {
			limit = { has_trait = lifestyle_reveler }
			add_trait_xp = {
				trait = lifestyle_reveler
				value = {
					integer_range = {
						min = small_lifestyle_random_xp_low
						max = small_lifestyle_random_xp_high
					}
				}
			}
		}
		add_random_tiered_trait_track_xp_effect = {
			TRAIT = tourney_participant
			TRACK = wit
			LEVEL_1 = yes
			LEVEL_3 = no
		}
	}
}
