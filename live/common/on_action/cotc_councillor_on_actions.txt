maintain_technology_chief_scientist_task_side_effects = {

		NOT = { has_variable = had_chief_scientist_task_side_effect }
		exists = cp:councillor_chief_scientist

		1000 = 0

	}
}
supervise_research_monthly_on_actions = {
	trigger = { exists = cp:councillor_chief_scientist }
	effect = {
		random_scheme = {
			limit = {
				OR = {
					scheme_type = research_project
					scheme_type = secret_research_project
				}
			}
			save_scope_as = scheme
		}
		trigger_event = { on_action = supervise_research_chief_scientist_task_side_effects }
	}
}
