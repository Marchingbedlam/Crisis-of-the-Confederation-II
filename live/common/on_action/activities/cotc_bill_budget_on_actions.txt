propose_random_budget_amendment_on_action = {
	random_on_action = {
		1 = propose_balanced_budget
		1 = propose_increased_military_spending
		1 = propose_increased_ministry_spending
		1 = propose_increased_salary_spending
	}
}
propose_balanced_budget = {
	weight_multiplier = {
		base = 1
		# Don't bother doing these calculations if the budget doesn't need to be balanced
		modifier = {
			factor = 0
			NOT = {
				scope:activity = {
					proposed_budget_deficit > 0
					any_attending_character = {
						is_in_guest_subset = { name = bill_opponents }
						OR = {
							has_activity_intent = balance_budget_intent
							has_activity_intent = balance_budget_maintain_military_intent
							has_activity_intent = balance_budget_maintain_ministry_intent
							has_activity_intent = balance_budget_maintain_salary_intent
						}
						OR = {
							NOT = { has_variable = vote_bargained_for }
							var:vote_strength < ai_rationality
						}
					}
				}
			}
		}
	}
	effect = {
		# Let's calculate some ways the budget could be balanced!
		# We can fix this by reducing just one budget
		if = {
			limit = { scope:activity.var:proposed_military_budget >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_military_budget value = scope:activity.var:proposed_military_budget }
			change_variable = { name = amended_military_budget subtract = scope:activity.proposed_budget_deficit }
		}
		if = {
			limit = { scope:activity.var:proposed_ministry_budget >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_ministry_budget value = scope:activity.var:proposed_military_budget }
			change_variable = { name = amended_ministry_budget subtract = scope:activity.proposed_budget_deficit }
		}
		if = {
			limit = { scope:activity.var:proposed_salary_budget >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_salary_budget value = scope:activity.var:proposed_military_budget }
			change_variable = { name = amended_salary_budget subtract = scope:activity.proposed_budget_deficit }
		}
		# We need to reduce two different budgets
		if = {
			limit = { scope:activity.combined_military_ministry_proposal >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_total_budget value = scope:activity.proposed_total_budget }
			set_variable = { name = amended_military_budget_milmin value = scope:activity.var:proposed_military_budget }
			set_variable = { name = amended_ministry_budget_milmin value = scope:activity.var:proposed_ministry_budget }
			while = {
				limit = { var:amended_total_budget > 100 }
				if = {
					limit = {
						var:amended_ministry_budget_milmin > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_ministry_budget_milmin subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				if = {
					limit = {
						var:amended_military_budget_milmin > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_military_budget_milmin subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				# Escape clause to prevent an infinite loop
				if = {
					limit = {
						var:amended_ministry_budget_milmin <= 0
						var:amended_military_budget_milmin <= 0
						var:amended_total_budget > 100
					}
					set_variable = { name = amended_total_budget value = 100 }
				}
			}
		}
		if = {
			limit = { scope:activity.combined_military_salary_proposal >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_total_budget value = scope:activity.proposed_total_budget }
			set_variable = { name = amended_military_budget_milsal value = scope:activity.var:proposed_military_budget }
			set_variable = { name = amended_salary_budget_milsal value = scope:activity.var:proposed_salary_budget }
			while = {
				limit = { var:amended_total_budget > 100 }
				if = {
					limit = {
						var:amended_salary_budget_milsal > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_salary_budget_milsal subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				if = {
					limit = {
						var:amended_military_budget_milsal > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_military_budget_milsal subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				# Escape clause to prevent an infinite loop
				if = {
					limit = {
						var:amended_salary_budget_milsal <= 0
						var:amended_military_budget_milsal <= 0
						var:amended_total_budget > 100
					}
					set_variable = { name = amended_total_budget value = 100 }
				}
			}
		}
		if = {
			limit = { scope:activity.combined_ministry_salary_proposal >= scope:activity.proposed_budget_deficit }
			set_variable = { name = amended_total_budget value = scope:activity.proposed_total_budget }
			set_variable = { name = amended_ministry_budget_minsal value = scope:activity.var:proposed_ministry_budget }
			set_variable = { name = amended_salary_budget_minsal value = scope:activity.var:proposed_salary_budget }
			while = {
				limit = { var:amended_total_budget > 100 }
				if = {
					limit = {
						var:amended_salary_budget_minsal > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_salary_budget_minsal subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				if = {
					limit = {
						var:amended_ministry_budget_minsal > 0
						var:amended_total_budget > 100
					}
					change_variable = { name = amended_ministry_budget_minsal subtract = 5 }
					change_variable = { name = amended_total_budget subtract = 5 }
				}
				# Escape clause to prevent an infinite loop
				if = {
					limit = {
						var:amended_salary_budget_minsal <= 0
						var:amended_ministry_budget_minsal <= 0
						var:amended_total_budget > 100
					}
					set_variable = { name = amended_total_budget value = 100 }
				}
			}
		}
		# Wow, you REALLY set those budgets too high...
		set_variable = { name = amended_total_budget value = scope:activity.proposed_total_budget }
		set_variable = { name = amended_military_budget_triple value = scope:activity.var:proposed_military_budget }
		set_variable = { name = amended_ministry_budget_triple value = scope:activity.var:proposed_ministry_budget }
		set_variable = { name = amended_salary_budget_triple value = scope:activity.var:proposed_salary_budget }
		while = {
			limit = { var:amended_total_budget > 100 }
			if = {
				limit = {
					var:amended_total_budget > 100
					var:amended_ministry_budget_triple > 0
				}
				change_variable = { name = amended_ministry_budget_triple subtract = 5 }
				change_variable = { name = amended_total_budget subtract = 5 }
			}
			if = {
				limit = {
					var:amended_total_budget > 100
					var:amended_salary_budget_triple > 0
				}
				change_variable = { name = amended_salary_budget_triple subtract = 5 }
				change_variable = { name = amended_total_budget subtract = 5 }
			}
			if = {
				limit = {
					var:amended_total_budget > 100
					var:amended_military_budget_triple > 0
				}
				change_variable = { name = amended_military_budget_triple subtract = 5 }
				change_variable = { name = amended_total_budget subtract = 5 }
			}
			# Escape clause to prevent an infinite loop
			if = {
				limit = {
					var:amended_military_budget_triple <= 0
					var:amended_ministry_budget_triple <= 0
					var:amended_salary_budget_triple <= 0
					var:amended_total_budget > 100
				}
				set_variable = { name = amended_total_budget value = 100 }
			}
		}
		remove_variable = amended_total_budget
		trigger_event = { on_action = propose_random_balanced_budget }
	}
}
