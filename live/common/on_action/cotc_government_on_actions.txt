# ROOT ON_ACTIONS #

global_government_setup_on_action = {
	effect = {
		every_ruler = { trigger_event = { on_action = government_setup_on_action } }
	}
}
government_setup_on_action = {
	on_actions = {
		# Federations
		initialize_starting_federation_on_action
		
		# Legislatures
		initialize_starting_legislature_on_action
		
		# Laws
		province_taxes_on_action
		
		# Planning
		plan_budget_on_action
	}
}
government_update_on_action = {
	on_actions = {		
		# Legislatures
		initialize_new_legislature_on_action
		
		abolish_legislature_on_action
		
		reattach_legislature_on_action
		merge_legislatures_on_action
		
		# Laws
		province_taxes_on_action
		
		# Planning
		plan_budget_on_action
	}
}
government_vassal_change_on_action = {
	effect = {
		trigger_event = { on_action = government_update_on_action }
		trigger_event = { on_action = ridings_vassal_change_on_action }
		liege ?= {
			if = {
				limit = { NOT = { THIS = PREV } }
				trigger_event = { on_action = government_update_on_action }
			}
		}
	}
}
government_destruction_on_action = {
}
party_change_on_action = {
	on_actions = { update_legislator_party_on_action }
}
on_legislator_death_on_action = {
	trigger = { has_variable = legislature }
	effect = {
		var:legislature = { save_temporary_scope_as = temp_legislature }
		remove_from_legislature_effect = yes
		scope:temp_legislature = { recalc_legislature_parties_effect = yes }
	}
}

initialize_starting_legislature_on_action = {
	trigger = {
		should_have_legislature_trigger = yes
		has_legislature_trigger = no
	}
	effect = {
		save_scope_as = ruler
		start_situation = {
			type = legislature
			save_scope_as = situation
			start_phase = legislature_in_session
		}
		every_held_title = {
			save_temporary_scope_as = temp_title
			scope:situation.situation_sub_region:legislature_title_region = {
				add_dejure_title_to_sub_region = scope:temp_title
			}
			clear_saved_scope = temp_title
		}
		primary_title = {
			save_scope_as = target_title
			set_variable = {
				name = legislature
				value = scope:situation
			}
		}
		scope:situation ?= {
			set_situation_center_province = scope:ruler.capital_province
			set_variable = {
				name = legislature_title
				value = scope:target_title
			}
			set_variable = { name = months_in_session value = 0 }
			generate_riding_list_effect = yes
			# Elect starting legislators
			every_in_list = {
				variable = ridings
				select_initial_legislator_effect = { LEGISLATURE = scope:situation }
			}
			every_in_list = {
				variable = legislators
				trigger_event = { on_action = plan_legislative_campaign_on_action }
			}
			recalc_legislature_parties_effect = yes
			every_participant_group = {
				every_situation_group_participant = {
					recalculate_participant_group = scope:situation
				}
			}
		}
		trigger_event = { on_action = set_initial_prime_minister }
	}
}
initialize_new_legislature_on_action = {
	trigger = {
		should_have_legislature_trigger = yes
		has_legislature_trigger = no
	}
	effect = {
		save_scope_as = ruler
		start_situation = {
			type = legislature
			save_scope_as = situation
			start_phase = legislature_election
		}
		every_held_title = {
			limit = { tier = prev.primary_title.tier }
			save_temporary_scope_as = temp_title
			scope:situation.situation_sub_region:legislature_title_region = {
				add_dejure_title_to_sub_region = scope:temp_title
			}
			clear_saved_scope = temp_title
		}
		primary_title = {
			save_scope_as = target_title
			set_variable = {
				name = legislature
				value = scope:situation
			}
		}
		scope:situation ?= {
			set_situation_center_province = scope:ruler.capital_province
			set_variable = {
				name = legislature_title
				value = scope:target_title
			}
			generate_riding_list_effect = yes
		}
		trigger_event = { on_action = set_initial_prime_minister }
	}
}
quarterly_progress_legislature_on_action = {
	trigger = {
		has_legislature_trigger = yes
		primary_title.var:legislature ?= { situation_current_phase = legislature_in_session }
	}
	effect = {
		trigger_event = { on_action = monthly_progress_legislature_on_action }
		trigger_event = { on_action = monthly_progress_legislature_on_action months = 1 }
		trigger_event = { on_action = monthly_progress_legislature_on_action months = 2 }
	}
}
monthly_progress_legislature_on_action = {
	trigger = {
		has_legislature_trigger = yes
		primary_title.var:legislature ?= { situation_current_phase = legislature_in_session }
	}
	effect = {
		# Advance the legislature's term
		primary_title.var:legislature ?= { change_variable = { name = months_in_session add = 1 } }
		# If we've hit the term limit, call an election
		if = {
			limit = {
				trigger_if = {
					limit = { has_realm_law = two_year_terms_law }
					primary_title.var:legislature ?= { var:months_in_session >= 21 }
				}
				trigger_else_if = {
					limit = { has_realm_law = four_year_terms_law }
					primary_title.var:legislature ?= { var:months_in_session >= 45 }
				}
				trigger_else_if = {
					limit = { has_realm_law = six_year_terms_law }
					primary_title.var:legislature ?= { var:months_in_session >= 69 }
				}
				trigger_else_if = {
					limit = { has_realm_law = eight_year_terms_law }
					primary_title.var:legislators ?= { var:months_in_session >= 93 }
				}
				trigger_else_if = {
					limit = { has_realm_law = ten_year_terms_law }
					primary_title.var:legislators ?= { var:months_in_session >= 117 }
				}
				trigger_else = { always = no }
			}
			primary_title.var:legislature ?= { trigger_situation_catalyst = catalyst_term_expired }
		}
	}
}
abolish_legislature_on_action = {
	# We're a despot or dictator and don't need a legislature, or are too low-ranking to have one
	trigger = {
		has_legislature_trigger = yes
		should_have_legislature_trigger = no
	}
	events = { legislature_events.100 }
}
reattach_legislature_on_action = {
	# We should and do have a legislature but it's attached to the wrong title
	trigger = {
		should_have_legislature_trigger = yes
		primary_title = { NOT = { has_variable = legislature } }
		any_held_title = { has_variable = legislature }
	}
	events = { legislature_events.200 }
}
merge_legislatures_on_action = {
	# We have more than one legislature
	trigger = {
		has_legislature_trigger = yes
		primary_title = { has_variable = legislature }
		any_held_title = {
			has_variable = legislature
			count >= 2
		}
	}
	events = { legislature_events.300 }
}
ridings_vassal_change_on_action = {
	# We were represented in a legislature, but now we aren't!
	trigger = {
		exists = scope:old_liege # We were not independent
		scope:old_liege ?= {
			add_to_temporary_list = old_lieges
			any_liege_or_above ?= {
				count = all
				add_to_temporary_list = old_lieges
			}
		}
		any_in_list = {
			list = old_lieges
			has_legislature_trigger = yes
			save_temporary_scope_as = temp_old_liege
			primary_title = {
				any_in_list = {
					variable = ridings
					save_temporary_scope_as = temp_riding
					NOT = {
						scope:temp_old_liege = {
							OR = {
								any_sub_realm_county = { this = scope:temp_riding }
								any_sub_realm_duchy = { this = scope:temp_riding }
							}
						}
					}
				}
			}
		}
	}
	effect = {
		# Remove us from the old ridings list and also remove any of our legislators
		every_sub_realm_county = { add_to_list = ridings_to_remove }
		every_sub_realm_duchy = { add_to_list = ridings_to_remove }
		
		# Build a list of all the lieges we no longer have
		scope:old_liege = {
			add_to_list = old_lieges
			every_liege_or_above = { add_to_list = old_lieges }
		}
		
		every_in_list = {
			list = old_lieges
			if = {
				limit = { has_legislature_trigger = yes }
				primary_title = {
					save_temporary_scope_as = temp_old_legislature
					var:legislature ?= {
						save_temporary_scope_as = temp_legislature
						every_in_list = {
							variable = ridings
							limit = { is_in_list = ridings_to_remove }
							
							# End all ongoing elections
							switch = {
								trigger = scope:temp_old_legislature.tier
								tier_county = { clear_variable_list = county_candidates }
								tier_duchy = { clear_variable_list = duchy_candidates }
								tier_kingdom = { clear_variable_list = kingdom_candidates }
								tier_empire = { clear_variable_list = empire_candidates }
								tier_hegemony = { clear_variable_list = hegemony_candidates }
							}
							
							save_temporary_scope_as = temp_riding
							scope:temp_legislature = {
								# Remove us from the ridings list
								remove_list_variable = { name = ridings target = scope:temp_riding }
								# And remove our senator if we have one
								every_in_list = {
									variable = legislators
									limit = { var:riding = scope:temp_riding }
									remove_from_legislature_effect = yes
								}
								# And then recalc parties
								recalc_legislature_parties_effect = yes
							}
						}
					}
				}
			}
		}
	}
}
update_legislator_party_on_action = {
	trigger = { has_variable = legislature }
	effect = { var:legislature = { recalc_legislature_parties_effect = yes } }
}

set_initial_prime_minister = {
	trigger = {
		government_has_flag = government_is_parliamentary
		has_legislature_trigger = yes
		NOT = { has_diarchy_type = cotc_premiership }
		any_vassal = { count > 0 }
	}
	effect = {
		try_start_diarchy = cotc_premiership
		ordered_vassal = {
			order_by = prime_minister_score
			ROOT = { set_diarch = PREV }
		}
	}
}

plan_legislative_campaign_on_action = {
	trigger = {
		is_ai = yes
		eligible_legislator_candidate_trigger = yes
		NOT = { any_scheme = { scheme_type = legislative_election_campaign } }
		ai_political_ambition > 0
		any_liege_or_above = {
			has_legislature_trigger = yes
			primary_title = {
				save_temporary_scope_as = temp_title
				ROOT = {
					# Must live in the sub-realm
					character_lives_in_de_facto_title = { TITLE = scope:temp_title }
					# Must be able to 
					OR = {
						influence >= { value = minor_influence_value multiply = scope:temp_title.title_basic_cost_multiplier }
						gold >= { value = minor_gold_value multiply = scope:temp_title.title_basic_cost_multiplier }
						AND = {
							has_variable = legislature
							var:legislature ?= { var:legislature_title ?= { this = scope:temp_title } }
						}
					}
				}
			}
		}
	}
	effect = {
		# Consider running for each legislature, in descending order of importance
		ordered_liege_or_above = {
			limit = {
				has_legislature_trigger = yes
				primary_title = {
					save_temporary_scope_as = temp_title
					ROOT = {
						character_lives_in_de_facto_title = { TITLE = scope:temp_title }
						OR = {
							influence >= { value = minor_influence_value multiply = scope:temp_title.title_basic_cost_multiplier }
							gold >= { value = minor_gold_value multiply = scope:temp_title.title_basic_cost_multiplier }
							AND = {
								has_variable = legislature
								var:legislature ?= { var:legislature_title ?= { this = scope:temp_title } }
							}
						}
					}
				}
			}
			order_by = { value = highest_held_title_tier multiply = -1 }
			primary_title = {
				var:legislature ?= {
					save_scope_as = legislature
					# If ROOT is an incumbent, check their existing riding
					if = {
						limit = {
							ROOT = {
								has_variable = var:riding
								var:riding ?= { scope:legislature = { is_target_in_variable_list = { name = ridings target = PREV } } }
							}
						}
						ROOT = {
							var:riding = { save_scope_as = riding }
							trigger_event = cotc_legislative_campaign_events.50
						}
					}
					# Otherwise, check the riding they live in
					else = {
						random_in_list = {
							variable = ridings
							limit = {
								save_temporary_scope_as = temp_riding
								ROOT = { character_lives_in_de_jure_title = { TITLE = scope:temp_riding } }
							}
							save_scope_as = riding
							ROOT = { trigger_event = cotc_legislative_campaign_events.50 }
						}
					}
				}
			}
		}
	}
}

		set_variable = { name = planned_budget_surplus value = target_budget_surplus }

			limit = { planned_total_budget < 100 }

							add = {
								value = 0
								subtract = ai_interventionism

							factor = 0

						change_variable = { name = planned_ministry_budget subtract = 5 }
					}

							add = {
								value = 0
								subtract = ai_intellectualism
							}
						}
						modifier = {
							factor = 0
							var:planned_salary_budget <= 0

		}
	}
