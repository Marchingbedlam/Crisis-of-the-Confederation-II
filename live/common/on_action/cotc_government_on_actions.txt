on_vassal_changed = {
	effect = {
		trigger_event = { on_action = government_update_on_action }
		liege ?= {
			if = {
				limit = { NOT = { THIS = PREV } }
				trigger_event = { on_action = government_update_on_action }
			}
		}
	}
}

	on_actions = {
		progress_legislature_on_action
		plan_budget_on_action
	}
}

global_government_setup_on_action = {
	effect = {
		every_ruler = { trigger_event = { on_action = government_setup_on_action } }
	}
}
government_setup_on_action = {
	on_actions = {
		# Legislatures
		initialize_starting_legislature_on_action
		
		# Laws
		province_taxes_on_action
		
		# Planning
		plan_budget_on_action
	}
}
government_update_on_action = {
	on_actions = {
		# Legislatures
		initialize_new_legislature_on_action
		abolish_legislature_on_action
		reattach_legislature_on_action
		merge_legislatures_on_action
		
		# Laws
		province_taxes_on_action
		
		# Planning
		plan_budget_on_action
	}
}

initialize_starting_legislature_on_action = {
	trigger = {
		should_have_legislature_trigger = yes
		has_legislature_trigger = no
	}
	effect = {
		save_scope_as = ruler
		debug_log_scopes = yes
		start_situation = {
			type = legislature
			save_scope_as = situation
			start_phase = legislature_in_session
		}
		every_held_title = {
			save_temporary_scope_as = temp_title
			scope:situation.situation_sub_region:legislature_title_region = {
				add_dejure_title_to_sub_region = scope:temp_title
			}
			clear_saved_scope = temp_title
		}
		primary_title = {
			save_scope_as = target_title
			set_variable = {
				name = legislature
				value = scope:situation
			}
		}
		scope:situation ?= {
			set_situation_center_province = scope:ruler.capital_province
			set_variable = {
				name = legislature_title
				value = scope:target_title
			}
			set_variable = { name = months_in_session value = 0 }
		}
		# Elect starting legislators
		every_sub_realm_county ?= { elect_new_legislator_effect = yes }
		scope:situation ?= {
			every_participant_group = {
				every_situation_group_participant = {
					recalculate_participant_group = scope:situation
				}
			}
		}
		trigger_event = { on_action = set_initial_prime_minister }
	}
}
initialize_new_legislature_on_action = {
	trigger = {
		should_have_legislature_trigger = yes
		has_legislature_trigger = no
	}
	effect = {
		save_scope_as = ruler
		start_situation = {
			type = legislature
			save_scope_as = situation
			start_phase = legislature_election
		}
		every_held_title = {
			limit = { tier = prev.primary_title.tier }
			save_temporary_scope_as = temp_title
			scope:situation.situation_sub_region:legislature_title_region = {
				add_dejure_title_to_sub_region = scope:temp_title
			}
			clear_saved_scope = temp_title
		}
		primary_title = {
			save_scope_as = target_title
			set_variable = {
				name = legislature
				value = scope:situation
			}
		}
		scope:situation ?= {
			set_situation_center_province = scope:ruler.capital_province
			set_variable = {
				name = legislature_title
				value = scope:target_title
			}
		}
		trigger_event = { on_action = set_initial_prime_minister }
	}
}
progress_legislature_on_action = {
	trigger = {
		has_legislature_trigger = yes
		primary_title.var:legislature ?= { situation_current_phase = legislature_in_session }
	}
	effect = { primary_title.var:legislature ?= { change_variable = { name = months_in_session add = 3 } } }
}
abolish_legislature_on_action = {
	# We're a despot or dictator and don't need a legislature, or are too low-ranking to have one
	trigger = {
		has_legislature_trigger = yes
		should_have_legislature_trigger = no
	}
	events = { legislature_events.100 }
}
reattach_legislature_on_action = {
	# We should and do have a legislature but it's attached to the wrong title
	trigger = {
		should_have_legislature_trigger = yes
		primary_title = { NOT = { has_variable = legislature } }
		any_held_title = { has_variable = legislature }
	}
	events = { legislature_events.200 }
}
merge_legislatures_on_action = {
	# We have more than one legislature
	trigger = {
		has_legislature_trigger = yes
		primary_title = { has_variable = legislature }
		any_held_title = {
			has_variable = legislature
			count >= 2
		}
	}
	events = { legislature_events.300 }
}

# PRIME MINISTERS #
###################
set_initial_prime_minister = {
	trigger = {
		government_has_flag = government_is_parliamentary
		has_legislature_trigger = yes
		NOT = { has_diarchy_type = cotc_premiership }
		any_vassal = { count > 0 }
	}
	effect = {
		try_start_diarchy = cotc_premiership
		ordered_vassal = {
			order_by = prime_minister_score
			ROOT = { set_diarch = PREV }
		}
	}
}

plan_budget_on_action = {
	trigger = {
		is_independent_ruler = yes
		should_have_legislature_trigger = yes
	}
	effect = {
		# Target budgets
		save_temporary_scope_as = budget_ruler
		set_variable = { name = planned_military_budget value = target_military_budget }
		set_variable = { name = planned_ministry_budget value = target_ministry_budget }
		set_variable = { name = planned_salary_budget value = target_salary_budget }
		set_variable = { name = planned_budget_surplus value = target_budget_surplus }
		while = {
			limit = { planned_total_budget < 100 }
			random_list = {
				200 = {
					modifier = { add = ai_hawkishness }
					change_variable = { name = planned_military_budget add = 5 }
				}
				200 = {
					modifier = { add = ai_interventionism }
					change_variable = { name = planned_ministry_budget add = 5 }
				}
				200 = {
					modifier = { add = ai_intellectualism }
					change_variable = { name = planned_salary_budget add = 5 }
				}
				200 = {
					modifier = { add = ai_greed }
					change_variable = { name = planned_budget_surplus add = 5 }
				}
			}
		}
		if = {
			limit = { planned_total_budget > 100 }
			set_variable = { name = counter value = planned_total_budget }
			change_variable = { name = counter subtract = 100 }
			while = {
				limit = { var:counter > 0 }
				change_variable = { name = counter subtract = 5 }
				random_list = {
					200 = {
						modifier = {
							add = {
								value = 0
								subtract = ai_hawkishness
							}
						}
						modifier = {
							factor = 0
							var:planned_military_budget <= 0
						}
						change_variable = { name = planned_military_budget subtract = 5 }
					}
					200 = {
						modifier = {
							add = {
								value = 0
								subtract = ai_interventionism
							}
						}
						modifier = {
							factor = 0
							var:planned_ministry_budget <= 0
						}
						change_variable = { name = planned_ministry_budget subtract = 5 }
					}
					200 = {
						modifier = {
							add = {
								value = 0
								subtract = ai_intellectualism
							}
						}
						modifier = {
							factor = 0
							var:planned_salary_budget <= 0
						}
						change_variable = { name = planned_salary_budget subtract = 5 }
					}
					200 = {
						modifier = {
							add = {
								value = 0
								subtract = ai_greed
							}
						}
						modifier = {
							factor = 0
							var:planned_budget_surplus <= 0
							ai_rationality >= high_positive_ai_value
						}
						modifier = {
							factor = 0
							var:planned_budget_surplus <= -10
							ai_rationality >= low_positive_ai_value
						}
						modifier = {
							factor = 0
							var:planned_budget_surplus <= -20
						}
						change_variable = { name = planned_budget_surplus subtract = 5 }
					}
				}
			}
			remove_variable = counter
		}
	}
}
