# root = the newborn
on_birth_child = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the new holder
on_title_gain = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the courtier
on_join_court = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# characters list is everyone travelling together
on_guest_arrived_from_pool = {
	effect = {
		every_in_list = {
			list = characters
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}
