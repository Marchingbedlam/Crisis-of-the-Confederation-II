##################################################
# COTC - Dynasty naming
##################################################

# root = the newborn
on_birth_child = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the new holder
on_title_gain = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the courtier
on_join_court = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# characters list is everyone travelling together
on_guest_arrived_from_pool = {
	effect = {
		every_in_list = {
			list = characters
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the dynasty, dynast = its founder.
# Fallback safety net as not all dynasty generation runs through cotc_misc_effects.
# The founder's own culture is passed, so a Terran dummy made in an Earth culture
# still yields a family name matching the given name
on_dynasty_created = {
	effect = {
		dynast ?= {
			cotc_assign_family_name_effect = { CULTURE = culture }
		}
	}
}

# Baronies with no holder get a generated holder with their dynasty named after the barony.
# They are not lowborn, so the other pass misses them.
on_game_start_after_lobby = {
	effect = {
		every_ruler = {
			limit = {
				highest_held_title_tier = tier_barony
				dynasty ?= { NOT = { cotc_is_canon_dynasty_trigger = yes } }
			}
			cotc_assign_family_name_effect = { CULTURE = culture }
		}
	}
}
