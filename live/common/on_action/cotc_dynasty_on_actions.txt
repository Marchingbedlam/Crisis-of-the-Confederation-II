##################################################
# COTC - Dynasty naming
##################################################

on_birth_child = { on_actions = { cotc_family_name_birth } }
on_title_gain = { on_actions = { cotc_family_name_title_gain } }
on_join_court = { on_actions = { cotc_family_name_join_court } }
on_guest_arrived_from_pool = { on_actions = { cotc_family_name_pool_arrival } }
on_dynasty_created = { on_actions = { cotc_family_name_dynasty_created } }
on_game_start_after_lobby = { on_actions = { cotc_family_name_game_start } }

##################################################
# No character reaches ennoblement lowborn, so no dynasty is founded from a title.
##################################################

# root = the newborn
cotc_family_name_birth = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the new holder. Last chance before a title can name the dynasty.
cotc_family_name_title_gain = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the character joining
cotc_family_name_join_court = {
	effect = {
		if = {
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# The characters list holds everyone who travelled together, root included.
cotc_family_name_pool_arrival = {
	effect = {
		every_in_list = {
			list = characters
			limit = { is_lowborn = yes }
			generate_family_name_effect = yes
		}
	}
}

# root = the dynasty, dynast = its founder.
cotc_family_name_dynasty_created = {
	effect = {
		dynast ?= {
			cotc_assign_family_name_effect = { CULTURE = culture }
		}
	}
}

# Baronies with no holder get a generated holder whose dynasty is named after the
# barony. They are not lowborn, so the lowborn pass misses them.
cotc_family_name_game_start = {
	effect = {
		every_ruler = {
			limit = {
				highest_held_title_tier = tier_barony
				dynasty ?= { NOT = { cotc_is_canon_dynasty_trigger = yes } }
			}
			cotc_assign_family_name_effect = { CULTURE = culture }
		}
	}
}
