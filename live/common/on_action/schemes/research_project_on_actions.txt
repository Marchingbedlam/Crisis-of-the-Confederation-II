# HOOKS TO CODE
on_death = { on_actions = { inherit_research_project_by_death_on_action } }
on_title_gain = { on_actions = { inherit_research_project_by_succession_on_action } }

# CORE PROJECT ON-ACTIONS
research_project_on_monthly = {
	effect = {
		scheme_owner = { trigger_event = { on_action = research_project_monthly_pulse days = { 1 30 } } }
	}
}
research_project_on_hud_click = {
	effect = {
		research_project_prep_effect = yes
	}
}
research_project_on_semiyearly = {
	effect = {
		scheme_owner = {
			if = {
				limit = { is_ai = yes }
				research_project_prep_effect = yes
			}
			trigger_event = { on_action = research_project_semiyearly_pulse days = { 1 180 } }
		}
	}
}
research_project_on_phase_completed = {
	effect = {
		# SUCCESS ROLL
		random = {
			chance = scope:scheme.scheme_success_chance
			save_scope_value_as = {
				name = scheme_successful
				value = yes
			}
		}
		### End success roll
		if = {
			limit = { exists = scope:scheme_successful }
			save_research_project_scopes_effect = yes
			scheme_owner = {
				if = {
					limit = { has_character_modifier = researching_modifier }
					remove_research_project_personal_involvement = yes
				}
				trigger_event = cotc_research_project_events.10
			}
		}
	}
}
research_project_on_invalidated = {
	effect = {
		scheme_owner = {
			if = {
				limit = { has_character_modifier = researching_modifier }
				remove_research_project_personal_involvement = yes
			}
			if = {
				limit = { has_character_flag = suppress_project_invalidation }
				remove_character_flag = suppress_project_invalidation
			}
			else = {
				scope:scheme = { save_research_project_scopes_effect = yes }
				trigger_event = { on_action = fail_research_project_on_action }
			}
		}
	}
}

# PULSE ON-ACTIONS
research_project_monthly_pulse = {
	effect = {
		# Track active contributors to the project
		scope:scheme = {
			every_scheme_agent_character = {
				limit = { NOT = { scope:scheme = { is_target_in_variable_list = { name = researchers target = PREV } } } }
				scope:scheme = { add_to_variable_list = { name = researchers target = PREV } }
			}
			every_in_list = {
				variable = researchers
				limit = { NOT = { scope:scheme = { scheme_is_character_agent = PREV } } }
				trigger_event = { on_action = research_project_researcher_purge months = 12 }
			}
		}
		# Guaranteed events
		scope:scheme = { every_scheme_agent_character = { trigger_event = cotc_research_project_events.200 } } # Reveal your knowledge of the subject if you want
	}
}
research_project_semiyearly_pulse = {
	effect = {
		# Check for setbacks
		trigger_event = { on_action = research_project_setback_roll days = { 1 180 } }
		
		# Random events
		random_list = { # Pick a random type of event to (maybe) fire
			10 = {
				modifier = { factor = { scope:scheme = { every_scheme_agent_character = { add = 1 } } } }
				scope:scheme = {
					random_scheme_agent_slot = {
						limit = {
							is_filled = yes
							slot_character ?= { NOT = { has_character_flag = research_project_event_cooldown } }
						}
						save_scope_as = first_researcher_slot
						slot_character = {
							save_scope_as = first_researcher
							trigger_event = { on_action = research_project_researcher_pulse days = { 1 180 } }
						}
					}
				}
			}
			10 = {
				trigger = { has_character_modifier = researching_modifier }
				trigger_event = { on_action = research_project_personal_research_on_action days = { 1 180 } }
			}
			40 = { }
		}
	}
}
research_project_researcher_pulse = {
	trigger = { exists = scope:scheme } # Check to make sure the project still exists
	random_events = {
		1 = cotc_research_project_events.210 # Likes a fellow researcher
		1 = cotc_research_project_events.220 # Dislikes a fellow researcher
		1 = cotc_research_project_events.230 # Lab accident
		1 = cotc_research_project_events.240 # Solo breakthrough
		1 = cotc_research_project_events.250 # Team breakthrough
		100 = 0
	}
}
research_project_personal_research_on_action = {
	trigger = {
		exists = scope:scheme
		has_character_modifier = researching_modifier
	}
	random_events = {
		1 = cotc_research_project_events.300 # Make a personal breakthrough
		1 = cotc_research_project_events.310 # Lab accident
		100 = 0
	}
}
# OTHER ON-ACTIONS
research_project_setback_roll = {
	trigger = { exists = scope:scheme }
	random_events = {
		1 = cotc_research_project_events.500
		500 = 0
	}
}
fail_research_project_on_action = {
	events = {
		cotc_research_project_events.20 # Project failed because you no longer have territory
		cotc_research_project_events.30 # Project failed because you got the tech some other way
	}
}
inherit_research_project_by_death_on_action = {
	trigger = { any_scheme = { scheme_type = research_project_public } }
	effect = {
		save_scope_as = old_project_patron
		random_scheme = {
			limit = { scheme_type = research_project_public }
			save_scope_as = old_scheme
		}
		if = {
			limit = { capital_county ?= { current_heir ?= { NOT = { any_scheme = { scheme_type = research_project_public } } } } }
			capital_county ?= {
				current_heir ?= {
					save_scope_as = owner
					add_character_flag = inheriting_research_project
					start_scheme = {
						type = research_project_public
						targets_nothing = yes
						save_scope_as = scheme
					}
					scope:scheme = {
						copy_research_project_data = { SCHEME = scope:old_scheme }
						set_variable = { name = owner value = scope:owner }
					}
					trigger_event = { id = cotc_research_project_events.50 days = 1 }
				}
			}
		}
	}
}
inherit_research_project_by_succession_on_action = {
	trigger = {
		OR = {
			scope:transfer_type = flag:inheritance
			scope:transfer_type = flag:abdication
			scope:transfer_type = flag:usurped
			scope:transfer_type = flag:election
		}
		NOT = { any_scheme = { scheme_type = research_project_public } }
		scope:previous_holder = {
			any_scheme = { scheme_type = research_project_public }
			NOT = { exists = capital_county }
		}
	}
	effect = {
		save_scope_as = owner
		scope:previous_holder = {
			save_scope_as = old_project_patron
			random_scheme = {
				limit = { scheme_type = research_project_public }
				save_scope_as = old_scheme
			}
			add_character_flag = suppress_project_invalidation
		}
		add_character_flag = inheriting_research_project
		start_scheme = {
			type = research_project_public
			targets_nothing = yes
			save_scope_as = scheme
		}
		scope:scheme = {
			copy_research_project_data = { SCHEME = scope:old_scheme }
			set_variable = { name = owner value = scope:owner }
		}
		scope:old_scheme = { invalidate_scheme = yes }
		trigger_event = { id = cotc_research_project_events.61 days = 1 }
		scope:previous_holder = { trigger_event = { id = cotc_research_project_events.62 days = 1 } }
	}
}
research_project_researcher_purge = { # You've been off the project for a year, you no longer count as a contributor
	trigger = {
		exists = scope:scheme
		scope:scheme ?= {
			is_target_in_variable_list = { name = researchers target = ROOT }
			NOT = { scheme_is_character_agent = ROOT }
		}
	}
	effect = { scope:scheme = { remove_list_variable = { name = researchers target = ROOT } } }
}
