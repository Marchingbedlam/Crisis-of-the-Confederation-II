tradition_cotc_asymmetric_doctrine = {
	category = combat

	layers = {
		0 = intrigue
		1 = western
		4 = soldiers4.dds
	}

	parameters = {
		next_level_military_camps = yes
		skirmisher_nomadic_cultrad_stationing_bonus = yes
	}
	
	character_modifier = {
		enemy_terrain_advantage = -0.25
		archers_damage_mult = 0.15
		archers_toughness_mult = 0.15
		archers_maintenance_mult = -0.05
		skirmishers_damage_mult = 0.15
		skirmishers_toughness_mult = 0.2
		skirmishers_maintenance_mult = -0.05	
	}
	
	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOR = {
						culture_pillar:ethos_bellicose = { is_in_list = traits }
						culture_pillar:ethos_spiritual = { is_in_list = traits }
						culture_pillar:ethos_stoic = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_bellicose_spiritual_or_stoic_desc
				}
			}
			if = {
				limit = {
					NOT = {
						scope:character = {
							number_maa_soldiers_of_base_type = {
								type = skirmishers
								value >= 500
							}	
							number_maa_soldiers_of_base_type = {
								type = archers
								value >= 500
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = culture_head_small_soldiers_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_cotc_asymmetric_doctrine }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 400
		if = {
			limit = {
				scope:character = {
					OR = {
						number_maa_soldiers_of_base_type = {
							type = heavy_infantry
							value >= 200
						}	
						number_maa_soldiers_of_base_type = {
							type = pikemen
							value >= 200
						}
					}
				}
			}
			multiply = 0
		}
		if = {
			limit = {
				OR = {
					culture_has_heavy_infantry_maa = yes
					culture_has_heavy_cavalry_maa = yes
				}
			}
			multiply = 0
		}
		else_if = {
			limit = {
				NOR = {
					culture_has_archer_maa = yes
					culture_has_skirmisher_maa = yes
				}
			}
			multiply = 0.25
		}
		else_if = {
			limit = {
				scope:character ?= {
					ai_has_warlike_personality = yes
				}
			}
			multiply = 2
		}
	}
}
