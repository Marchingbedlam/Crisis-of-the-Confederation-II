state_capacity = {
	default = state_capacity_0
	cumulative = yes
	flag = realm_law
	
	state_capacity_0 = {
		modifier = {
			belligerent_opinion = -10
			barons_and_minor_landholders_opinion = 20
			glory_hound_opinion = 10
			parochial_opinion = 10
			courtly_opinion = 5
		}
		flag = uses_state_capacity
		flag = can_have_tributaries

		can_keep = {
			realm_law_uses_state_capacity = yes
			trigger_if = {
				limit = { top_liege != this }
				liege = { has_realm_law = state_capacity_0 }
			}
		}
		
		can_pass = {
			trigger_if = {
				limit = {
					top_liege != this
					liege = { has_realm_law = state_capacity_0 }
				}
				custom_tooltip = {
					text = imperial_bureaucracy_pending_law_change
					always = no
				}
			}
			custom_description = {
				text = "no_powerful_vassal_with_negative_opinion"
				NOT = {
					any_vassal = {
						is_powerful_vassal = yes
						opposes_succession_law_change_trigger = yes
					}
				}
			}
		}

		on_pass = {
			# Remove modifiers.
			remove_law_related_modifiers_effect = yes
		}
	}
	
	state_capacity_1 = {

		}
		flag = uses_strain

		flag = vassal_retraction_allowed
		flag = can_change_partition_succession_laws
		flag = vassal_all_wars_banned

		triggered_flag = {
			trigger = { NOT = { government_has_flag = government_is_military } }

		triggered_flag = {

		triggered_flag = {

		}
		triggered_flag = {
			trigger = { NOT = { government_has_flag = government_is_theocratic } }

		}

		can_keep = {
			realm_law_uses_state_capacity = yes
			trigger_if = {
				limit = { top_liege != this }
				liege = { has_realm_law = state_capacity_1 }
			}
			trigger_else = { legitimacy_level >= 1 }
		}

		can_pass = {
			trigger_if = {
				limit = {
					has_realm_law = state_capacity_0
					top_liege = this
				}
				custom_description = {
					subject = root
					text = "has_imperial_bureaucracy_cooldown"
					NAND = {
						has_variable = imperial_bureaucracy_cooldown
						NOT = { culture = { has_innovation = innovation_all_things } }
					}
				}
			}
			trigger_if = {
				limit = {
					top_liege != this

						is_powerful_vassal = yes
						opposes_succession_law_change_trigger = yes
					}
				}
			}
		}
		
		pass_cost = {
			prestige = {
				if = {
					limit = {
						NOR = {
							has_realm_law = state_capacity_2
							has_realm_law = state_capacity_3
						}
					}
					add = increase_crown_authority_prestige_cost

			}
		}
		
		on_pass = {
			# Increment cooldown breaker tally, if appropriate.
			calculate_authority_cooldown_break_effect = yes
			# Set regular cooldown.
			set_variable = {
				name = imperial_bureaucracy_cooldown
				years = @imperial_bureaucracy_cooldown_years_1
			}
			# Remove modifiers.
			remove_law_related_modifiers_effect = yes
		}

		ai_will_do = {
			if = {
				limit = { has_realm_law = state_capacity_0 }
				value = 1
			}
		}
	}

military_provinces_laws = {
	default = military_provinces_self_determined_law
	cumulative = no
	flag = imperial_policy
	
	military_provinces_self_determined_law = {
		can_keep = { can_have_vassal_government_laws_trigger = yes }
		can_have = { can_have_vassal_government_laws_trigger = yes }
		can_pass = { legislature_blocks_law_trigger = no }
		on_pass = {
			set_variable = { name = provincial_law_cooldown years = @admin_law_cooldown_value }
			pass_vassal_government_law_effect = { POWER_BASE = flag:military LEGITIMACY = flag:free }
		}
	}
	military_provinces_democratic_law = {
		can_keep = {
			can_have_vassal_government_laws_trigger = yes
			has_realm_law_flag = can_set_military_province_governments
		}
		should_start_with = {
			faith = {
				has_doctrine_parameter = prefers_military_power
				has_doctrine_parameter = prefers_democracy
			}
		}
		can_have = {
			can_have_vassal_government_laws_trigger = yes
			has_realm_law_flag = can_set_military_province_governments
		}
		can_pass = {
			legislature_blocks_law_trigger = no
			OR = {
				should_have_legislature_trigger = yes
				AND = {
					custom_tooltip = {
						text = provincial_law_cooldown_desc
						NOT = { has_variable = provincial_law_cooldown }
					}
				}
			}
		}
		pass_cost = { prestige = { add = admin_law_prestige_cost } }
		on_pass = {
			set_variable = { name = provincial_law_cooldown years = @admin_law_cooldown_value }
			pass_vassal_government_law_effect = { POWER_BASE = flag:military LEGITIMACY = flag:democratic }
		}
		ai_will_do = {
			if = { # We want to bring the military in line with our democratic ideals
				limit = {
					trigger_if = {
						limit = { exists = secret_faith }
						secret_faith = { has_doctrine_parameter = prefers_democracy }
					}
					trigger_else = { faith = { has_doctrine_parameter = prefers_democracy } }
					ai_zeal > ai_rationality
				}
				value = 1
			}
		}
	}
	military_provinces_aristocratic_law = {

	military_provinces_meritocratic_law = {
		can_keep = {

			has_realm_law_flag = can_set_military_province_governments
		}
		should_start_with = {
			faith = {
				OR = {
					has_doctrine_parameter = prefers_commercial_power
					has_doctrine_parameter = prefers_administrative_power
					has_doctrine_parameter = prefers_meritocracy
				}
			}
		}
		can_have = {
			can_have_vassal_government_laws_trigger = yes

			pass_vassal_government_law_effect = { POWER_BASE = flag:military LEGITIMACY = flag:meritocratic }
		}
		ai_will_do = {
			if = { # We want to bring the military under tighter control
				limit = {
					NOT = { government_has_flag = government_is_military }
					OR = {
						trigger_if = {
							limit = { exists = secret_faith }
							secret_faith = { has_doctrine_parameter = prefers_meritocracy }
						}
						trigger_else = { faith = { has_doctrine_parameter = prefers_meritocracy } }
						ai_rationality > ai_zeal

				}

						limit = { exists = secret_faith }
						secret_faith = { has_doctrine_parameter = prefers_meritocracy }
					}
					trigger_else = { faith = { has_doctrine_parameter = prefers_meritocracy } }
