activity_invite_rule_legislators = {
	effect = {
		every_courtier = {
			limit = {
				has_variable = legislature
				var:legislature ?= { var:legislature_title = { holder = ROOT } }
			}
			add_to_list = characters
		}
		every_courtier_away = {
			limit = {
				has_variable = legislature
				var:legislature ?= { var:legislature_title = { holder = ROOT } }
			}
			add_to_list = characters
		}
		every_vassal_or_below = {
			every_courtier = {
				limit = {
					has_variable = legislature
					var:legislature ?= { var:legislature_title = { holder = ROOT } }
				}
				add_to_list = characters
			}
			every_courtier_away = {
				limit = {
					has_variable = legislature
					var:legislature ?= { var:legislature_title = { holder = ROOT } }
				}
				add_to_list = characters
			}
		}
	}
}
