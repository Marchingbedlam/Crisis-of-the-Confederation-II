	on_start = {
		scope:activity = {
			set_variable = { name = ruler value = scope:host }
			set_variable = { name = title value = scope:host.primary_title }

	on_complete = {
		scope:activity = {
			var:ruler = { save_scope_as = ruler }
			var:title = { save_scope_as = title }
			var:legislature = { save_scope_as = legislature }
			save_scope_value_as = { name = final_supporters value = total_bill_supporters }
			save_scope_value_as = { name = final_opponents value = total_bill_opponents }
			save_scope_value_as = { name = proposed_military_tax_rate value = proposed_military_tax_rate }
			save_scope_value_as = { name = proposed_commercial_tax_rate value = proposed_commercial_tax_rate }
			save_scope_value_as = { name = proposed_technocratic_tax_rate value = proposed_technocratic_tax_rate }
			save_scope_value_as = { name = proposed_theocratic_tax_rate value = proposed_theocratic_tax_rate }
			save_scope_value_as = { name = pork_value value = var:pork }
			every_in_list = {
				variable = pork_recipients
				add_to_list = pork_distribution
			}
		}
		if = {
			limit = { scope:activity = { has_variable = bill_passed } }
			if = {
				limit = { ROOT = scope:host }
				trigger_event = legislature_events.3100
				every_vassal = { trigger_event = legislature_events.3105 }
				every_tributary = { trigger_event = legislature_events.3105 }
			}
			else_if = {
				limit = { is_in_guest_subset = { name = bill_supporters } }
				trigger_event = legislature_events.3101
			}
			else_if = {
				limit = { is_in_guest_subset = { name = bill_opponents } }
				trigger_event = legislature_events.3102
			}
		}
		else = {
			if = {
				limit = { ROOT = scope:host }
				trigger_event = legislature_events.3110
			}
			else_if = {
				limit = { is_in_guest_subset = { name = bill_supporters } }
				trigger_event = legislature_events.3111
			}
			else_if = {
				limit = { is_in_guest_subset = { name = bill_opponents } }
				trigger_event = legislature_events.3112
			}
		}
		trigger_event = { on_action = bill_clean_legislators_on_action }
	}
	
	phases = {
		debate = {
			order = 1
			is_predefined = yes
			
			is_shown = { always = yes }

			on_enter_phase = {
				scope:activity = { 
					progress_activity_phase_after = { days = 1 }
				}
			}
			
			on_phase_active = {
				if = {
					limit = { this = scope:host } # Run only once
					scope:activity = {
						# First, check that we've met quorum
						# For budget bills, quorum is a simple majority
						if = {
							limit = {
								any_attending_character = {
									save_temporary_scope_as = evaluated_character
									is_character_in_legislature_trigger = { LEGISLATURE = scope:activity.var:legislature CHARACTER = scope:evaluated_character }
									count < scope:activity.var:legislature.simple_majority
								}
							}
							set_variable = quorum_not_met
						}
						# We've met quorum, so collect everyone's initial votes
						else = {
							every_attending_character = {
								limit = {
									save_temporary_scope_as = evaluated_character
									is_character_in_legislature_trigger = { LEGISLATURE = scope:activity.var:legislature CHARACTER = scope:evaluated_character }
								}
								trigger_event = legislature_events.3000
							}
							# Check whether we can proceed directly to the voting phase,
							# or if further debate is needed
							scope:host = {
								trigger_event = { id = legislature_events.1100 delayed = yes }
								trigger_event = { id = legislature_events.1101 delayed = yes }
							}
						}
					}
				}
			}
			
			on_weekly_pulse = {
				if = {
					limit = { ROOT = scope:host }
					# Offer to amend the budget in order to get more supporters
					# Give someone a chance to give a speech
					trigger_event = { on_action = bill_debate_pulse_speech_on_action days = { 2 6 } }
				}
				else_if = {
					limit = { is_character_in_legislature_trigger = { LEGISLATURE = scope:activity.var:legislature CHARACTER = ROOT } }
					# Attending legislators do things
					trigger_event = { on_action = bill_debate_pulse_legislator_on_action days = { 2 6 } }
				}
			}
		}
		voting = {
			order = 2
			is_predefined = yes
			
			is_shown = { always = yes }

			on_enter_phase = {
				scope:activity = { 
					progress_activity_phase_after = { days = 1 } 
				}		
			}
			
			on_phase_active = {
				# The final vote! Did we succeed or fail?
				if = {
					limit = { ROOT = scope:host }
					scope:host = {
						trigger_event = legislature_events.1200 # Bill passed
						trigger_event = legislature_events.1210 # Bill failed
					}
				}
			}
		}
	}
