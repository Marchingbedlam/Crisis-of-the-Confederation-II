### COTC - Freelancer (landless adventurer) effects

# Decides the government a settling Freelancer receives.
# Vanilla's ep3_become_landed_transfer_effect derives the new government from the SELLER's government.
# COTC governments still carry vanilla compatibility flags so they end up resolving onto vanilla government types (feudal_government, clan_government)  which we don't want.
#
# Instead, the settling character's Charter decides their power base.
# Legitimacy and authority still come from ideology, exactly as they do for any other route into a state government.
cotc_freelancer_settle_government_effect = {
	if = {
		# Privateers settle as an outlaw regime rather than a state.
		limit = { has_realm_law = camp_purpose_brigands }
		set_to_default_non_state_government = yes
	}
	else = {
		if = {
			limit = { has_realm_law = camp_purpose_mercenaries }
			save_scope_value_as = { name = power_base value = flag:military }
		}
		else_if = {
			limit = { has_realm_law = camp_purpose_scholars }
			save_scope_value_as = { name = power_base value = flag:technocratic }
		}
		else_if = {
			limit = { has_realm_law = camp_purpose_explorers }
			save_scope_value_as = { name = power_base value = flag:commercial }
		}
		else_if = {
			# Suppressed in Phase 1
			limit = { has_realm_law = camp_purpose_legitimists }
			save_scope_value_as = { name = power_base value = flag:military }
		}
		# camp_purpose_wanderers deliberately sets nothing: an unregistered group has declared no trade, so the ideology chooses

		set_to_default_state_government = { KEEP_CURRENT_POWER_BASE = flag:no }
	}
}
