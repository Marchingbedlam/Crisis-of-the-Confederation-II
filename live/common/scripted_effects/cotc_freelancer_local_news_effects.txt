# Root = the character
cotc_local_news_flag_effect = {
	if = {
		limit = { scope:1025_offer_$SLOT$ ?= { has_variable = cotc_news_article } }
		set_variable = {
			name = cotc_news_$SLOT$
			value = scope:1025_offer_$SLOT$.var:cotc_news_article
		}
	}
	else = {
		set_variable = { name = cotc_news_$SLOT$  value = flag:generic }
	}
}

# Root = the character. Every entry is a whole sign-off line
# Avoided custom loc since it has no access to the scope
cotc_local_news_network_effect = {
	random_list = {
		1 = { set_variable = { name = cotc_news_network  value = flag:network_1 } }
		2 = { set_variable = { name = cotc_news_network  value = flag:network_2 } }
		3 = { set_variable = { name = cotc_news_network  value = flag:network_3 } }
	}
}

cotc_local_news_clear_flags_effect = {
	remove_variable = cotc_news_first
	remove_variable = cotc_news_second
	remove_variable = cotc_news_third
	remove_variable = cotc_news_network
}
