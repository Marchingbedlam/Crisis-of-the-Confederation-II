# AGENT EFFECTS
# These will depend on what agent types are defined for research projects
add_research_project_agent_slots = {
	while = {
		limit = { any_scheme_agent_slot = { count < var:max_agent_slots } }
		add_new_research_project_agent_slot = { TYPE = economist }
		add_new_research_project_agent_slot = { TYPE = sociologist }
		add_new_research_project_agent_slot = { TYPE = officer }
		add_new_research_project_agent_slot = { TYPE = doctor }
		add_new_research_project_agent_slot = { TYPE = engineer }
		add_new_research_project_agent_slot = { TYPE = physicist }
	}
}
copy_research_project_data = {
	save_temporary_scope_as = temp_new_scheme
	# Copy basic data
	set_variable = { name = research_subject value = $SCHEME$.var:research_subject }
	set_variable = { name = codename_adjective value = $SCHEME$.var:codename_adjective }
	set_variable = { name = codename_noun value = $SCHEME$.var:codename_noun }
	set_variable = { name = grant_money value = $SCHEME$.var:grant_money }
	set_variable = { name = max_agent_slots value = $SCHEME$.var:max_agent_slots }
	add_scheme_progress = $SCHEME$.scheme_progress
	set_research_project_difficulty_effect = yes
	
	# Copy flags
	$SCHEME$ = {
		every_in_list = {
			variable = flags
			scope:temp_new_scheme = { add_to_variable_list = { name = flag target = PREV } }
		}
	}
	
	# Copy agents
	if = {
		limit = { exists = $SCHEME$.var:max_economist_slots }
		set_variable = { name = max_economist_slots value = $SCHEME$.var:max_economist_slots }
	}
	if = {
		limit = { exists = $SCHEME$.var:max_sociologist_slots }
		set_variable = { name = max_sociologist_slots value = $SCHEME$.var:max_sociologist_slots }
	}
	if = {
		limit = { exists = $SCHEME$.var:max_officer_slots }
		set_variable = { name = max_officer_slots value = $SCHEME$.var:max_officer_slots }
	}
	if = {
		limit = { exists = $SCHEME$.var:max_doctor_slots }
		set_variable = { name = max_doctor_slots value = $SCHEME$.var:max_doctor_slots }
	}
	if = {
		limit = { exists = $SCHEME$.var:max_engineer_slots }
		set_variable = { name = max_engineer_slots value = $SCHEME$.var:max_engineer_slots }
	}
	if = {
		limit = { exists = $SCHEME$.var:max_physicist_slots }
		set_variable = { name = max_physicist_slots value = $SCHEME$.var:max_physicist_slots }
	}
	add_research_project_agent_slots = yes
	$SCHEME$ = {
		every_scheme_agent_slot = {
			limit = { is_filled = yes }
			save_temporary_scope_as = old_agent_slot
			switch = {
				trigger = is_agent_slot_type
				agent_research_economist = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = economist } }
				agent_research_sociologist = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = sociologist } }
				agent_research_officer = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = officer } }
				agent_research_doctor = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = doctor } }
				agent_research_engineer = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = engineer } }
				agent_research_physicist = { copy_research_project_agent_slot = { NEW_SCHEME = scope:temp_new_scheme TYPE = physicist } }
			}
		}
	}
}

# STRING EFFECTS
# These are modular and will not need as much constant maintenance
add_tech_by_string = {
	save_temporary_scope_as = temp_county_name
	culture_innovation:innovation_$TECH$ ?= { save_temporary_scope_as = temp_tech }
	if = {
		limit = {
			exists = scope:temp_tech
			NOT = { county_has_tech_by_string = { TECH = $TECH$ } }
		}
		custom_tooltip = {
			text = add_tech_to_county.tt
			add_to_variable_list = { name = county_techs target = culture_innovation:innovation_$TECH$ }
			add_to_global_variable_list = { name = all_discovered_innovations target = culture_innovation:innovation_$TECH$ }
			culture = { add_to_variable_list = { name = culture_discovered_innovations target = culture_innovation:innovation_$TECH$ } }
			add_tech_to_gui = { TECH = $TECH$ }
			
			# Apply the new tech
			apply_county_tech_by_string = { TECH = $TECH$ }
			every_county_province = {
				province_owner ?= {
					add_to_temporary_list = county_characters
					every_courtier = { add_to_temporary_list = county_characters }
					every_pool_character = { add_to_temporary_list = county_characters }
				}
			}
			every_in_list = {
				list = county_characters
				apply_character_tech_by_string = { TECH = $TECH$ }
			}
		}
	}
}
remove_tech_by_string = {
	if = {
		limit = { county_has_tech_by_string = { TECH = $TECH$ } }
		remove_list_variable = { name = county_techs target = culture_innovation:innovation_$TECH$ }
		remove_tech_from_gui = { TECH = $TECH$ }
		
		# Verify the removed tech
		apply_county_tech_by_string = { TECH = $TECH$ }
		every_county_province = {
			province_owner ?= {
				add_to_temporary_list = county_characters
				every_courtier = { add_to_temporary_list = county_characters }
				every_pool_character = { add_to_temporary_list = county_characters }
			}
		}
		every_in_list = {
			list = county_characters
			apply_character_tech_by_string = { TECH = $TECH$ }
		}
	}
}

apply_tech_by_string = {
	switch = {
		trigger = $TARGET$
		flag:county = { apply_county_tech_by_string = { TECH = $TECH$ } }
		flag:character = { apply_character_tech_by_string = { TECH = $TECH$ } }
		flag:province = { apply_province_tech_by_string = { TECH = $TECH$ } }
	}
}
apply_county_tech_by_string = {
	if = {
		limit = { has_global_variable = tech_$TECH$_county_modifier_defined }
		if = {
			limit = {
				county_has_tech_by_string = { TECH = $TECH$ }
				NOT = { has_county_modifier = tech_$TECH$_county_modifier }
			}
			add_county_modifier = tech_$TECH$_county_modifier
		}
		else_if = {
			limit = {
				has_county_modifier = tech_$TECH$_county_modifier
				NOT = { county_has_tech_by_string = { TECH = $TECH$ } }
			}
			remove_all_county_modifier_instances = tech_$TECH$_county_modifier
		}
	}
	if = {
		limit = { has_global_variable = tech_$TECH$_province_modifier_defined }
		if = {
			limit = { county_has_tech_by_string = { TECH = $TECH$ } }
			every_county_province = {
				limit = { NOT = { has_province_modifier = tech_$TECH$_province_modifier } }
				add_province_modifier = tech_$TECH$_province_modifier
			}
		}
		else = {
			every_county_province = {
				limit = { has_province_modifier = tech_$TECH$_province_modifier }
				remove_all_province_modifier_instances = tech_$TECH$_province_modifier
			}
		}
	}
}

add_new_research_project_agent_slot = {
	if = {
		limit = {
			exists = var:max_$TYPE$_slots
			any_scheme_agent_slot = { count < var:max_agent_slots }
		}
		if = {
			limit = {
				any_scheme_agent_slot = {
					count < var:max_$TYPE$_slots
					is_agent_slot_type = agent_research_$TYPE$
				}
			}
			add_agent_slot = agent_research_$TYPE$
		}
	}
}

save_research_project_scopes_effect = { # Run this when it's possible the project will end before some scopes need to be checked
	var:research_subject = { save_scope_as = research_subject }
	save_scope_value_as = { name = grant_money value = var:grant_money }
	scheme_owner = { save_scope_as = patron }
	every_scheme_agent_character = { add_to_list = project_staff }
	every_in_list = {
		variable = flags
		add_to_list = research_project_flags
	}
}
set_research_project_difficulty_effect = {	
	# Set difficulty
	var:research_subject = {
		switch = {
			trigger = has_innovation_parameter
			colonial_civic = { scope:scheme = { add_scheme_modifier = { type = colonial_research_subject_difficulty } } }
			colonial_military = { scope:scheme = { add_scheme_modifier = { type = colonial_research_subject_difficulty } } }
			confederate_civic = { scope:scheme = { add_scheme_modifier = { type = confederate_research_subject_difficulty } } }
			confederate_military = { scope:scheme = { add_scheme_modifier = { type = confederate_research_subject_difficulty } } }
			trimillennial_civic = { scope:scheme = { add_scheme_modifier = { type = trimillennial_research_subject_difficulty } } }
			trimillennial_military = { scope:scheme = { add_scheme_modifier = { type = trimillennial_research_subject_difficulty } } }
			neorenaissance_civic = { scope:scheme = { add_scheme_modifier = { type = neorenaissance_research_subject_difficulty } } }
			neorenaissance_military = { scope:scheme = { add_scheme_modifier = { type = neorenaissance_research_subject_difficulty } } }
		}
	}
	scheme_owner = {
		capital_county ?= {
			culture = {
				if = {
					limit = { has_cultural_era_or_later = culture_era_early_medieval }
					scope:scheme = { add_scheme_modifier = { type = confederate_era_research_bonus } }
				}
				if = {
					limit = { has_cultural_era_or_later = culture_era_high_medieval }
					scope:scheme = { add_scheme_modifier = { type = trimillennial_era_research_bonus } }
				}
				if = {
					limit = { has_cultural_era_or_later = culture_era_late_medieval }
					scope:scheme = { add_scheme_modifier = { type = neorenaissance_era_research_bonus } }
				}
				every_culture_county = {
					limit = { county_has_tech = { TECH = scope:scheme.var:research_subject } }
					scope:scheme = { add_scheme_modifier = { type = culture_research_bonus } }
				}
			}
			every_neighboring_county = {
				limit = { county_has_tech = { TECH = scope:scheme.var:research_subject } }
				scope:scheme = { add_scheme_modifier = { type = neighbor_research_bonus } }
			}
		}
		if = {
			limit = {
				any_secret = {
					has_variable = research_subject
					var:research_subject ?= scope:scheme.var:research_subject
				}
			}
			scope:scheme = { add_scheme_modifier = { type = already_researched_bonus } }
		}
	}
}
add_research_project_stress_effects = {
	if = {
		limit = {
			any_in_list = {
				list = research_project_flags
				this = flag:military
			}
		}
		positive_hawkishness_stress_gain_effect = { SCALE = -0.5 }
		negative_hawkishness_stress_gain_effect = { SCALE = 1 }
	}
	if = {
		limit = { exists = secret_faith }
		secret_faith ?= { save_temporary_scope_as = real_ideology }
	}
	else = { faith = { save_temporary_scope_as = real_ideology } }
	if = {
		limit = {
			any_in_list = {
				list = research_project_flags
				this = flag:cybernetics
			}
		}
		if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = hates_cybernetics } }
			positive_zeal_stress_gain_effect = { SCALE = 1 }
		}
		else_if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = supports_cybernetics } }
			positive_zeal_stress_gain_effect = { SCALE = 0.5 }
		}
	}
	if = {
		limit = {
			any_in_list = {
				list = research_project_flags
				this = flag:gene_modding
			}
		}
		if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = hates_gene_modding } }
			positive_zeal_stress_gain_effect = { SCALE = 1 }
		}
		else_if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = supports_gene_modding } }
			positive_zeal_stress_gain_effect = { SCALE = 0.5 }
		}
	}
	if = {
		limit = {
			any_in_list = {
				list = research_project_flags
				this = flag:cloning
			}
		}
		if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = hates_cloning } }
			positive_zeal_stress_gain_effect = { SCALE = 1 }
		}
		else_if = {
			limit = { scope:real_ideology ?= { has_doctrine_parameter = supports_cloning } }
			positive_zeal_stress_gain_effect = { SCALE = 0.5 }
		}
	}
}
add_research_project_secret_effect = {
	save_temporary_scope_as = temp_secret_owner
	every_secret = {
		limit = { secret_type = secret_researched_innovation }
		add_to_temporary_list = temp_existing_secrets
	}
	if = {
		limit = {
			NOT = {
				any_in_list = {
					list = temp_existing_secrets
					trigger_if = {
						limit = { has_variable = research_subject }
						var:research_subject = scope:discovered_innovation
					}
				}
			}
		}
		add_secret = { type = secret_researched_innovation }
		random_secret = {
			limit = {
				secret_type = secret_researched_innovation
				NOT = { is_in_list = temp_existing_secrets }
			}
			set_variable = { name = research_subject value = scope:discovered_innovation }
			save_temporary_scope_as = temp_new_secret
			every_in_list = {
				list = research_project_flags
				limit = {
					OR = {
						this = flag:military
						this = flag:cybernetics
						this = flag:gene_modding
						this = flag:cloning
					}
				}
				scope:temp_new_secret = { add_to_variable_list = { name = flags target = PREV } }
			}
		}
	}
	random_secret = {
		limit = {
			secret_type = secret_researched_innovation
			trigger_if = {
				limit = { has_variable = research_subject }
				var:research_subject = scope:discovered_innovation
			}
		}
		if = { # Set who knows that we know this
			limit = { exists = scope:secret }
			save_temporary_scope_as = temp_secret
			scope:patron = { add_to_temporary_list = temp_project_participants }
			every_in_list = {
				list = project_staff
				add_to_temporary_list = temp_project_participants
			}
			every_in_list = {
				list = temp_project_participants
				limit = { NOT = { this = scope:temp_secret_owner } }
				scope:temp_secret = { add_secret_participant = PREV }
			}
		}
		else = { expose_secret = yes }
	}
}
research_project_prep_effect = {
	save_scope_as = scheme
	if = {
		limit = { NOT = { exists = scope:suppress_next_event } }
		scheme_owner = { trigger_event = cotc_research_project_events.100 }
	}
}
add_research_project_personal_involvement = {
	add_character_modifier = researching_modifier
	add_character_modifier = personal_involvement_learning_bonus
	if = {
		limit = { scope:scheme = { is_target_in_variable_list = { name = flags target = flag:social } } }
		add_character_modifier = personal_involvement_diplomacy_bonus
	}
	if = {
		limit = { scope:scheme = { is_target_in_variable_list = { name = flags target = flag:economic } } }
		add_character_modifier = personal_involvement_stewardship_bonus
	}
	if = {
		limit = { scope:scheme = { is_target_in_variable_list = { name = flags target = flag:military } } }
		add_character_modifier = personal_involvement_military_bonus
	}
	if = {
		limit = { basic_eligible_for_diarchy_trigger = yes }
		custom_tooltip = {
			trigger_event = cotc_research_project_events.101
			text = enter_temporary_regency.tt
		}
	}
}
remove_research_project_personal_involvement = {
	remove_character_modifier = researching_modifier
	remove_character_modifier = personal_involvement_learning_bonus
	if = {
		limit = { has_character_modifier = personal_involvement_diplomacy_bonus }
		remove_character_modifier = personal_involvement_diplomacy_bonus
	}
	if = {
		limit = { has_character_modifier = personal_involvement_stewardship_bonus }
		remove_character_modifier = personal_involvement_stewardship_bonus
	}
	if = {
		limit = { has_character_modifier = personal_involvement_martial_bonus }
		remove_character_modifier = personal_involvement_martial_bonus
	}
	if = {
		limit = {
			exists = diarch
			is_character_interaction_valid = {
				interaction = liege_dismiss_temporary_regency_interaction
				recipient = diarch
			}
		}
		custom_tooltip = {
			trigger_event = diarchy.0112
			text = dismiss_temporary_regency.tt
		}
	}
}
