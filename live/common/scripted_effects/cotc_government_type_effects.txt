sort_federation_province_effect = { # Make a province either a territory or a member-state, depending on what makes the most sense
	if = { # This character has been flagged as a member-state
		limit = { has_character_flag = new_member_state }
		vassal_contract_set_obligation_level = { type = federal_status level = federal_status_member_state }
		remove_character_flag = new_member_state
	}
	else_if = { # This character has been flagged as a territory
		limit = { has_character_flag = new_territory }
		vassal_contract_set_obligation_level = { type = federal_status level = federal_status_territory }
		remove_character_flag = new_territory
	}
	else_if = { # This character hasn't been flagged and is unsorted
		limit = { subject_contract_has_flag = federal_status_unsorted }
		if = { # This character is too low-ranking to be a member-state
			limit = { highest_held_title_tier = tier_barony }
			vassal_contract_set_obligation_level = { type = federal_status level = federal_status_territory }
		}
		else_if = { # This character is too high-ranking to be a territory
			limit = { tier_difference = { target = liege value >= -1 } }
			vassal_contract_set_obligation_level = { type = federal_status level = federal_status_member_state }
		}
		else_if = { # This character is a de jure vassal of a federal title
			limit = {
				primary_title = { save_temporary_scope_as = temp_province_title }
				liege = {
					any_held_title = {
						tier < holder.highest_held_title_tier
						is_de_jure_liege_or_above_target = scope:temp_province_title
					}
				}
			}
			vassal_contract_set_obligation_level = { type = federal_status level = federal_status_territory }
		}
		else = { vassal_contract_set_obligation_level = { type = federal_status level = federal_status_member_state } }
	}
}
pass_vassal_government_law_effect = {
	save_scope_value_as = { name = power_base value = $POWER_BASE$ }
	save_scope_value_as = { name = legitimacy value = $LEGITIMACY$ }
	switch = {
		trigger = scope:power_base
		flag:military = {
			switch = {
				trigger = has_realm_law
				military_provinces_self_determined_law = { save_scope_value_as = { name = old_legitimacy value = flag:free } }
				military_provinces_democratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:democratic } }
				military_provinces_aristocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:aristocratic } }
				military_provinces_meritocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:meritocratic } }
			}
		}
		flag:commercial = {
			switch = {
				trigger = has_realm_law
				commercial_provinces_self_determined_law = { save_scope_value_as = { name = old_legitimacy value = flag:free } }
				commercial_provinces_democratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:democratic } }
				commercial_provinces_aristocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:aristocratic } }
				commercial_provinces_meritocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:meritocratic } }
			}
		}
		flag:technocratic = {
			switch = {
				trigger = has_realm_law
				technocratic_provinces_self_determined_law = { save_scope_value_as = { name = old_legitimacy value = flag:free } }
				technocratic_provinces_democratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:democratic } }
				technocratic_provinces_aristocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:aristocratic } }
				technocratic_provinces_meritocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:meritocratic } }
			}
		}
		flag:theocratic = {
			switch = {
				trigger = has_realm_law
				theocratic_provinces_self_determined_law = { save_scope_value_as = { name = old_legitimacy value = flag:free } }
				theocratic_provinces_democratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:democratic } }
				theocratic_provinces_aristocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:aristocratic } }
				theocratic_provinces_meritocratic_law = { save_scope_value_as = { name = old_legitimacy value = flag:meritocratic } }
			}
		}
	}
	save_scope_as = enacting_ruler
	if = {
		limit = { has_variable = bill_passed }
		save_scope_value_as = { name = passed_by_legislature value = flag:yes }
	}
	else = { save_scope_value_as = { name = passed_by_legislature value = flag:no } }
	if = {
		limit = { has_variable = bill_sponsor }
		var:bill_sponsor = { save_scope_as = passed_by }
		remove_variable = bill_sponsor
	}
	else = { save_scope_as = passed_by }
	custom_tooltip = {
		text = change_province_governments_tt
		every_vassal_or_below = {
			debug_log = "[THIS.Char.Custom('GetBestName')] checking for government change..."
			if = {
				limit = {
					subject_to_vassal_government_laws_trigger = { LIEGE = scope:enacting_ruler }
					debug_log = yes
				}
				trigger_event = government_change_events.100 # Bring government in line with law
			}
			else = { trigger_event = government_change_events.101 } # Keep other vassals in the loop on how the state is being run
		}
	}
}

set_to_default_state_government = {
	if = {
		limit = { $KEEP_CURRENT_POWER_BASE$ = flag:yes }
		faith ?= { save_temporary_scope_as = temp_faith }
		capital_province ?= { select_power_base_by_province = { FAITH = scope:temp_faith } }
	}
	faith ?= { # Choose all other government values by ideology
		if = {
			limit = { NOT = { exists = scope:power_base } }
			select_ideology_default_power_base = yes
		}
		select_ideology_default_legitimacy = yes
		select_ideology_default_authority = yes
		save_scope_as = new_state_ideology
	}
	if = {
		limit = {
			trigger_if = {
				limit = { exists = scope:legitimacy }
				scope:legitimacy = flag:federal
				highest_held_title_tier < tier_duchy
			}
		}
		random_list = {
			10 = { save_scope_value_as = { name = legitimacy value = flag:democratic } }
			10 = { save_scope_value_as = { name = legitimacy value = flag:aristocratic } }
			10 = { save_scope_value_as = { name = legitimacy value = flag:meritocratic } }
		}
	}
	if = {
		limit = {
			exists = scope:power_base
			exists = scope:legitimacy
			exists = scope:authority
		}
		become_specified_state_government = { POWER_BASE = scope:power_base LEGITIMACY = scope:legitimacy AUTHORITY = scope:authority }
	}
	else = { debug_log = "Attempted to set default state government for [THIS.Char.Custom('GetBestName')], but failed due to missing specifications." }
}

	switch = {
		trigger = has_holding_type
		castle_holding = { save_scope_value_as = { name = power_base value = flag:military } }
		city_holding = { save_scope_value_as = { name = power_base value = flag:commercial } }
		church_holding = {
			if = {
				limit = { $FAITH$ = { has_doctrine = religious_hostility_doctrine } }
				save_scope_value_as = { name = power_base value = flag:theocratic }
			}
			else = { save_scope_value_as = { name = power_base value = flag:technocratic } }
		}
	}
}
