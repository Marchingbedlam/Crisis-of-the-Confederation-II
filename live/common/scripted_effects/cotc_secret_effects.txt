give_official_bribe_secret_effect = {
	if = {
		limit = {
			NOR = {
				# Bribing criminals isn't a crime, it's just business
				$TARGET$.liege = { government_has_flag = government_is_criminal }
				$TARGET$ = {
					is_ruler = yes
					government_has_flag = government_is_criminal
				}
				# Neither is bribery where everyone involved is a despot
				AND = {
					is_ruler = yes
					government_has_flag = government_is_despotic
					$TARGET$ = {
						OR = {
							AND = {
								is_ruler = yes
								government_has_flag = government_is_despotic
							}
							AND = {
								is_ruler = no
								liege = { government_has_flag = government_is_despotic }
							}
						}
					}
				}
			}
		}
		# Okay, this is a real secret, add it
		add_secret = {
			type = secret_official_bribe
			target = $TARGET$
		}
		random_secret = {
			limit = {
				secret_type = secret_official_bribe
				secret_target = $TARGET$
				NOT = { any_secret_participant = { this = $TARGET$ } }
			}
			add_secret_participant = $TARGET$
		}
	}
}
give_performed_cybersurgery_secret_effect = {
	if = { # Make sure this is worth keeping secret
		limit = {
			OR = {
				is_implant_illegal_for_character = { IMPLANT = $IMPLANT$ }
				rite = { rite_has_parameter = hates_cybernetics }
				any_liege_or_above = { rite = { rite_has_parameter = hates_cybernetics } }
			}
		}
		# This is a real secret, add it
		add_secret = {
			type = secret_performed_cybersurgery
			target = $TARGET$
		}
		random_secret = {
			limit = {
				secret_type = secret_performed_cybersurgery
				target = $TARGET$
				NOT = { any_secret_participant = { this = $TARGET$ } }
			}
			add_secret_participant = $TARGET$
			set_variable = { name = implant value = $IMPLANT$ }
		}
	}
}
