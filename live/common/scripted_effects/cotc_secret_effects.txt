give_official_bribe_secret_effect = {
	if = {
		limit = {
			NOR = {
				# Bribing criminals isn't a crime, it's just business
				$TARGET$.liege = { government_has_flag = government_is_criminal }
				$TARGET$ = {
					is_ruler = yes
					government_has_flag = government_is_criminal
				}
				# Neither is bribery where everyone involved is a despot
				AND = {
					is_ruler = yes
					government_has_flag = government_is_despotic
					$TARGET$ = {
						OR = {
							AND = {
								is_ruler = yes
								government_has_flag = government_is_despotic
							}
							AND = {
								is_ruler = no
								liege = { government_has_flag = government_is_despotic }
							}
						}
					}
				}
			}
		}
		# Okay, this is a real secret, add it
		add_secret = {
			type = secret_official_bribe
			target = $TARGET$
		}
		random_secret = {
			limit = {
				type = secret_official_bribe
				secret_target = $TARGET$
				NOT = { any_secret_participant = { this = $TARGET$ } }
			}
			add_secret_participant = $TARGET$
		}
	}
}
