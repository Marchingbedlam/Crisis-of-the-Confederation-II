build_supporters_gained_list_effect = {
	scope:activity = {
		every_guest_subset = {
			name = bill_opponents
			limit = {
				trigger_if = {
					limit = { scope:activity = { has_activity_type = budget_bill_activity } }
					supports_budget_amendment_trigger = yes
				}
				trigger_else_if = {
					limit = { scope:activity = { has_activity_type = tax_bill_activity } }
					supports_tax_code_amendment_trigger = yes
				}
				trigger_else = { always = no }
				vote_is_open_trigger = yes
			}
			scope:host = { add_to_variable_list = { name = supporters_gained target = PREV } }
		}
	}
	scope:host = {
		ordered_in_list = {
			variable = supporters_gained
			order_by = diplomacy
			save_scope_as = representative_opponent
		}
	}
}

pork_spending_effect = {
	if = {
		limit = { scope:pork_value > 0 }
		scope:ruler = { remove_treasury = scope:pork_value }
		every_in_list = {
			list = pork_distribution
			limit = { tier = tier_county }
			save_temporary_scope_as = pork_target
			scope:legislature = {
				random_in_list = {
					variable = legislators
					limit = {
						has_variable = riding
						var:riding ?= { this = scope:pork_target }
					}
					save_temporary_scope_as = pork_legislator
				}
			}
			holder = {
				send_interface_message = {
					type = msg_vote_change_good
					title = pork_spending_effect.tt
					desc = pork_spending_effect.desc
				}
			}
			if = {
				limit = {
					OR = {
						has_county_modifier = reinvested_in_province_modifier
						has_county_modifier = invested_in_province_modifier
					}
				}
				add_county_modifier = {
					modifier = reinvested_in_province_modifier
					years = 5
				}
			}
			else = {
				add_county_modifier = {
					modifier = invested_in_province_modifier
					years = 5
				}
			}
		}
		every_in_list = {
			variable = pork_distribution
			limit = { tier >= tier_duchy }
			save_temporary_scope_as = pork_target
			scope:legislature = {
				random_in_list = {
					variable = legislators
					limit = {
						has_variable = riding
						var:riding ?= { this = scope:pork_target }
					}
					save_temporary_scope_as = pork_legislator
				}
			}
			holder = {
				send_interface_message = {
					type = msg_vote_change_good
					title = pork_spending_effect.tt
					desc = pork_spending_effect.desc
				}
			}
			every_de_jure_county = {
				limit = { target_is_de_facto_liege_or_above = scope:pork_target }
				if = {
					limit = {
						OR = {
							has_county_modifier = reinvested_in_province_modifier
							has_county_modifier = invested_in_province_modifier
						}
					}
					add_county_modifier = {
						modifier = reinvested_in_province_modifier
						years = 5
					}
				}
				else = {
					add_county_modifier = {
						modifier = invested_in_province_modifier
						years = 5
					}
				}
			}
		}
	}
}

	if = {
		limit = { scope:activity = { has_activity_type = budget_bill_activity } }
		vote_for_budget_strength_effect = yes
	}
	else_if = {
		limit = { scope:activity = { has_activity_type = tax_bill_activity } }
		vote_for_tax_code_strength_effect = yes

vote_for_budget_strength_effect = {
	hidden_effect = {
		set_variable = { name = vote_strength value = 0 }
		if = {
			limit = { has_activity_intent = support_bill_intent }
			change_variable = { name = vote_strength add = "opinion(scope:host)" }
		}
		else_if = {
			limit = {
				OR = {
					has_activity_intent = increase_military_budget_intent
					has_activity_intent = balance_budget_maintain_military_intent
				}
			}
			change_variable = { name = vote_strength add = scope:host.current_military_budget }
			change_variable = { name = vote_strength subtract = scope:activity.var:proposed_military_budget }
		}
		else_if = {
			limit = {
				OR = {
					has_activity_intent = increase_ministry_budget_intent
					has_activity_intent = balance_budget_maintain_ministry_intent
				}
			}
			change_variable = { name = vote_strength add = scope:host.current_ministry_budget }
			change_variable = { name = vote_strength subtract = scope:activity.var:proposed_ministry_budget }
		}
		else_if = {
			limit = {
				OR = {
					has_activity_intent = increase_salary_budget_intent
					has_activity_intent = balance_budget_maintain_salary_intent
				}
			}
			change_variable = { name = vote_strength add = scope:host.current_salary_budget }
			change_variable = { name = vote_strength subtract = scope:activity.var:proposed_salary_budget }
		}
		if = {
			limit = {
				OR = {
					has_activity_intent = balance_budget_intent
					has_activity_intent = balance_budget_maintain_military_intent
					has_activity_intent = balance_budget_maintain_ministry_intent
					has_activity_intent = balance_budget_maintain_salary_intent
				}
			}
			change_variable = { name = vote_strength add = scope:host.current_total_budget }
			change_variable = { name = vote_strength subtract = scope:activity.proposed_total_budget }
		}
	}
}
