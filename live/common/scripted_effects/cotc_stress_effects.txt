positive_lawfulness_stress_gain_effect = {
	if = {
		limit = { ai_lawfulness >= very_low_positive_ai_value }
		custom_tooltip = pos_lawfulness_desc
		add_stress = {
			value = $SCALE$
			if = {
				limit = { ai_lawfulness >= very_high_positive_ai_value }
				multiply = massive_stress_gain
			}
			else_if = {
				limit = { ai_lawfulness >= high_positive_ai_value }
				multiply = major_stress_gain
			}
			else_if = {
				limit = { ai_lawfulness >= medium_positive_ai_value }
				multiply = medium_stress_gain
			}
			else_if = {
				limit = { ai_lawfulness >= low_positive_ai_value }
				multiply = minor_stress_gain
			}
			else = { multiply = miniscule_stress_gain }
		}
	}
}
