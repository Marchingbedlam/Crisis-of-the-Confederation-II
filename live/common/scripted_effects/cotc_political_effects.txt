abolish_legislature_effect = {
	# Clear the legislature from the title
	var:legislature_title ?= { remove_variable = legislature }
	# Clear ridings from all legislators
	every_in_list = {
		variable = legislators
		remove_variable = riding
		remove_variable = legislature
	}
	# Clear all candidate lists in ridings
	switch = {
		trigger = $LEGISLATURE_TITLE$.tier
		tier_county = {
			every_in_list = {
				variable = ridings
				clear_variable_list = county_candidates
			}
		}
		tier_duchy = {
			every_in_list = {
				variable = ridings
				clear_variable_list = duchy_candidates
			}
		}
		tier_kingdom = {
			every_in_list = {
				variable = ridings
				clear_variable_list = kingdom_candidates
			}
		}
		tier_empire = {
			every_in_list = {
				variable = ridings
				clear_variable_list = empire_candidates
			}
		}
		tier_hegemony = {
			every_in_list = {
				variable = ridings
				clear_variable_list = hegemony_candidates
			}
		}
	}
	# Remove the legislature
	end_situation = yes
}

					scope:temp_legislature = { add_to_variable_list = { name = ridings target = PREV } }
				}
			}
		}
	}
}

hold_legislative_election_effect = {
	# Grab the list of candidates
	if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_county } }
		every_in_list = {
			variable = county_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = county_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_duchy } }
		every_in_list = {
			variable = duchy_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = duchy_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_kingdom } }
		every_in_list = {
			variable = kingdom_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = kingdom_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_empire } }
		every_in_list = {
			variable = empire_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = empire_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_hegemony } }
		every_in_list = {
			variable = hegemony_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = hegemony_candidates
	}
	elect_legislative_election_winner_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = this }
	every_in_list = {
		variable = candidates
		remove_variable = legislative_campaign_score
	}
	clear_variable_list = candidates
	recalc_legislature_parties_effect = yes
}
elect_legislative_election_winner_effect = {
	$LEGISLATURE$ = { save_temporary_scope_as = temp_legislature }
	$RIDING$ = { save_temporary_scope_as = temp_riding }
	if = { # Elect the winning candidate
		limit = { any_in_list = { variable = candidates exists = yes } }
		ordered_in_list = {
			variable = candidates
			limit = {
				eligible_legislator_candidate_trigger = yes
				legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
			}
			order_by = legislator_total_score
			# This is a re-election
			if = {
				limit = {
					has_variable = riding
					has_variable = legislature
					var:riding ?= { this = $RIDING$ }
					var:legislature ?= { this = $LEGISLATURE$ }
				}
				reelect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
			}
			# This is an election to a different legislature
			else_if = {
				limit = {
					has_variable = riding
					has_variable = legislature
					NOT = { var:legislature = { this = $LEGISLATURE$ } }
				}
				remove_from_legislature_effect = yes
				elect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
			}
			else = { elect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ } }
		}
	}
	else = { # If nobody ran, make up a character
		create_new_politician_effect = yes
		scope:new_politician = { elect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ } }
	}
}
create_new_politician_effect = {
	save_temporary_scope_as = temp_riding
	# Select a home province for the new legislator
	if = {
		limit = { tier = tier_county }
		random_county_province = {
			limit = { has_holding = yes }
			save_temporary_scope_as = temp_province
		}
	}
	else_if = {
		limit = { holder = { any_sub_realm_county = { target_is_de_jure_liege_or_above = scope:temp_riding } } }
		holder = {
			random_sub_realm_county = {
				limit = {
					target_is_de_jure_liege_or_above = scope:temp_riding
					any_county_province = { has_holding = yes }
				}
				random_county_province = {
					limit = { has_holding = yes }
					save_temporary_scope_as = temp_province
				}
			}
		}
	}
	else = {
		holder = {
			random_sub_realm_county = {
				limit = { any_county_province = { has_holding = yes } }
				random_county_province = {
					limit = { has_holding = yes }
					save_temporary_scope_as = temp_province
				}
			}
		}
	}
	# Generate the new legislator in the court of the province's owner
	scope:temp_province = {
		province_owner = {
			save_temporary_scope_as = temp_ruler
			create_character = {
				template = cotc_politician_template
				culture = scope:temp_province.culture
				faith = scope:temp_province.faith
				location = scope:temp_province
				save_scope_as = new_politician
			}
		}
	}
}

recalc_legislature_parties_effect = {
	save_temporary_scope_as = party_legislature
	clear_variable_list = parties
	every_in_list = {
		variable = legislators
		faith = { add_to_list = unsorted_parties }
	}
	ordered_in_list = {
		list = unsorted_parties
		order_by = ideology_represented_in_legislature
		min = 0
		check_range_bounds = yes
		if = {
			limit = { ideology_represented_in_legislature > 0 }
			scope:party_legislature = { add_to_variable_list = { name = parties target = PREV } }
		}
	}
}
start_legislative_election_campaign_effect = {
	$RIDING$ = { save_scope_as = temp_riding }
	custom_tooltip = {
		text = start_legislative_election_campaign_effect_desc
		save_scope_as = candidate
		every_scheme = { add_to_list = current_schemes_list }
		start_scheme = {
			type = legislative_election_campaign
			target_title = $RIDING$
		}
		random_scheme = {
			limit = { NOT = { is_in_list = current_schemes_list } }
			set_variable = { name = legislature value = $LEGISLATURE$ }
			set_variable = { name = public_campaign_strength value = 1 }
			set_variable = { name = hidden_campaign_strength value = 1 }
			save_scope_as = scheme
			scheme_target_title = { save_scope_as = riding }
			var:legislature = { save_scope_as = legislature }
		}
		stand_for_legislator_election_effect = { LEGISLATURE = scope:legislature RIDING = scope:riding CAMPAIGN = scope:scheme }
		# Inform everyone who might be interested in this
		every_player = { # The AI does not need news events
			limit = {
				in_diplomatic_range = scope:candidate
				OR = {
					# We care about the candidate
					cares_about_character_trigger = { TARGET = scope:candidate }
					# We care about the riding
					cares_about_title_trigger = { TARGET = scope:riding }
				}
			}
			trigger_event = cotc_legislative_campaign_events.1
		}
	}
}

vote_for_challenger_strength_effect = {
	set_variable = { name = vote_strength value = "opinion(scope:activity.var:challenger)" }
	change_variable = { name = vote_strength subtract = "opinion(scope:activity.var:incumbent)" }
	if = {
		limit = {
			faith = {
				this = scope:activity.var:challenger.faith
				this != scope:activity.var:incumbent.faith
			}
		}
		change_variable = { name = vote_strength add = { value = ai_zeal min = 0 } }
	}
	else_if = {
		limit = {
			faith = {
				this != scope:activity.var:challenger.faith
				this = scope:activity.var:incumbent.faith
			}
		}
		change_variable = { name = vote_strength subtract = { value = ai_zeal min = 0 } }
	}
	if = {
		limit = {
			culture = {
				this = scope:activity.var:challenger.culture
				this != scope:activity.var:incumbent.culture
			}
		}
		change_variable = { name = vote_strength add = { value = ai_parochialism min = 0 } }
	}
	else_if = {
		limit = {
			culture = {
				this != scope:activity.var:challenger.culture
				this = scope:activity.var:incumbent.culture
			}
		}
		change_variable = { name = vote_strength subtract = { value = ai_parochialism min = 0 } }
	}
}
