select_initial_legislator_effect = {
	save_temporary_scope_as = temp_riding
	# Make a list of every possible candidate
	holder = {
		every_courtier = {
			limit = {
				eligible_legislator_candidate_trigger = yes
				initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
				legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
			}
			scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
		}
		every_pool_guest = {
			limit = {
				eligible_legislator_candidate_trigger = yes
				initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
				legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
			}
			scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
		}
	}
	every_county_province ?= {
		province_owner ?= {
			every_courtier = {
				limit = {
					eligible_legislator_candidate_trigger = yes
					initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
					legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
				}
				scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
			}
			every_pool_guest = {
				limit = {
					eligible_legislator_candidate_trigger = yes
					initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
					legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
				}
				scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
			}
		}
		every_province_domicile ?= {
			owner = {
				every_courtier = {
					limit = {
						eligible_legislator_candidate_trigger = yes
						initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
						legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
					}
					scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
				}
				every_pool_guest = {
					limit = {
						eligible_legislator_candidate_trigger = yes
						initial_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
						legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
					}
					scope:temp_riding = { add_to_variable_list = { name = candidates target = PREV } }
				}
			}
		}
	}
	# Elect the initial legislator
	elect_legislative_election_winner_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = scope:temp_riding }
	clear_variable_list = candidates
}
hold_legislative_election_effect = {
	# Grab the list of candidates
	if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_county } }
		every_in_list = {
			variable = county_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = county_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_duchy } }
		every_in_list = {
			variable = duchy_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = duchy_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_kingdom } }
		every_in_list = {
			variable = kingdom_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = kingdom_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_empire } }
		every_in_list = {
			variable = empire_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = empire_candidates
	}
	else_if = {
		limit = { $LEGISLATURE$.var:legislature_title = { tier = tier_hegemony } }
		every_in_list = {
			variable = hegemony_candidates
			save_temporary_scope_as = candidate
			PREV = { add_to_variable_list = { name = candidates target = PREV } }
		}
		clear_variable_list = hegemony_candidates
	}
	elect_legislative_election_winner_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = this }
	every_in_list = {
		variable = candidates
		remove_variable = legislative_campaign_score
	}
	clear_variable_list = candidates
}
elect_legislative_election_winner_effect = {
	$LEGISLATURE$ = { save_temporary_scope_as = temp_legislature }
	$RIDING$ = { save_temporary_scope_as = temp_riding }
	ordered_in_list = {
		variable = candidates
		limit = {
			eligible_legislator_candidate_trigger = yes
			legal_legislator_candidate_trigger = { LEGISLATURE = $LEGISLATURE$ }
		}
		order_by = legislator_total_score
		# This is a re-election
		if = {
			limit = {
				has_variable = riding
				has_variable = legislature
				var:riding ?= { this = $RIDING$ }
				var:legislature ?= { this = $LEGISLATURE$ }
			}
			reelect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
		}
		# This is an election to a different legislature
		else_if = {
			limit = {
				has_variable = riding
				has_variable = legislature
				NOT = { var:legislature = { this = $LEGISLATURE$ } }
			}
			remove_from_legislature_effect = yes
			elect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
		}
		else = { elect_as_legislator_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ } }
	}
}

	}
}

elect_as_legislator_effect = {
	add_to_legislature_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
	create_character_memory = { type = elected_legislator_memory }
	scope:new_memory = {
		set_variable = { name = legislature_title value = $LEGISLATURE$.var:legislature_title }
		set_variable = { name = riding value = $RIDING$ }
	}
	$LEGISLATURE$ = { save_scope_as = legislature }
	$RIDING$ = { save_scope_as = riding }
	trigger_event = { id = legislature_events.10 }
}
reelect_as_legislator_effect = {
	add_to_legislature_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
	create_character_memory = { type = reelected_legislator_memory }
	scope:new_memory = {
		set_variable = { name = legislature_title value = $LEGISLATURE$.var:legislature_title }
		set_variable = { name = riding value = $RIDING$ }
	}
	$LEGISLATURE$ = { save_scope_as = legislature }
	$RIDING$ = { save_scope_as = riding }
	trigger_event = { id = legislature_events.11 }
}
unelect_as_legislator_effect = {
	remove_from_legislature_effect = yes
	create_character_memory = { type = not_reelected_legislator_memory }
	scope:new_memory = { set_variable = { name = legislature_title value = $LEGISLATURE$.var:legislature_title } }
	$LEGISLATURE$ = { save_scope_as = legislature }
	$RIDING$ = { save_scope_as = riding }
	trigger_event = { id = legislature_events.12 }
}
add_to_legislature_effect = {
	$LEGISLATURE$ = {
		add_to_variable_list = {
			name = legislators
			target = PREV
		}
	}
	set_variable = { name = legislature value = $LEGISLATURE$ }
	set_variable = { name = riding value = $RIDING$ }
	# Incumbents should automatically stand for re-election
	stand_for_legislator_election_effect = { LEGISLATURE = $LEGISLATURE$ RIDING = $RIDING$ }
}
remove_from_legislature_effect = {
	var:legislature = { remove_list_variable = { name = legislators target = PREV } }
	remove_variable = legislature
	remove_variable = riding
}
vote_for_challenger_strength_effect = {
	set_variable = { name = vote_strength value = "opinion(scope:activity.var:challenger)" }
	change_variable = { name = vote_strength subtract = "opinion(scope:activity.var:incumbent)" }
	if = {
		limit = {
			faith = {
				this = scope:activity.var:challenger.faith
				this != scope:activity.var:incumbent.faith
			}
		}
		change_variable = { name = vote_strength add = { value = ai_zeal min = 0 } }
	}
	else_if = {
		limit = {
			faith = {
				this != scope:activity.var:challenger.faith
				this = scope:activity.var:incumbent.faith
			}
		}
		change_variable = { name = vote_strength subtract = { value = ai_zeal min = 0 } }
	}
	if = {
		limit = {
			culture = {
				this = scope:activity.var:challenger.culture
				this != scope:activity.var:incumbent.culture
			}
		}
		change_variable = { name = vote_strength add = { value = ai_parochialism min = 0 } }
	}
	else_if = {
		limit = {
			culture = {
				this != scope:activity.var:challenger.culture
				this = scope:activity.var:incumbent.culture
			}
		}
		change_variable = { name = vote_strength subtract = { value = ai_parochialism min = 0 } }
	}
}
