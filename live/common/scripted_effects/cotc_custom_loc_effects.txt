set_interest_group = {
	set_variable = { name = interest_group_title value = $TITLE$ days = 5 }
	if = {
		limit = { $TITLE$ = { owner = { government_has_flag = government_is_disorganized } } }
		set_variable = { name = interest_group_title_format value = flag:adjective days = 5 }
		set_variable = { name = interest_group_name_format value = flag:title_type days = 5 }
		set_disorganized_interest_group = yes
	}
	else = {
		# Where does the name of the title go?
		random_list = {
			1 = { set_variable = { name = interest_group_title_format value = flag:adjective days = 5 } }
			1 = { set_variable = { name = interest_group_title_format value = flag:name days = 5 } }
		}
		# What kind of interest group is endorsing us?
		random_list = {
			1 = { # Business interest group
				modifier = { add = stewardship }
				modifier = { factor = { value = faith.ideology_interventionism multiply = -1 } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_business_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_business_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = adjective_type }
						set_business_interest_group_plural = yes
					}
				}
			}
			1 = { # Labor interest group
				modifier = { add = { value = piety_level multiply = 3.5 } }
				modifier = { factor = { value = faith.ideology_interventionism } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_labor_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_labor_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = adjective_type }
						set_labor_interest_group_plural = yes
					}
				}
			}
			1 = { # Military interest group
				modifier = { add = martial }
				modifier = { factor = { value = faith.ideology_hawkishness } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_military_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_military_interest_group = yes
					}
				}
			}
			1 = { # Pacifist interest group
				modifier = { add = diplomacy }
				modifier = { factor = { value = faith.ideology_hawkishness multiply = -1 } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_pacifist_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_pacifist_interest_group = yes
					}
				}
			}
			1 = { # Academic interest group
				modifier = { add = learning }
				modifier = { factor = { value = faith.ideology_intellectualism } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_academic_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_academic_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = adjective_type }
						set_academic_interest_group_plural = yes
					}
				}
			}
			# 1 = { # Transhumanist interest group
				# trigger = { faith = { is_transhumanist_ideology = yes } }
				# modifier = { add = learning }
				# random_list = {
					# 1 = {
						# set_variable = { name = interest_group_name_format value = type_group }
						# set_transhumanist_interest_group = yes
					# }
					# 1 = {
						# set_variable = { name = interest_group_name_format value = group_type }
						# set_transhumanist_interest_group = yes
					# }
				# }
			# }
			1 = { # Conservative interest group
				modifier = { add = { value = piety_level multiply = 3.5 } }
				modifier = { factor = { value = faith.ideology_intellectualism multiply = -1 } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_transhumanist_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_transhumanist_interest_group = yes
					}
				}
			}
			1 = { # Religious interest group
				modifier = { add = { value = piety_level multiply = 3.5 } }
				modifier = { factor = { value = faith.ideology_mysticism } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_religious_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_religious_interest_group = yes
					}
				}
			}
			1 = { # Libertarian interest group
				modifier = { add = { value = piety_level multiply = 3.5 } }
				modifier = { factor = { value = faith.ideology_authoritarianism multiply = -1 } }
				random_list = {
					1 = {
						set_variable = { name = interest_group_name_format value = type_group }
						set_libertarian_interest_group = yes
					}
					1 = {
						set_variable = { name = interest_group_name_format value = group_type }
						set_libertarian_interest_group = yes
					}
				}
			}
			1 = { # Authoritarian interest group
				modifier = { add = { value = piety_level multiply = 3.5 } }

						set_authoritarian_interest_group = yes
					}
				}
			}
		}
	}
}

set_labor_interest_group = {
	random_list = {
		1 = { set_variable = { name = interest_group_noun value = flag:union days = 5 } }
		1 = { set_variable = { name = interest_group_noun value = flag:syndicate days = 5 } }
		1 = { set_variable = { name = interest_group_noun value = flag:guild days = 5 } }
		1 = { set_variable = { name = interest_group_noun value = flag:federation days = 5 } }
		1 = { set_variable = { name = interest_group_noun value = flag:alliance days = 5 } }
	}
	random_list = {
		5 = { set_variable = { name = interest_group_type value = flag:labor days = 5 } }
		5 = { set_variable = { name = interest_group_type value = flag:workers days = 5 } }
		1 = { set_variable = { name = interest_group_type value = flag:steelworkers days = 5 } }
		1 = {
			trigger = {
				var:interest_group_title ?= {
					OR = {
						tier > tier_barony
						has_minerals_trigger = yes
					}
				}
			}
			set_variable = { name = interest_group_type value = flag:miners days = 5 }
		}
		1 = { set_variable = { name = interest_group_type value = flag:shipbreakers days = 5 } }
		1 = { set_variable = { name = interest_group_type value = flag:dockworkers days = 5 } }
		1 = {
			trigger = {
				var:interest_group_title ?= {
					OR = {
						tier > tier_barony
						has_breathable_atmosphere_trigger = yes
					}
				}
			}
			set_variable = { name = interest_group_type value = flag:farmers days = 5 }
		}
	}
}
