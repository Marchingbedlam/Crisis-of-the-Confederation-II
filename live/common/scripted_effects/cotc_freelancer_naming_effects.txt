##################################################
# COTC - Freelancer group renaming
#
# Called from each Charter law's on_pass in 00_realm_laws.txt, and from
# cotc_freelancer_posture_changed_effect when Posture changes.
#
# Charter picks the VOCABULARY (CotcRoamingGroupNoun / CotcSettledGroupNoun)
# Posture picks the TEMPLATE SHAPE:
#   roaming r01-r05 - Iron Corsairs, Hedd Runners
#   settled s01-s04 - Hedd Survey, Ceres Combine
#
# Every Freelancer holds the default Charter by the time create_landless_adventurer_title_effect runs
# set_title_name_dynamic resolves it again
##################################################

cotc_freelancer_rename_group_effect = {
	if = {
		limit = {
			government_has_flag = government_is_landless_adventurer
			exists = primary_title
			# Players who have settled on a name will keep it.
			NOT = { has_variable = cotc_group_name_locked }
		}
		house ?= { save_scope_as = cotc_group_house }

		if = {
			limit = { has_realm_law = posture_settled }
			domicile ?= {
				domicile_location ?= { save_scope_as = cotc_group_location }
			}
		}
		else = {
			location ?= { save_scope_as = cotc_group_location }
		}

		if = {
			limit = { has_realm_law_flag = static_camp_posture }
			# SETTLED
			random_list = {
				10 = {
					trigger = { exists = scope:cotc_group_house }
					primary_title = { set_title_name_dynamic = cotc_group_name_s01 }
				}
				10 = {
					trigger = { exists = scope:cotc_group_location }
					primary_title = { set_title_name_dynamic = cotc_group_name_s02 }
				}
				10 = {
					trigger = { exists = scope:cotc_group_location }
					primary_title = { set_title_name_dynamic = cotc_group_name_s03 }
				}
				10 = {
					primary_title = { set_title_name_dynamic = cotc_group_name_s04 }
				}
			}
		}
		else = {
			# ROAMING
			random_list = {
				10 = {
					primary_title = { set_title_name_dynamic = cotc_group_name_r01 }
				}
				10 = {
					primary_title = { set_title_name_dynamic = cotc_group_name_r02 }
				}
				10 = {
					trigger = { exists = scope:cotc_group_house }
					primary_title = { set_title_name_dynamic = cotc_group_name_r03 }
				}
				10 = {
					trigger = { exists = scope:cotc_group_location }
					primary_title = { set_title_name_dynamic = cotc_group_name_r04 }
				}
				10 = {
					primary_title = { set_title_name_dynamic = cotc_group_name_r05 }
				}
			}
		}
	}
}

# Stops the group being renamed on any later Charter or Posture change
cotc_freelancer_lock_group_name_effect = {
	set_variable = cotc_group_name_locked
}
