##################################################
# COTC - Freelancer group renaming
##################################################

# Root = the boss
# The AI takes the new name. The player is asked first.
cotc_freelancer_rename_group_effect = {
	if = {
		limit = {
			government_has_flag = government_is_landless_adventurer
			exists = primary_title
		}
		custom_tooltip = cotc_group_renamed_tt
		hidden_effect = {
			if = {
				limit = { is_ai = yes }
				cotc_freelancer_apply_group_name_effect = yes
			}
			else = { trigger_event = cotc_freelancer_events.0001 }
		}
	}
}

# Root = the boss. Rolls a name to suit the current Charter and Posture and
# applies it. Called by the AI branch above and by the player event's accept.
cotc_freelancer_apply_group_name_effect = {
	house ?= { save_scope_as = cotc_group_house }

	if = {
		limit = { has_realm_law = posture_settled }
		domicile ?= { domicile_location ?= { cotc_save_group_location_effect = yes } }
	}
	else = {
		location ?= { cotc_save_group_location_effect = yes }
	}

	if = {
		limit = { has_realm_law_flag = static_camp_posture }
		# SETTLED
		random_list = {
			10 = {
				trigger = { exists = scope:cotc_group_house }
				primary_title = { set_title_name_dynamic = cotc_group_name_s01 }
			}
			10 = {
				trigger = { exists = scope:cotc_group_location }
				primary_title = { set_title_name_dynamic = cotc_group_name_s02 }
			}
			10 = {
				trigger = { exists = scope:cotc_group_location }
				primary_title = { set_title_name_dynamic = cotc_group_name_s03 }
			}
			10 = {
				primary_title = { set_title_name_dynamic = cotc_group_name_s04 }
			}
		}
	}
	else = {
		# ROAMING
		random_list = {
			10 = {
				primary_title = { set_title_name_dynamic = cotc_group_name_r01 }
			}
			10 = {
				trigger = { exists = scope:cotc_group_house }
				primary_title = { set_title_name_dynamic = cotc_group_name_r02 }
			}
		}
	}
}


# Province scope, saves the county name
cotc_save_group_location_effect = {
	if = {
		limit = { NOT = { geographical_region = unsuitable_placename_region } }
		county ?= { save_scope_as = cotc_group_location }
	}
}
