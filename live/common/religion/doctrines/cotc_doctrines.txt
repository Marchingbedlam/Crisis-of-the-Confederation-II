# This determines how much authority rulers are supposed to have
doctrine_authority = {
	group = "politics"
	
	# The ruler holds absolute power and there is no distinction between ruler and state
	doctrine_despotism = {
		piety_cost = {
			value = faith_doctrine_cost_mid
			if = {
				limit = { has_doctrine = doctrine_despotism }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_separation_of_powers }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_parliamentarianism }
				multiply = faith_changed_doctrine_cost_mult_three_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_anarchism }
				multiply = faith_changed_doctrine_cost_mult_four_step
			}
		}
		parameters = {
			prefers_dictatorships = yes
			dictatorial_revolutions = yes
		}
		character_modifier = {
			tyranny_gain_mult = -0.5
			direct_vassal_opinion = -10
			opinion_of_liege = 10
		}
	}
	# The ruler is a dictator but remains part of a greater legal state
	doctrine_dictatorship = {
		piety_cost = {
			value = faith_doctrine_cost_mid
			if = {
				limit = { has_doctrine = doctrine_dictatorship }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_parliamentarianism }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_anarchism }
				multiply = faith_changed_doctrine_cost_mult_three_step
			}
		}
		parameters = {
			prefers_dictatorships = yes
			dictatorial_revolutions = yes
		}
	}
	# The ruler is a co-equal governor with some form of parliament
	doctrine_separation_of_powers = {
		piety_cost = {
			value = faith_doctrine_cost_mid
			if = {
				limit = { has_doctrine = doctrine_separation_of_powers }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_despotism }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_anarchism }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
		}
		parameters = {
			prefers_presidential_system = yes
			presidential_revolutions = yes
		}
	}
	# The ruler must defer to some form of parliament which votes on important decisions; no one rules alone
	doctrine_parliamentarianism = {
		piety_cost = {
			value = faith_doctrine_cost_mid
			if = {
				limit = { has_doctrine = doctrine_parliamentarianism }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_despotism }
				multiply = faith_changed_doctrine_cost_mult_three_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_dictatorship }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
		}
		parameters = {
			prefers_parliamentary_system = yes
			parliamentary_revolutions = yes
		}
		character_modifier = {
			tyranny_gain_mult = 0.5
			direct_vassal_opinion = 10
			county_opinion = -10
		}
	}
	
	# The ruler should be almost entirely a figurehead, with real power in the hands of the citizens
	doctrine_anarchism = {
		piety_cost = {
			value = faith_doctrine_cost_high
			if = {
				limit = { has_doctrine = doctrine_anarchism }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_despotism }
				multiply = faith_changed_doctrine_cost_mult_four_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_dictatorship }
				multiply = faith_changed_doctrine_cost_mult_three_step
			}
			else_if = {
				limit = { has_doctrine = doctrine_separation_of_powers }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
		}
		parameters = {
			prefers_anarchy = yes
			anarchist_revolutions = yes
		}
		character_modifier = {
			tyranny_gain_mult = 1.0
			direct_vassal_opinion = 25
			county_opinion = -25
		}
	}
}

# Criminal Activities for criminal ideologies
doctrine_extortion = {
	group = "crimes"
	is_available_on_create = { has_doctrine = doctrine_criminal }
	
	doctrine_extortion_allowed = {
		piety_cost = {
			value = faith_doctrine_cost_medium
			if = {
				limit = { has_doctrine = doctrine_extortion_allowed }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_extortion_forbidden }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
		}
		parameters = {
			can_extort_provinces = yes
			can_extort_rulers = yes
			can_issue_criminal_loans = yes
		}
	}
	doctrine_extortion_rulers_only = {
		piety_cost = {
			value = faith_doctrine_cost_medium
			if = {
				limit = { has_doctrine = doctrine_extortion_rulers_only }
				multiply = faith_unchanged_doctrine_cost_mult
			}
		}
		parameters = {
			can_extort_provinces = no
			can_extort_rulers = yes
			can_issue_criminal_loans = yes
		}
	}
	doctrine_extortion_forbidden = {
		piety_cost = {
			value = faith_doctrine_cost_medium
			if = {
				limit = { has_doctrine = doctrine_extortion_forbidden }
				multiply = faith_unchanged_doctrine_cost_mult
			}
			else_if = {
				limit = { has_doctrine = doctrine_extortion_allowed }
				multiply = faith_changed_doctrine_cost_mult_two_step
			}
		}
		can_pick = {
			custom_description = {
				text = incompatible_doctrine_must_do_some_crime_trigger
				OR = {
					doctrine:doctrine_piracy_allowed = { is_in_list = selected_doctrines }
					doctrine:doctrine_smuggling_allowed = { is_in_list = selected_doctrines }
					doctrine:doctrine_illegal_technology_allowed = { is_in_list = selected_doctrines }
				}
			}
		}
		parameters = {
			can_extort_provinces = no
			can_extort_rulers = no
			can_issue_criminal_loans = no
		}
	}
}
