GetPoliticalCampaignRidingName = {
	type = character
	text = {
		setup_scope = {
			random_scheme = {
				limit = { scheme_type = legislative_election_campaign }
				var:riding ?= { save_scope_as = loc_riding }
			}
		}
		localization_key = "[loc_riding.GetName]"
	}
}
