# Gives us the character's given and family name, in an appropriate
# order for their culture
GetFullPersonalName = {
	type = character
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = {
			NOT = { culture = terran } # Common uses western name order
			OR = {
				culture = { has_name_list = name_list_cotc_japanese }
				culture = { has_name_list = name_list_cotc_chinese }
				culture = { has_name_list = name_list_cotc_vietnamese }
			}
		}
		localization_key = personal_name_eastern_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		localization_key = personal_name_western_custom_loc
	}
}
