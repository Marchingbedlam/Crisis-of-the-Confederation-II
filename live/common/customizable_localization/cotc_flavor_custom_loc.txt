# Gives us the character's given and family name, in an appropriate
# order for their culture
GetFullPersonalName = {
	type = character
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = {
			NOT = { culture = terran } # Common uses western name order
			OR = {
				culture = { has_name_list = name_list_cotc_japanese }
				culture = { has_name_list = name_list_cotc_chinese }
				culture = { has_name_list = name_list_cotc_vietnamese }
			}
		}
		localization_key = personal_name_eastern_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		localization_key = personal_name_western_custom_loc
	}
}
# Gives us the name of a character and also their highest political office
GetBestName = {
	type = character
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = { is_diarch = yes }
		localization_key = title_diarch_of_state_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = { is_landed = yes }
		localization_key = title_ruler_of_state_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = { is_councillor = yes }
		localization_key = title_councillor_of_state_custom_loc
	}
	text = {
		setup_scope = {
			save_scope_as = custom_loc_character
			var:legislature ?= { save_scope_as = custom_loc_legislature }
			var:riding ?= { save_scope_as = custom_loc_riding }
		}
		localization_key = title_legislator_of_riding_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		localization_key = "[custom_loc_character.Custom('GetFullPersonalName')]"
	}
}
# A brief description of this character's spouse or spouses
GetSpousesDesc = {
	type = character
	# One Spouse
	text = {
		setup_scope = {
			save_scope_as = custom_loc_character
			primary_spouse ?= { save_scope_as = custom_loc_spouse }
		}
		trigger = {
			any_spouse = {
				limit = { is_alive = yes }
				count = 1
			}
		}
		localization_key = one_spouse_custom_loc
	}
	# Multiple spouses
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = {
			any_spouse = {
				limit = { is_alive = yes }
				count = 2
			}
		}
		localization_key = two_spouses_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = {
			any_spouse = {
				limit = { is_alive = yes }
				count = 3
			}
		}
		localization_key = three_spouses_custom_loc
	}
	text = {
		setup_scope = { save_scope_as = custom_loc_character }
		trigger = {
			any_spouse = {
				limit = { is_alive = yes }
				count > 3
			}
		}
		localization_key = many_spouses_custom_loc
	}
}

# What happens to this character's family?
GetDeadCharacterConsequences = {
	type = character
	
	# Who inherits the throne of a monarchy?
	# Did this character have any surviving family?
	text = {
		setup_scope = { var:dead_character = { save_scope_as = custom_loc_character } }
		trigger = {
			scope:custom_loc_character = {
				OR = {
					any_spouse = { is_alive = yes }
					any_child = { is_alive = yes }
				}
			}
		}
		localization_key = "[custom_loc_character.Custom('GetDeadCharacterFamily')]"
	}
}
GetDeadCharacterFamily = {
	type = character
	text = {
		setup_scope = {
			var:dead_character = { save_scope_as = custom_loc_character }
			var:death_title_heir ?= { save_scope_as = custom_loc_heir }
		}
		trigger = {
			exists = scope:custom_loc_heir
			scope:custom_loc_character = { government_has_flag = government_is_aristocratic }
		}
		localization_key = throne_will_pass_to_custom_loc
	}
	# Had both spouses and children
	text = {
		setup_scope = { var:dead_character = { save_scope_as = custom_loc_character } }
		trigger = {
			scope:custom_loc_character = {
				any_spouse = { is_alive = yes }
				any_child = { is_alive = yes }
			}
		}
		localization_key = survived_by_spouses_and_children_custom_loc
	}
	# Had only spouses
	text = {
		setup_scope = { var:dead_character = { save_scope_as = custom_loc_character } }
		trigger = { scope:custom_loc_character = { any_spouse = { is_alive = yes } } }
		localization_key = survived_by_spouses_custom_loc
	}
	# Had only children
	text = {
		setup_scope = { var:dead_character = { save_scope_as = custom_loc_character } }
		trigger = { scope:custom_loc_character = { any_child = { is_alive = yes } } }
		localization_key = survived_by_children_custom_loc
	}
	# Had no-one
	text = { localization_key = blank_custom_loc }
}

		trigger = { var:dead_character_prestige = 5 }
		localization_key = legendary_death_custom_loc
	}
	# Empire or Hegemony

			var:dead_character = { save_scope_as = custom_loc_character }
			var:news_source_title = { save_scope_as = custom_loc_title }
		}
		trigger = { scope:custom_loc_title ?= { tier >= tier_empire } }
		localization_key = ruler_death_custom_loc
	}
	# Exalted
	text = {
		setup_scope = { var:dead_character = { save_scope_as = custom_loc_character } }
