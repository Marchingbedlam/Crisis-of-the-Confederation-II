GetBillType = {
	type = activity
	text = {
		trigger = {
			OR = {
				has_activity_type = tax_bill_activity
				has_activity_type = liege_tax_bill_activity
			}
		}
		localization_key = "tax_bill_custom_loc"
	}
	text = {
		trigger = { has_activity_type = budget_bill_activity }
		localization_key = "budget_bill_custom_loc"
	}
	text = {
		trigger = { has_activity_type = no_confidence_motion_activity }
		localization_key = "no_confidence_motion_custom_loc"
	}
	text = {
		localization_key = "bill_custom_loc"
	}
}
GetBillProvisions = {
	type = activity
	text = {
		setup_scope = {
			this = { save_scope_as = bill }
			var:title = { save_scope_as = bill_title }
		}
		trigger = {
			OR = {
				has_activity_type = tax_bill_activity
				has_activity_type = liege_tax_bill_activity
			}
		}
		localization_key = "tax_bill_provisions_custom_loc"
	}
	text = {
		setup_scope = {
			this = { save_scope_as = bill }
			var:title = { save_scope_as = bill_title }
		}
		trigger = { has_activity_type = budget_bill_activity }
		localization_key = "budget_bill_provisions_custom_loc"
	}
	text = {
		setup_scope = {
			this = { save_scope_as = bill }
			var:title ?= { save_scope_as = bill_title }
			var:ruler ?= { save_scope_as = bill_ruler }
			var:incumbent ?= { save_scope_as = bill_incumbent }
			var:challenger ?= { save_scope_as = bill_challenger }
		}
		trigger = { has_activity_type = no_confidence_motion_activity }
		localization_key = "no_confidence_provisions_custom_loc"
	}
}
GetBillEarmarks = {
	type = activity
	text = {
		setup_scope = { save_scope_value_as = { name = pork_value value = var:pork } }
		trigger = {
			trigger_if = {
				limit = { has_variable = pork }
				var:pork > 0
			}
			trigger_else = {
				always = no
			}
		}
		localization_key = "bill_earmarks_custom_loc"
	}
	text = { localization_key = "blank_custom_loc" }
}
GetPostBillEarmarks = {
	type = character
	text = {
		setup_scope = { save_scope_value_as = { name = pork_value value = pork_value } }
		trigger = {
			trigger_if = {
				limit = { exists = scope:pork_value }
				scope:pork_value > 0
			}
			else = { always = no }
		}
		localization_key = "bill_post_earmarks_custom_loc"
	}
	text = { localization_key = "blank_custom_loc" }
}
