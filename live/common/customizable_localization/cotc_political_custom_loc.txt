GetBillType = {
	type = activity
	text = {
		trigger = { has_activity_type = budget_bill_activity }
		localization_key = "budget_bill_custom_loc"
	}
	text = {
		localization_key = "bill_custom_loc"
	}
}
GetBillProvisions = {
	type = activity
	text = {
		setup_scope = {
			this = { save_scope_as = bill }
			var:title = { save_scope_as = bill_title }
		}
		trigger = { has_activity_type = budget_bill_activity }
		localization_key = "budget_bill_provisions_custom_loc"
	}
}
GetBillSellDifficulty = {
	type = character
	text = {
		trigger = { scope:activity.bill_sell_difficulty = 10 }
		localization_key = "very_easy_to_sell_bill_custom_loc"
	}
	text = {
		trigger = { scope:activity.bill_sell_difficulty < 15 }
		localization_key = "easy_to_sell_bill_custom_loc"
	}
	text = {
		trigger = { scope:activity.bill_sell_difficulty < 20 }
		localization_key = "moderate_difficulty_to_sell_bill_custom_loc"
	}
	text = {
		trigger = { scope:activity.bill_sell_difficulty < 25 }
		localization_key = "hard_to_sell_bill_custom_loc"
	}
	text = { localization_key = "very_hard_to_sell_bill_custom_loc" }
}
