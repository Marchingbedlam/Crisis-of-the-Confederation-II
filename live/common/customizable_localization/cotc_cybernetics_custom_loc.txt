GetDNIUpgrades = {
	type = artifact
	
	text = {
		setup_scope = { save_scope_as = desc_implant }
		trigger = {
			OR = {
				has_artifact_feature_group = iaei
				has_artifact_feature_group = pci
				has_artifact_feature_group = taccomp
				has_artifact_feature_group = pluto
				has_artifact_feature_group = thoth
				has_artifact_feature_group = neurolink
				has_artifact_feature_group = amam
			}
		}
		localization_key = dni_has_upgrades_custom_loc
	}
	text = {
		localization_key = dni_has_no_upgrades_custom_loc
		fallback = yes
	}
}
