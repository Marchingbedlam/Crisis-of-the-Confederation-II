### Form the Barnard Accord ###
# Offered to any ruler holding land in the four core duchies of the Horsehead Nebula, and takeable once all four are inside their realm.
form_barnard_accord_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}
	desc = form_barnard_accord_decision_desc
	selection_tooltip = form_barnard_accord_decision_tooltip
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 120
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		is_ruler = yes
		is_playable_character = yes
		is_confederation_member = no
		# The Accord has not been signed yet. Vacating it makes the decision available again.
		NOT = { exists = title:k_barnard_accord.holder }
		# Shown to anyone with land in one of the four founding duchies, vassals included.
		any_sub_realm_county = {
			OR = {
				de_jure_liege = title:d_arjona
				de_jure_liege = title:d_galilei
				de_jure_liege = title:d_miltia
				de_jure_liege = title:d_abraxas
			}
		}
	}

	is_valid = {
		completely_controls = title:d_arjona
		completely_controls = title:d_galilei
		completely_controls = title:d_miltia
		completely_controls = title:d_abraxas
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = {
			value = 400
			if = {
				limit = {
					has_treasury = yes
				}
				multiply = 0
			}
		}
		treasury = {
			value = 400
			if = {
				limit = {
					has_treasury = no
				}
				multiply = 0
			}
		}
		prestige = 750
	}

	effect = {
		form_barnard_accord_scripted_effect = yes
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
