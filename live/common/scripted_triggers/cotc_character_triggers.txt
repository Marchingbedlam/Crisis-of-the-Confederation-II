lives_in_capital_trigger = {
	court_owner ?= { capital_province ?= { county ?= { this = $TARGET$.capital_county } } }

	trigger_if = {
		limit = { is_ruler = yes }
		government_has_flag = government_is_federal
	}
	trigger_else = {
		liege = { government_has_flag = government_is_federal }
	}
}
