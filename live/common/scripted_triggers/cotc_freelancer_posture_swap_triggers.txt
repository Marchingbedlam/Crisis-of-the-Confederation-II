##################################################
# COTC - Posture swap triggers
##################################################

realm_law_use_cotc_posture = {
	government_has_flag = government_is_landless_adventurer
}

cotc_posture_no_cooldown_trigger = {
	custom_tooltip = {
		subject = root
		text = cotc_posture_cooldown_tt
		NOT = { has_variable = cotc_posture_cooldown }
	}
}

# Changing Posture re-anchors the Base, so it cannot happen mid-journey
cotc_posture_can_switch_now_trigger = {
	is_travelling = no
	is_in_army = no
	is_imprisoned = no
	exists = domicile
	domicile = { is_domicile_type = camp }
}
