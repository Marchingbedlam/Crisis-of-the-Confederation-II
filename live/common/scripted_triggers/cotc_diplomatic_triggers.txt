is_diplomatic_ally_of = { # Are scope and target diplomatic allies?
	OR = {
		is_allied_to = $TARGET$
		AND = {
			exists = confederation
			$TARGET$ = { is_member_of = PREV.confederation }
		}
	}
}
is_potential_threat_to = { # Is scope a potential threat to target?
	NOT = { is_diplomatic_ally_of = $TARGET$ }
	OR = {
		has_any_cb_on = $TARGET$
		is_at_war_with = $TARGET$
	}
}
is_threatened_trigger = { # Do any of our neighbors threaten us?
	save_temporary_scope_as = threat_target
	any_neighboring_top_liege_realm_owner = {
		limit = { is_potential_threat_to = { TARGET = scope:threat_target } }
		threat_to >= 20
	}
}
