# SECRET BRIBE
secret_official_bribe_is_valid_trigger = {
	# Can't bribe yourself.
	# Probably.
	NOT = { $OWNER$ = $TARGET$ }
}
# Official bribes are always crimes except in despotic regimes
secret_official_bribe_is_criminal_trigger = {
	NOR = {
		$OWNER$ = { government_has_flag = government_is_despotic }
		$TARGET$ = { government_has_flag = government_is_despotic }
	}
}
# SECRET RESEARCHED INNOVATION
# Researching some innovations is shunned by some ideologies
secret_researched_innovation_is_shunned_trigger = {
	OR = {
		AND = {
			is_target_in_variable_list = { name = flags target = flag:military }
			$OWNER$ = {
				OR = {
					rite = { rite_hawkishness < 0 }
					any_liege_or_above = { rite = { rite_hawkishness < 0 } }
				}
			}
		}
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = cybernetics }
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = gene_modding }
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = cloning }
	}
}
secret_innovation_flag_shunned = {
	is_target_in_variable_list = { name = flags target = flag:$INNOVATION_FLAG$ }
	$OWNER$ = {
		OR = {
			rite = { rite_has_parameter = hates_$INNOVATION_FLAG$ }
			any_liege_or_above = { rite = { rite_has_parameter = hates_$INNOVATION_FLAG$ } }
		}
	}
}
# SECRET CYBERSURGERY
secret_cybersurgery_is_criminal_trigger = {
	trigger_if = {
		limit = { has_variable = implant }
		var:implant = { save_temporary_scope_as = temp_implant }
		$OWNER$ = { is_implant_illegal_for_character = { IMPLANT = scope:temp_implant } }
	}
}
secret_cybersurgery_is_shunned_trigger = {
	$OWNER$ = {
		OR = {
			rite = { rite_has_parameter = hates_cybernetics }
			any_liege_or_above = { rite = { rite_has_parameter = hates_cybernetics } }
		}
	}
}
