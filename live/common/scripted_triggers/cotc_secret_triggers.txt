# SECRET BRIBE
secret_official_bribe_is_valid_trigger = {
	# Can't bribe yourself.
	# Probably.
	NOT = { $OWNER$ = $TARGET$ }
}
# Official bribes are always crimes except in despotic regimes
secret_official_bribe_is_criminal_trigger = {
	NOR = {
		$OWNER$ = { has_government_flag = government_is_despotic }
		$TARGET$ = { has_government_flag = government_is_despotic }
	}
}
# SECRET RESEARCHED INNOVATION
# Researching some innovations is shunned by some ideologies
secret_researched_innovation_is_shunned_trigger = {
	OR = {
		AND = {
			is_target_in_variable_list = { name = flags target = flag:military }
			$OWNER$ = {
				OR = {
					faith = { ideology_hawkishness < 0 }
					any_liege_or_above = { faith = { ideology_hawkishness < 0 } }
				}
			}
		}
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = cybernetics }
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = gene_modding }
		secret_innovation_flag_shunned = { OWNER = $OWNER$ INNOVATION_FLAG = cloning }
	}
}
secret_innovation_flag_shunned = {
	is_target_in_variable_list = { name = flags target = flag:$INNOVATION_FLAG$ }
	$OWNER$ = {
		OR = {
			faith = { has_doctrine_parameter = hates_$INNOVATION_FLAG$ }
			any_liege_or_above = { faith = { has_doctrine_parameter = hates_$INNOVATION_FLAG$ } }
		}
	}
}
