# Triggers that tell us various things about planets/celestial objects/holdings

# Returns whether a target holding is a planet or not
is_planet_trigger = {
	title_province = {
		OR = {
			terrain = atmospheric
			terrain = barren
			terrain = frozen
			terrain = corrosive
			terrain = gas_giant
			terrain = frozen_giant
		}
	}
}
# Returns planets where someone could actually live on the surface
is_terrestrial_planet_trigger = {
	title_province = {
		OR = {
			terrain = atmospheric
			terrain = barren
			terrain = frozen
			terrain = corrosive
		}
	}
}
# Returns planets with water (not necessarily surface water!)
has_water_trigger = {
	OR = {
		terrain = atmospheric
		terrain = frozen
	}
}
