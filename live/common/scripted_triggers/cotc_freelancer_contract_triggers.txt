# Root = a candidate contract, OTHER = an offer already taken.
cotc_local_news_distinct_trigger = {
	trigger_if = {
		limit = {
			has_variable = cotc_news_type
			exists = $OTHER$
			$OTHER$ = { has_variable = cotc_news_type }
		}
		NOT = { var:cotc_news_type = $OTHER$.var:cotc_news_type }
	}
}
