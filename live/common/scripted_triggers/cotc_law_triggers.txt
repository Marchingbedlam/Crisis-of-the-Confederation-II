realm_government_is_democratic_trigger = {
	custom_tooltip = {
		text = succession_laws_must_have_democratic_government
		OR = {
			government_has_flag = government_is_democratic
			var:secondary_legitimacy ?= { this = flag:democratic }
		}
	}
}

can_have_popular_elective_succession_law_trigger = {
	government_has_mechanic = administrative
	realm_government_is_democratic_trigger = yes
}
can_keep_popular_elective_succession_law_trigger = {
	trigger_if = {
		limit = { has_realm_law = popular_elective_succession_law }
		can_have_popular_elective_succession_law_trigger = yes
	}
}
can_change_popular_elective_succession_law_trigger = { always = yes }

can_have_military_appointment_succession_law_trigger = {
	government_has_mechanic = administrative
	OR = {
		AND = {
			government_has_flag = government_is_military
			realm_government_is_meritocratic_trigger = yes
		}
		vassal_contract_has_flag = military_subordinate
	}
}
