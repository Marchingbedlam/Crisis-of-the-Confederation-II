cloning_possible_trigger = {
	character_has_tech = { TECH = culture_innovation:innovation_cloning }
}

single_donor_cloning_allowed_trigger = {
	cloning_possible_trigger = yes
	$DONOR$ = { is_alive = yes }
}
double_donor_cloning_allowed_trigger = {
	cloning_possible_trigger = yes
	$FIRST_DONOR$ = { is_alive = yes }
	$SECOND_DONOR$ = { is_alive = yes }
}
clone_can_be_male_trigger = {
	OR = { # This is a matter of chromosomal sex, so we check the game-mechanical gender directly
		scope:first_donor ?= { is_female = no }
		scope:second_donor ?= { is_female = no }
	}
}
clone_can_be_female_trigger = {
	OR = { # This is a matter of chromosomal sex, so we check the game-mechanical gender directly
		scope:first_donor ?= { is_female = yes }
		scope:second_donor ?= { is_female = yes }
	}
}
