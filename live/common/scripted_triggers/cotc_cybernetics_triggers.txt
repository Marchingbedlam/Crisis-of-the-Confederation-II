# General checks
is_cybernetic_implant = {
	OR = {
		artifact_type = neural_implant
		artifact_type = dermal_implant
		artifact_type = internal_implant
		artifact_type = monocular_implant
		artifact_type = binocular_implant
		artifact_type = artificial_arm
		artifact_type = artificial_leg
	}
}

	character_has_tech = { TECH = culture_innovation:innovation_direct_neural_interface }
}

}

already_has_artificial_arm_trigger = {
	trigger_if = {
		limit = {
			OR = {
				has_personal_tenet_flag = wants_implants
				has_trait = armless
			}
		}
		any_equipped_character_artifact = {
			artifact_type = artificial_arm
			count = 2
		}
	}
	trigger_else = { any_equipped_character_artifact = { artifact_type = artificial_arm } }
}
already_has_artificial_leg_trigger = {
	trigger_if = {
		limit = {
			OR = {
				has_personal_tenet_flag = wants_implants
				has_trait = legless
			}
		}
		any_equipped_character_artifact = {
			artifact_type = artificial_leg
			count = 2
		}
	}
	trigger_else = { any_equipped_character_artifact = { artifact_type = artificial_leg } }
}
might_want_implant_gift_trigger = {
	$TYPE$_self_score >= 10
	already_has_$TYPE$_trigger = no
	opinion = { target = PREV value > 0 }
	reverse_opinion = { target = PREV value > 0 }
	NOT = { rite = { rite_has_parameter = hates_cybernetics } }
	NOT = { has_personal_tenet_flag = abstains_from_cybernetics }
	NOT = { any_character_artifact = { artifact_type = $TYPE$ } }
}

# Legality
is_implant_illegal_in_realm_trigger = {
	# Later this will be a law but for now we'll tie it directly to the ruler's ideology
	OR = {
		rite = { rite_has_parameter = hates_cybernetics }
		primary_title = { state_rite ?= { rite_has_parameter = hates_cybernetics } }
	}
	trigger_if = { # This is a placeholder for later making different implants differently legal
		limit = { exists = $IMPLANT$ }
		always = yes
	}
}
is_implant_illegal_for_character = {
	OR = {
		AND = {
			is_ruler = yes
			is_implant_illegal_in_realm_trigger = { IMPLANT = $IMPLANT$ }
		}
		any_liege_or_above = { is_implant_illegal_in_realm_trigger = { IMPLANT = $IMPLANT$ } }
	}
}
