install_implant_self_interaction = {
	category = interaction_category_friendly
	common_interaction = no
	interface_priority = 30
	desc = install_implant_self_interaction.desc
	icon = artisan_inspiration # Placeholder
	
	target_type = artifact
	target_filter = actor_artifacts
	show_all_targets = no
	
	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			has_any_unequipped_artifact = yes
			NOT = { has_personal_tenet_flag = abstains_from_cybernetics }
			NAND = {
				rite = { rite_has_parameter = hates_cybernetics }
				ai_zeal >= 0
			}
		}
	}
	is_shown = {
		scope:puppet_or_actor = scope:recipient
		scope:puppet_or_actor = { character_has_tech = { TECH = culture_innovation:innovation_cybernetics } }
	}
	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			is_capable_adult = yes
			is_imprisoned = no
			custom_tooltip = {
				text = NEEDS_UNEQUIPPED_IMPLANT
				any_character_artifact = {
					is_equipped = no
					is_cybernetic_implant = yes
				}
			}
			NOT = { has_character_modifier = recovering_from_surgery }
			trigger_if = { # Needs someone to perform the surgery
				limit = { is_ruler = yes }
				employs_court_position = court_physician_court_position
			}
			trigger_else = { liege_or_court_owner ?= { employs_court_position = court_physician_court_position } }
		}
	}
	can_be_picked_artifact = {
		scope:target = {
			is_equipped = no
			is_cybernetic_implant = yes
		}
	}
	send_option = {
		localization = INSTALL_NEW_IMPLANT_WITHOUT_SURGERY
		is_shown = {
			scope:recipient = {
				trigger_if = {
					limit = { scope:target = { artifact_slot_type = artificial_arm } }
					OR = {
						AND = {
							has_trait_rank = { trait = missing_arms rank = 1 }
							num_of_limbs = 0
						}
						AND = {
							has_trait_rank = { trait = missing_arms rank = 2 }
							num_of_limbs < 2
						}
					}
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = artificial_leg } }
					OR = {
						AND = {
							has_trait_rank = { trait = missing_legs rank = 1 }
							num_of_limbs = 0
						}
						AND = {
							has_trait_rank = { trait = missing_legs rank = 2 }
							num_of_limbs < 2
						}
					}
				}
				trigger_else = {
					NOT = { any_equipped_character_artifact = { artifact_slot_type = scope:target.artifact_slot_type } }
					trigger_if = {
						limit = { scope:target = { artifact_slot_type = neural_implant } }
						has_trait = datajacked
					}
					trigger_else_if = {
						limit = { scope:target = { artifact_slot_type = dermal_implant } }
						has_trait = disfigured
					}
					trigger_else_if = {
						limit = { scope:target = { artifact_slot_type = internal_implant } }
						has_trait = implant_dependent
					}
					trigger_else_if = {
						limit = { scope:target = { artifact_slot_type = monocular_implant } }
						has_trait_rank = { trait = missing_eyes rank > 0 }
					}
					trigger_else_if = {
						limit = { scope:target = { artifact_slot_type = binocular_implant } }
						has_trait_rank = { trait = missing_eyes rank > 1 }
					}
				}
			}
		}
		is_valid = { always = yes }
		flag = new
	}
	send_option = {
		localization = INSTALL_NEW_IMPLANT_WITH_SURGERY
		is_shown = {
			scope:recipient = {
				trigger_if = {
					limit = { scope:target = { artifact_slot_type = artificial_arm } }
					has_trait_rank = { trait = missing_arms rank < 2 }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = artificial_leg } }
					has_trait_rank = { trait = missing_legs rank < 2 }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = neural_implant } }
					NOT = { has_trait = datajacked }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = dermal_implant } }
					NOT = { has_trait = disfigured }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = internal_implant } }
					NOT = { has_trait = implant_dependent }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = monocular_implant } }
					has_trait_rank = { trait = missing_eyes rank < 1 }
				}
				trigger_else_if = {
					limit = { scope:target = { artifact_slot_type = binocular_implant } }
					has_trait_rank = { trait = missing_eyes rank < 2 }
				}
			}
		}
		is_valid = { always = yes }
		flag = new
		flag = first_surgery
	}
	send_option = {
		localization = REPLACE_EXISTING_IMPLANT
		is_shown = { scope:recipient = { any_equipped_character_artifact = { artifact_slot_type = scope:target.artifact_slot_type } } }
		is_valid = { always = yes }
		flag = replace
	}
	send_options_exclusive = yes
	
	on_accept = {
		scope:actor = { save_scope_as = sponsor }
		scope:puppet_or_actor = {
			save_scope_as = patient
			if = {
				limit = { is_ruler = yes }
				court_position:court_physician_court_position = { save_scope_as = physician }
			}
			else_if = { liege_or_court_owner = { court_position:court_physician_court_position = { save_scope_as = physician } } }
		}
		if = {
			limit = { scope:replace = yes }
			if = {
				limit = {
					scope:target = { artifact_slot_type = artificial_arm }
					scope:puppet_or_actor = {
						any_equipped_character_artifact = {
							artifact_slot_type = artificial_arm
							count >= 2
						}
					}
				}
				scope:actor = {
					custom_tooltip = WILL_CHOOSE_ARM_TO_UNINSTALL
					trigger_event = cotc_cybernetics_events.1050
				}
			}
			else_if = {
				limit = {
					scope:target = { artifact_slot_type = artificial_leg }
					scope:puppet_or_actor = {
						any_equipped_character_artifact = {
							artifact_slot_type = artificial_leg
							count >= 2
						}
					}
				}
				scope:actor = {
					custom_tooltip = WILL_CHOOSE_LEG_TO_UNINSTALL
					trigger_event = cotc_cybernetics_events.1051
				}
			}
			else = {
				scope:puppet_or_actor = {
					random_equipped_character_artifact = {
						limit = { artifact_slot_type = scope:target.artifact_slot_type }
						custom_tooltip = WILL_BE_UNINSTALLED
						save_scope_as = replaced_implant
					}
				}
				scope:physician = { trigger_event = cotc_cybernetics_events.1000 }
			}
		}
		else = { scope:physician = { trigger_event = cotc_cybernetics_events.1000 } }
		scope:physician = { custom_tooltip = WILL_INSTALL_TARGET_IN_PATIENT }
	}
	
	ai_targets = { ai_recipients = self }
	ai_will_do = {
		base = 0
		cybersurgery_suite_modifier = { IMPLANT = scope:target PATIENT = scope:puppet_or_actor SELECTOR = scope:actor }
	}
}
