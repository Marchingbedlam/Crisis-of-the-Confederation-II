#Events related to the Chief Scientist's tasks

namespace = chief_scientist_task

### MAINTAIN TECHNOLOGY
chief_scientist_task.1000 = { # Incompetent engineer damages artifact
	hidden = yes
	trigger = {
		exists = cp:councillor_chief_scientist
		scope:councillor = cp:councillor_chief_scientist
		cp:councillor_chief_scientist = { learning < high_skill_rating }
		has_any_artifact = yes
	}
	weight_multiplier = {
		base = 1
		compare_modifier = {
			target = scope:councillor
			value = learning
			multiplier = -0.2
			offset = inverted_high_skill_rating
		}
	}
	immediate = {
		cp:councillor_chief_scientist = { save_scope_as = active_councillor }
		set_variable = {
			name = had_chief_scientist_task_side_effect
			value = yes
			days = chief_scientist_task_side_effect_cooldown
		}
		random_character_artifact = {
			save_scope_as = artifact
			weight = {
				base = 0
				modifier = { add = artifact_durability }
			}
			save_scope_value_as = {
				name = durability_loss
				value = {
					add = artifact_durability
					divide = {
						integer_range = {
							min = 2
							max = 4
						}
					}
					multiply = -1
				}
			}
		}
		send_interface_message = {
			type = msg_chief_scientist_task_bad
			title = chief_scientist_task.1000.title
			desc = {
				triggered_desc = {
					trigger = { scope:active_councillor ?= { learning <= average_skill_level } }
					desc = task_learning_bad_unskilled_notification_tooltip
				}
				triggered_desc = {
					trigger = { scope:active_councillor ?= { learning > average_skill_level } }
					desc = task_learning_bad_skilled_notification_tooltip
				}
			}
			tooltip = task_maintain_technology_notification_tooltip

			left_icon = scope:active_councillor
			right_icon = scope:artifact

			scope:artifact ?= { add_durability = scope:durability_gain }
		}
	}
}
chief_scientist_task.1100 = { # Competent engineer repairs artifact
	hidden = yes

	trigger = {
		exists = cp:councillor_chief_scientist
		scope:councillor = cp:councillor_chief_scientist
		cp:councillor_chief_scientist = { learning > mediocre_skill_rating }
		any_character_artifact = { artifact_durability < artifact_max_durability }
	}

	weight_multiplier = {
		base = 1
		compare_modifier = {
			target = scope:councillor
			value = learning
			multiplier = 0.2
			offset = inverted_mediocre_skill_rating
		}
	}

	immediate = {
		cp:councillor_chief_scientist = { save_scope_as = active_councillor }
		set_variable = {
			name = had_chief_scientist_task_side_effect
			value = yes
			days = chief_scientist_task_side_effect_cooldown
		}
		random_character_artifact = {
			limit = { artifact_durability < artifact_max_durability }
			weight = {
				base = 0
				modifier = {
					add = {
						value = artifact_max_durability
						subtract = artifact_durability
					}
				}
			}
			save_scope_as = artifact
			save_scope_value_as = {
				name = durability_gain
				value = {
					add = artifact_max_durability
					subtract = artifact_durability
					divide = {
						integer_range = {
							min = 1
							max = 4
						}
					}
				}
			}
		}

		send_interface_message = {
			type = msg_chief_scientist_task_good
			title = chief_scientist_task.1100.title
			desc = {
				triggered_desc = {
					trigger = { scope:active_councillor ?= { learning <= average_skill_level } }
					desc = task_learning_good_unskilled_notification_tooltip
				}
				triggered_desc = {
					trigger = { scope:active_councillor ?= { learning > average_skill_level } }
					desc = task_learning_good_skilled_notification_tooltip
				}
			}
			tooltip = task_maintain_technology_notification_tooltip

			left_icon = scope:active_councillor
			right_icon = scope:artifact

			scope:artifact ?= { add_durability = scope:durability_gain }
		}
	}
}

				if = {
					limit = { has_scheme_modifier = chief_scientist_supervise_research_fast_project }
					remove_scheme_modifier = chief_scientist_supervise_research_fast_project
				}
				else = {
					add_scheme_modifier = {
						type = chief_scientist_supervise_research_slow_project
						days = chief_scientist_task_modifier_duration
					}
				}
			}
		}
	}
}
