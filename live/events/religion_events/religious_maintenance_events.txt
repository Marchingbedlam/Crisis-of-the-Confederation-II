namespace = religious_maintenance

# These are under-the-hood events that handle things like assigning religious opinion modifiers

# Opinion of liege with the wrong power base
# Fires on yearly pulse to clear stray bad opinion targets
religious_maintenance.001 = {
	hidden = yes
	immediate = {
		# Start by clearing out former lieges, we'll rebuild the list later if necessary
		if = {
			limit = {
				has_variable_list = wrong_power_base_liege
				any_in_list = {
					variable = wrong_power_base_liege
					NOT = { ROOT = { liege = PREV } }
				}
			}
			every_in_list = {
				variable = wrong_power_base_liege
				ROOT = {
					remove_opinion = { modifier = opinion_not_commercial target = PREV }
					remove_opinion = { modifier = opinion_not_military target = PREV }
					remove_opinion = { modifier = opinion_not_administrative target = PREV }
				}
			}
			clear_variable_list = wrong_power_base_liege
		}
		
		# If we are a ruler of the relevant power base, and our liege isn't, we dislike them
		# Otherwise, we have no problem with them
		if = {
			limit = {
				is_ruler = yes
				faith = { has_doctrine_parameter = doctrine_commercial_power }
				NOT = { has_opinion_modifier = { target = liege modifier = opinion_not_commercial } }
				has_commercial_government = yes
				liege = { has_commercial_government = no }
			}
			opinion = { modifier = opinion_not_commercial target = liege }
			add_to_variable_list = { name = wrong_power_base_liege target = liege }
		}
		else_if = {
			limit = {
				is_ruler = yes
				faith = { has_doctrine_parameter = doctrine_military_power }
				NOT = { has_opinion_modifier = { target = liege modifier = opinion_not_military } }
				has_military_government = yes
				liege = { has_military_government = no }
			}
			opinion = { modifier = opinion_not_military target = liege }
			add_to_variable_list = { name = wrong_power_base_liege target = liege }
		}
		else_if = {
			limit = {
				is_ruler = yes
				faith = { has_doctrine_parameter = doctrine_administrative_power }
				NOT = { has_opinion_modifier = { target = liege modifier = opinion_not_administrative } }
				has_administrative_government = yes
				liege = { has_administrative_government = no }
			}
			opinion = { modifier = opinion_not_administrative target = liege }
			add_to_variable_list = { name = wrong_power_base_liege target = liege }
		}
		else = {
			every_in_list = {
				variable = wrong_power_base_liege
				ROOT = {
					remove_opinion = { modifier = opinion_not_commercial target = PREV }
					remove_opinion = { modifier = opinion_not_military target = PREV }
					remove_opinion = { modifier = opinion_not_administrative target = PREV }
				}
			}
			clear_variable_list = wrong_power_base_liege
		}
	}
}

# Fires on title gain, title loss, and faith change; recurses to get the full chain of vassals affected
religious_maintenance.002 = {
	hidden = yes
	
	immediate = {
		trigger_event = religious_maintenance.001
		any_vassal = { trigger_event = religious_maintenance.002 }
	}
}
