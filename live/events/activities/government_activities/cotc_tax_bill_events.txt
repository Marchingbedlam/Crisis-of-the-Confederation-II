namespace = cotc_tax_bill_events

# Everyone has arrived for a tax code session, let's consider the new tax code!
cotc_tax_bill_events.100 = {
	type = activity_event
	desc = cotc_budget_bill_events.100.desc
	
	theme = realm
	override_background = council_chamber
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	left_portrait = {
		character = scope:host
		animation = idle
	}
	
	# Intend to support the bill
	option = {
		name = cotc_budget_bill_events.100.a
		scope:activity = {
			add_to_guest_subset = {
				name = bill_supporters
				target = ROOT
			}
		}
		vote_for_bill_strength_effect = yes
		hidden_effect = {
			save_scope_as = voter
			scope:host = {
				send_interface_message = {
					type = msg_vote_change_good
					title = cotc_budget_bill_events.100.votes_for.tt
					desc = cotc_budget_bill_events.100.votes_for.desc
					left_icon = scope:voter
				}
			}
			scope:activity = {
				add_activity_log_entry = {
					key = voted_for_bill
					tags = { positive intent }
					character = scope:voter
					target = scope:host
					custom_tooltip = voted_for_bill_intent_desc
				}
			}
		}
		ai_chance = {
			base = 50
			# This bill does not support the status quo (well, it shouldn't)
			modifier = {
				factor = 0
				has_activity_intent = maintain_status_quo_intent
			}
			# I want to hold out for political favors
			modifier = {
				factor = 0
				has_activity_intent = trade_favors_intent
			}
			# I just want to block any bills, period
			modifier = {
				factor = 0
				has_activity_intent = block_bill_intent
			}
			# I want to increase revenue, and this bill doesn't do that
			modifier = {
				factor = 0
				scope:activity = { proposed_total_tax_income <= var:ruler.current_total_tax_income }
				OR = {
					has_activity_intent = increase_revenue_intent
					has_activity_intent = increase_revenue_maintain_military_intent
					has_activity_intent = increase_revenue_maintain_commercial_intent
					has_activity_intent = increase_revenue_maintain_technocratic_intent
					has_activity_intent = increase_revenue_maintain_theocratic_intent
				}
			}
			# This bill raises taxes I don't like
			modifier = {
				factor = 0
				scope:activity = { var:proposed_military_taxes > var:ruler.current_military_taxes }
				has_activity_intent = increase_revenue_maintain_military_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_commercial_taxes > var:ruler.current_commercial_taxes }
				has_activity_intent = increase_revenue_maintain_commercial_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_technocratic_taxes > var:ruler.current_technocratic_taxes }
				has_activity_intent = increase_revenue_maintain_technocratic_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_theocratic_taxes > var:ruler.current_theocratic_taxes }
				has_activity_intent = increase_revenue_maintain_theocratic_intent
			}
			# This bill doesn't make tax cuts I want
			modifier = {
				factor = 0
				scope:activity = { var:proposed_military_taxes >= var:ruler.current_military_taxes }
				has_activity_intent = cut_military_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_commercial_taxes >= var:ruler.current_commercial_taxes }
				has_activity_intent = cut_commercial_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_technocratic_taxes >= var:ruler.current_technocratic_taxes }
				has_activity_intent = cut_technocratic_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_theocratic_taxes >= var:ruler.current_theocratic_taxes }
				has_activity_intent = cut_theocratic_taxes_intent
			}
		}
	}
	# Intend to oppose the bill
	option = {
		name = cotc_budget_bill_events.100.b
		scope:activity = {
			add_to_guest_subset = {
				name = bill_opponents
				target = ROOT
			}
		}
		vote_against_bill_strength_effect = yes
		hidden_effect = {
			save_scope_as = voter
			scope:host = {
				send_interface_message = {
					type = msg_vote_change_bad
					title = cotc_budget_bill_events.100.votes_against.tt
					desc = cotc_budget_bill_events.100.votes_against.desc
					left_icon = scope:voter
				}
			}
			scope:activity = {
				add_activity_log_entry = {
					key = voted_against_bill
					tags = { negative intent }
					character = scope:voter
					target = scope:host
					custom_tooltip = voted_against_bill_intent_desc
				}
			}
		}
		ai_chance = {
			base = 50
			# I am a rubber stamp
			modifier = {
				factor = 0
				has_activity_intent = support_bill_intent
			}
			# I want to increase revenue and this bill does that
			modifier = {
				factor = 0
				scope:activity = { proposed_total_tax_income > var:ruler.current_total_tax_income }
				OR = {
					has_activity_intent = increase_revenue_intent
					AND = {
						scope:activity = { var:proposed_military_taxes <= var:ruler.current_military_taxes }
						has_activity_intent = increase_revenue_maintain_military_intent
					}
					AND = {
						scope:activity = { var:proposed_commercial_taxes <= var:ruler.current_commercial_taxes }
						has_activity_intent = increase_revenue_maintain_commercial_intent
					}
					AND = {
						scope:activity = { var:proposed_technocratic_taxes <= var:ruler.current_technocratic_taxes }
						has_activity_intent = increase_revenue_maintain_technocratic_intent
					}
					AND = {
						scope:activity = { var:proposed_theocratic_taxes <= var:ruler.current_theocratic_taxes }
						has_activity_intent = increase_revenue_maintain_theocratic_intent
					}
				}
			}
			# This bill makes tax cuts I like
			modifier = {
				factor = 0
				scope:activity = { var:proposed_military_taxes < var:ruler.current_military_taxes }
				has_activity_intent = cut_military_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_commercial_taxes < var:ruler.current_commercial_taxes }
				has_activity_intent = cut_commercial_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_technocratic_taxes < var:ruler.current_technocratic_taxes }
				has_activity_intent = cut_technocratic_taxes_intent
			}
			modifier = {
				factor = 0
				scope:activity = { var:proposed_theocratic_taxes < var:ruler.current_theocratic_taxes }
				has_activity_intent = cut_theocratic_taxes_intent
			}
		}
	}
}
# Your tax code has successfully passed!
cotc_tax_bill_events.200 = {
	type = activity_event
	desc = cotc_tax_bill_events.200.desc
	
	theme = realm
	override_background = council_chamber
	
	right_portrait = {
		character = ROOT
		animation = war_over_win
	}
	
	option = {
		name = cotc_budget_bill_events.200.a
		add_prestige = {
			value = 50
			multiply = scope:ruler.tier_multiplier
		}
		hidden_effect = { set_variable = bill_passed }
		pass_military_tax_code_effect = yes
		pass_commercial_tax_code_effect = yes
		pass_technocratic_tax_code_effect = yes
		pass_theocratic_tax_code_effect = yes
		pork_spending_effect = yes
		hidden_effect = {
			trigger_event = { on_action = province_taxes_on_action }
			if = {
				limit = { var:ai_faith_in_democracy <= 10 }
				change_variable = { name = ai_faith_in_democracy add = 1 }
			}
		}
	}
	
	after = { remove_variable = bill_passed }
}
# A bill you voted for has passed!
cotc_tax_bill_events.210 = {
	type = activity_event
	desc = cotc_tax_bill_events.210.desc
	
	theme = realm
	override_background = council_chamber
	
	right_portrait = {
		character = ROOT
		animation = war_over_win
	}
	left_portrait = {
		character = scope:host
		animation = war_over_win
	}
	
	immediate = {
		if = {
			limit = { tax_code_passed_voter_intent_trigger = { BUDGET = scope:activity } }
			complete_activity_intent = yes
		}
		if = {
			limit = { bill_is_ideologically_necessary = { BILL = scope:activity } }
			change_variable = { name = ai_faith_in_democracy add = 1 }
		}
		else_if = {
			limit = { var:ai_faith_in_democracy <= 10 }
			change_variable = { name = ai_faith_in_democracy add = 1 }
		}
	}
	
	option = {
		name = cotc_budget_bill_events.210.a
		add_prestige = {
			value = 5
			multiply = scope:ruler.tier_multiplier
		}
		reverse_add_opinion = { modifier = opinion_voted_for_my_bill target = scope:host }
		show_as_tooltip = {
			pass_military_tax_code_effect = yes
			pass_commercial_tax_code_effect = yes
			pass_technocratic_tax_code_effect = yes
			pass_theocratic_tax_code_effect = yes
			pork_spending_effect = yes
		}
	}
}

# Someone proposes an amendment!
cotc_tax_bill_events.400 = {
	type = activity_event
	desc = cotc_tax_bill_events.400.desc
	
	theme = realm
	override_background = council_chamber
	
	immediate = {
		ordered_in_list = {
			variable = supporters_gained
			order_by = legislative_seniority
			save_scope_as = representative_opponent
		}
		scope:activity = {
			add_activity_log_entry = {
				key = proposed_amendment
				tags = { neutral }
				character = scope:representative_opponent
				target = scope:host
				custom_tooltip = proposed_amendment_desc
				custom_tooltip = tax_bill_amendment_proposed_desc
				show_in_conclusion = yes
			}
		}
	}
	
	right_portrait = {
		character = scope:representative_opponent
		animation = debate
	}
	left_portrait = {
		character = ROOT
		animation = thinking
	}
	
	# Agree to the amendment
	option = {
		name = cotc_budget_bill_events.400.a
		custom_tooltip = amend_bill_desc
		custom_tooltip = gain_supporters_desc
		custom_tooltip = lose_supporters_desc
		hidden_effect = {
			scope:activity = {
				add_activity_log_entry = {
					key = accepted_amendment
					tags = { positive }
					character = scope:host
					target = scope:representative_opponent
					custom_tooltip = accepted_amendment_desc
					show_in_conclusion = yes
				}
				set_variable = { name = proposed_military_taxes value = ROOT.var:amended_military_taxes }
				set_variable = { name = proposed_commercial_taxes value = ROOT.var:amended_commercial_taxes }
				set_variable = { name = proposed_technocratic_taxes value = ROOT.var:amended_technocratic_taxes }
				set_variable = { name = proposed_theocratic_taxes value = ROOT.var:amended_theocratic_taxes }
			}
			every_in_list = {
				variable = supporters_gained
				trigger_event = { id = cotc_bill_events.300 }
			}
			every_in_list = {
				variable = supporters_lost
				trigger_event = { id = cotc_bill_events.310 }
			}
		}
		ai_chance = {
			base = 90
			modifier = {
				add = ai_rationality
			}
			modifier = {
				add = ai_greed
				amended_total_tax_income > scope:activity.proposed_total_tax_income
			}
			# Don't do this if you would lose more supporters than you gain
			modifier = {
				factor = 0
				total_supporters_lost >= total_supporters_gained
			}
		}
	}
	
	# Do not agree to the amendment
	option = {
		name = cotc_budget_bill_events.400.b
		hidden_effect = {
			scope:activity = {
				add_activity_log_entry = {
					key = rejected_amendment
					tags = { negative }
					character = scope:host
					target = scope:representative_opponent
					custom_tooltip = rejected_amendment_desc
					show_in_conclusion = yes
				}
			}
			every_in_list = {
				variable = supporters_gained
				send_interface_message = {
					type = msg_vote_change_bad
					title = cotc_budget_bill_events.400.refused_amendment.tt
					desc = cotc_budget_bill_events.400.refused_amendment.desc
					left_icon = scope:legislator
				}
			}
			every_in_list = {
				variable = supporters_lost
				send_interface_message = {
					type = msg_vote_change_good
					title = cotc_budget_bill_events.400.refused_amendment.tt
					desc = cotc_budget_bill_events.400.refused_amendment.desc
					left_icon = scope:legislator
				}
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = ai_vengefulness }
			modifier = {
				add = ai_greed
				amended_total_tax_income < scope:activity.proposed_total_tax_income
			}
		}
	}
	
	# Clean up all the now-unnecessary variables
	after = {
		remove_variable = amended_military_taxes
		remove_variable = amended_commercial_taxes
		remove_variable = amended_technocratic_taxes
		remove_variable = amended_theocratic_taxes
		clear_variable_list = supporters_gained
		clear_variable_list = supporters_lost
	}
}
