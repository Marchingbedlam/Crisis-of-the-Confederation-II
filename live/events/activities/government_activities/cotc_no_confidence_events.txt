namespace = cotc_no_confidence_events

scripted_effect cotc_prime_minister_vote_for_challenger_debate = {
	custom_tooltip = vote_for_challenger_desc
	hidden_effect = {
		save_scope_as = voter
		scope:activity.var:challenger = {
			send_interface_message = {
				type = msg_vote_change_good
				title = cotc_no_confidence_events.100.votes_for_challenger.tt
				desc = cotc_no_confidence_events.100.votes_for_challenger.desc
				left_icon = scope:voter
			}
		}
		scope:activity.var:incumbent = {
			send_interface_message = {
				type = msg_vote_change_bad
				title = cotc_no_confidence_events.100.votes_for_challenger.tt
				desc = cotc_no_confidence_events.100.votes_for_challenger.desc
				left_icon = scope:voter
			}
		}
		scope:activity.var:ruler = {
			send_interface_message = {
				type = msg_vote_change_neutral
				title = cotc_no_confidence_events.100.votes_for_challenger.tt
				desc = cotc_no_confidence_events.100.votes_for_challenger.desc
				left_icon = scope:voter
			}
		}
		scope:activity = {
			if = {
				limit = { scope:voter = { is_in_guest_subset = { name = undecideds } } }
				remove_from_guest_subset = { name = undecideds target = scope:voter }
			}
			else_if = {
				limit = { scope:voter = { is_in_guest_subset = { name = bill_opponents } } }
				remove_from_guest_subset = { name = bill_opponents target = scope:voter }
			}
			add_to_guest_subset = { name = bill_supporters target = scope:voter }
			add_activity_log_entry = {
				key = voted_for_challenger
				tags = { positive }
				character = scope:voter
				target = scope:activity.var:challenger
				if = {
					limit = { scope:opposition = flag:opposed }
					custom_tooltip = voted_for_challenger_debate_desc
				}
				else_if = {
					limit = { scope:actor = scope:activity.var:challenger }
					custom_tooltip = voted_for_challenger_speech_success_desc
				}
				else = { custom_tooltip = voted_for_challenger_speech_failure_desc }
			}
		}
	}
}

scripted_effect cotc_prime_minister_opposed_leadership_challenge = {
	hidden_effect = { save_scope_value_as = { name = opposition value = flag:opposed } }
	duel = {
		skill = diplomacy
		target = scope:opponent
		50 = {
			compare_modifier = {
				value = scope:duel_value
				multiplier = 3.5
				min = -49
			}
			desc = scope_actor_wins_debate_desc
			custom_tooltip = scope_actor_wins_qualification_debate_results_desc
			hidden_effect = {
				scope:actor = {
					send_interface_toast = {
						type = msg_debate_result_good
						title = cotc_no_confidence_events.300.won_debate.tt
						desc = successful_leadership_debate_desc
						left_icon = scope:actor
						right_icon = scope:opponent
					}
				}
				scope:opponent = {
					send_interface_toast = {
						type = msg_debate_result_bad
						title = cotc_no_confidence_events.300.lost_debate.tt
						desc = successful_leadership_debate_desc
						left_icon = scope:actor
						right_icon = scope:opponent
					}
				}
				scope:activity = {
					if = {
						limit = { scope:actor = scope:activity.var:challenger }
						add_activity_log_entry = {
							key = successful_leadership_debate
							tags = { positive }
							character = scope:actor
							target = scope:opponent
							custom_tooltip = successful_leadership_debate_desc
						}
					}
					else = {
						add_activity_log_entry = {
							key = successful_leadership_debate
							tags = { negative }
							character = scope:actor
							target = scope:opponent
							custom_tooltip = successful_leadership_debate_desc
						}
					}
					every_attending_character = {
						limit = { legislator_cares_about_qualifications_trigger = yes }
						trigger_event = cotc_no_confidence_events.308
					}
				}
			}
		}

			}

						desc = unsuccessful_leadership_debate_desc
						left_icon = scope:actor
