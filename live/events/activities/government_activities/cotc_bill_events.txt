# Build list of lobbyists for a legislator
cotc_bill_events.155 = {
	hidden = yes
	immediate = {
		# The ruler of your constituency, who may have had a hand in your election
		if = {
			limit = { exists = var:riding.holder }
			add_to_variable_list = { name = possible_lobbyists target = var:riding.holder }
		}
		# Friends and lovers
		every_relation = {
			type = friend
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = lover
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = soulmate
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = elder
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		# The leader of this legislator's ideology in the legislature
		scope:activity = {
			ordered_attending_character = {
				limit = {
					save_temporary_scope_as = evaluated_character
					is_character_in_legislature_trigger = { LEGISLATURE = scope:activity.var:legislature CHARACTER = scope:evaluated_character }
					faith = ROOT.faith
				}
				order_by = legislative_seniority
				ROOT = {
					if = {
						limit = {
							NOR = {
								ROOT = PREV
								is_target_in_variable_list = { target = PREV name = possible_lobbyists }
							}
						}
						add_to_guest_subset = { name = possible_lobbyists target = PREV }
					}
				}
			}
		}
	}
}

# Let's do some horse trading
cotc_bill_events.400 = {
	type = activity_event
	desc = cotc_bill_events.400.desc
	
	theme = realm
	override_background = council_chamber
	
	immediate = {
		# For bribe calculations
		save_scope_as = actor
		scope:target_opponent = { save_scope_as = recipient }
		# Calculate cost of earmarks
		save_scope_value_as = { name = pork_cost value = scope:target_opponent.var:riding.treasury_required_for_stimulus }
	}
	
	right_portrait = {
		character = scope:target_opponent
		animation = thinking
	}
	left_portrait = {
		character = ROOT
		animation = debate
	}
	
	# Perhaps a simple bribe?
	option = {
		name = cotc_bill_events.400.a
		trigger = { gold >= bribe_value }
		show_as_unavailable = { always = yes }
		show_as_tooltip = {
			pay_short_term_gold = { target = scope:target_opponent gold = bribe_value }
			add_piety = {
				value = bribe_piety
				multiply = 10
			}
		}
		positive_lawfulness_stress_gain_effect = yes
		if = {
			limit = { ai_rationality < ai_greed }
			custom_tooltip = pos_greed_desc
			add_stress = medium_stress_gain
		}
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.410 } } }
		ai_chance = {
			base = 100
			# Personality effects
			modifier = {
				add = {
					value = 0
					subtract = ai_greed
					max = 0
				}
			}
			modifier = {
				add = {
					value = 0
					subtract = ai_lawfulness
					max = 0
					multiply = 3
				}
			}
			modifier = {
				add = {
					value = 0
					subtract = ai_lawfulness
					min = 0
				}
			}
			# Judgement of recipient's personality - don't try to bribe someone who won't accept it!
			modifier = {
				add = scope:target_opponent.ai_greed
				ai_rationality >= very_low_positive_ai_value
			}
			modifier = {
				add = {
					value = 0
					subtract = scope:target_opponent.ai_lawfulness
				}
				ai_rationality >= low_positive_ai_value
			}
			# Can't afford the bribe
			modifier = {
				add = {
					value = 0
					subtract = ai_rationality
					max = 0
				}
				short_term_gold < bribe_value
			}
		}
	}
	# A gift of influence
	option = {
		name = cotc_bill_events.400.f
		trigger = { influence >= influence_bribe_value }
		show_as_unavailable = { always = yes }
		show_as_tooltip = {
			change_influence = { value = influence_bribe_value multiply = -1 }
			scope:target_opponent = { change_influence = influence_bribe_value }
		}
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.450 } } }
		ai_chance = {
			base = 0
			modifier = {
				add = {
					value = influence
					divide = { value = influence_bribe_value min = 1 }
				}
			}
		}
	}
	# Throw in some pork
	option = {
		name = cotc_bill_events.400.b
		trigger = {
			scope:activity = { has_variable = pork }
			treasury >= scope:pork_cost
		}
		show_as_unavailable = { scope:activity = { has_variable = pork } }
		show_as_tooltip = { pay_short_term_treasury = { target = scope:target_opponent treasury = scope:pork_cost } }
		if = {
			limit = { scope:activity.var:pork > 0 }
			custom_tooltip = already_spending_pork_desc
		}
		negative_interventionism_stress_gain_effect = yes
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.420 } } }
		ai_chance = {
			base = 50
			# Personality
			modifier = { add = ai_rationality }
			modifier = { add = ai_interventionism }
			# Assessment of target's personality
			modifier = {
				add = scope:target_opponent.ai_interventionism
				ai_rationality >= low_positive_ai_value
			}
			modifier = {
				add = {
					value = 0
					subtract = ai_rationality
					max = 0
				}
				scope:target_opponent.var:riding.treasury_required_for_stimulus > {
					value = short_term_treasury
					subtract = scope:activity.var:pork
				}
				short_term_treasury < scope:target_opponent.var:riding.treasury_required_for_stimulus
			}
		}
	}
	# Use a strong hook
	option = {
		name = cotc_bill_events.400.c
		trigger = { has_strong_usable_hook = scope:target_opponent }
		show_as_unavailable = { always = yes }
		use_hook = scope:target_opponent
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.430 } } }
		ai_chance = {
			base = 100
			modifier = {
				add = {
					value = 0
					subtract = ai_compassion
				}
			}
		}
	}
	# Use a weak hook
	option = {
		name = cotc_bill_events.400.c
		trigger = { has_weak_hook = scope:target_opponent }
		use_hook = scope:target_opponent
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.435 } } }
		ai_chance = {
			base = 100
			modifier = {
				add = {
					value = 0
					subtract = ai_compassion
				}
			}
			modifier = {
				factor = 0
				has_strong_usable_hook = scope:target_opponent
			}
		}
	}
	# Offer a future favor
	option = {
		name = cotc_bill_events.400.d
		trigger = { scope:target_opponent = { can_add_hook = { target = ROOT type = favor_hook } } }
		show_as_unavailable = { always = yes }
		show_as_tooltip = { scope:target_opponent = { add_hook = { target = ROOT type = favor_hook } } }
		hidden_effect = { scope:target_opponent = { trigger_event = { id = cotc_bill_events.440 } } }
		ai_chance = {
			base = 100
			modifier = {
				add = {
					value = 0
					subtract = ai_vengefulness
				}
			}
		}
	}
	# Do nothing
	option = {
		name = cotc_bill_events.400.e
		ai_chance = {
			base = 100
			modifier = {
				add = {
					value = 0
					subtract = ai_energy
				}
			}
		}
	}
}

					desc = cotc_bill_events.400.bribe_refused.desc

	type = activity_event
	desc = cotc_bill_events.420.desc
	
	theme = realm
	override_background = council_chamber
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	left_portrait = {
		character = scope:actor
		animation = debate
	}
	
	# Accept the deal
	option = {
		name = cotc_bill_events.420.a
		custom_tooltip = pork_barrel_effect_desc
		negative_interventionism_stress_gain_effect = yes
		add_piety = {
			value = pork_piety
			multiply = 10
		}
		if = {
			limit = { has_activity_intent = trade_favors_intent }
			complete_activity_intent = yes
		}
		hidden_effect = {
			scope:activity = { change_variable = { name = pork add = scope:pork_cost } }
			save_temporary_scope_as = voter
			scope:activity = {
				add_activity_log_entry = {
					key = added_pork_to_bill
					tags = { neutral }
					character = scope:voter
					custom_tooltip = added_pork_to_bill_desc
					show_in_conclusion = yes
				}
				add_activity_log_entry = {
					key = voted_for_bill
					tags = { positive }
					character = scope:actor
					target = scope:voter
					custom_tooltip = voted_for_bill_backroom_deal_desc
				}
				remove_from_guest_subset = { name = bill_opponents target = ROOT }
				add_to_guest_subset = { name = bill_supporters target = ROOT }
				add_to_variable_list = { name = pork_recipients target = ROOT.var:riding }
			}
			set_variable = { name = vote_strength value = 300 }
			change_variable = { name = vote_strength add = ai_honor }
			set_variable = vote_bargained_for
			scope:actor = {
				send_interface_toast = {
					type = msg_legislature_response_good
					title = cotc_bill_events.400.pork_accepted.tt
					desc = cotc_bill_events.400.pork_accepted.desc
					left_icon = scope:target_opponent
				}
			}
		}
		ai_chance = {
			base = 100
			modifier = { add = ai_greed }
			modifier = {
				add = {
					value = ai_interventionism
					multiply = 3
				}
			}
			modifier = {
				add = {
					value = 0
					subtract = ai_zeal
					multiply = 3
					max = 0
				}
				bill_is_ideologically_unacceptable = { BILL = scope:activity }
			}
		}
	}
	# Refuse the deal
	option = {
		name = cotc_bill_events.420.b
		positive_interventionism_stress_gain_effect = yes
		hidden_effect = {
			save_temporary_scope_as = voter
			scope:activity = {
				add_activity_log_entry = {
					key = refused_backroom_deal
					tags = { neutral }
					character = scope:actor
					target = scope:voter
					custom_tooltip = refused_backroom_deal_pro_desc
				}
			}
			scope:actor = {
				send_interface_toast = {
					type = msg_legislature_response_bad
					title = cotc_bill_events.400.pork_refused.tt
					desc = cotc_bill_events.400.pork_refused.desc
					left_icon = scope:target_opponent
				}
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				add = {
					value = var:vote_strength
					min = 0
				}
			}
			modifier = {
				add = {
					value = 0
					subtract = 50
				}
				has_activity_intent = trade_favors_intent
			}
		}
	}
}

# Make a speech advocating for the bill
cotc_bill_events.500 = {
	type = activity_event
	desc = cotc_bill_events.500.desc
	
	theme = realm
	override_background = council_chamber
	
	immediate = {
		save_scope_as = speaker
		set_variable = { name = had_speech_opportunity value = yes months = 2 }
	}
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	
	# Let's do it
	option = {
		name = cotc_bill_events.500.a
		duel = {
			skill = diplomacy
			value = scope:activity.bill_sell_difficulty
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				custom_tooltip = cotc_bill_events.500.a.success
				add_prestige = 100 # We look good
				hidden_effect = {
					send_interface_toast = {
						type = msg_vote_speech_good
						title = cotc_bill_events.500.successful_speech_self.tt
						desc = cotc_bill_events.500.successful_speech_self.desc
					}
					scope:host = {
						if = {
							limit = { NOT = { this = ROOT } }
							send_interface_toast = {
								type = msg_vote_speech_good
								title = cotc_bill_events.500.successful_speech_host.tt
								desc = cotc_bill_events.500.successful_speech_host.desc
								left_icon = scope:speaker
							}
						}
					}
					scope:activity = {
						add_activity_log_entry = {
							key = made_successful_speech
							tags = { positive }
							character = scope:speaker
							custom_tooltip = made_successful_speech_for_bill_desc
							show_in_conclusion = yes
						}
						every_guest_subset = {
							name = bill_supporters
							send_interface_message = {
								type = msg_vote_speech_good
								title = cotc_bill_events.500.successful_speech.tt
								desc = cotc_bill_events.500.successful_speech.desc
								left_icon = scope:speaker
							}
						}
						every_guest_subset = {
							name = bill_opponents
							trigger_event = { id = cotc_bill_events.505 }
						}
					}
				}
			}
			50 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				add_prestige = -100 # This is an embarassment
				hidden_effect = {
					send_interface_toast = {
						type = msg_vote_speech_bad
						title = cotc_bill_events.500.unsuccessful_speech_self.tt
						desc = cotc_bill_events.500.unsuccessful_speech_self.desc
					}
					scope:host = {
						if = {
							limit = { NOT = { this = ROOT } }
							send_interface_toast = {
								type = msg_vote_speech_bad
								title = cotc_bill_events.500.unsuccessful_speech_host.tt
								desc = cotc_bill_events.500.unsuccessful_speech_host.desc
								left_icon = scope:speaker
							}
						}
					}
					scope:activity = {
						add_activity_log_entry = {
							key = made_unsuccessful_speech
							tags = { neutral }
							character = scope:speaker
							custom_tooltip = made_unsuccessful_speech_for_bill_desc
							show_in_conclusion = yes
						}
						every_guest_subset = {
							name = bill_supporters
							send_interface_message = {
								type = msg_vote_speech_bad
								title = cotc_bill_events.500.unsuccessful_speech.tt
								desc = cotc_bill_events.500.unsuccessful_speech.desc
								left_icon = scope:speaker
							}
						}
						every_guest_subset = {
							name = bill_opponents
							send_interface_message = {
								type = msg_vote_speech_neutral
								title = cotc_bill_events.500.unsuccessful_speech.tt
								desc = cotc_bill_events.500.unsuccessful_speech.desc
								left_icon = scope:speaker
							}
						}
					}

		}
		ai_chance = {
			base = 0
			modifier = {
				add = {
					value = diplomacy
					multiply = 10
				}
			}
			modifier = { add = ai_boldness }
			modifier = { add = ai_energy }
			modifier = {
				add = {
					value = ai_sociability
					divide = 2
				}
			}
		}
	}
	# Let's not
	option = {
		name = cotc_bill_events.500.b
		ai_chance = { base = 200 }
	}
}
# Someone made a convincing speech advocating for the bill
cotc_bill_events.505 = {
	type = activity_event
	desc = cotc_bill_events.505.desc
	
	theme = realm
	override_background = council_chamber
	
	immediate = { save_scope_as = voter }
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	left_portrait = {
		character = scope:speaker
		animation = debate
	}
	
	# I guess I'll vote for it after all
	option = {
		name = cotc_bill_events.505.a
		if = {
			limit = { has_variable = vote_bargained_for }
			if = {
				limit = { ai_rationality < ai_honor }
				custom_tooltip = pos_honor_desc
				add_stress = medium_stress_gain
			}
		}
		if = {
			limit = {
				bill_is_ideologically_unacceptable = { BILL = scope:activity }
				ai_zeal >= 0
			}
			custom_tooltip = pos_zeal_desc
			add_stress = major_stress_gain
		}
		hidden_effect = {
			save_scope_as = voter
			scope:activity = {
				add_activity_log_entry = {
					key = voted_for_bill
					tags = { positive }
					character = scope:speaker
					target = scope:voter
					custom_tooltip = voted_for_bill_speech_desc
				}
				remove_from_guest_subset = { name = bill_opponents target = ROOT }
				add_to_guest_subset = { name = bill_supporters target = ROOT }
			}
			set_variable = { name = vote_strength value = 50 }
			change_variable = { name = vote_strength add = scope:speaker.diplomacy }
			remove_variable = vote_bargained_for
			scope:host = {
				send_interface_message = {
					type = msg_vote_change_good
					title = cotc_bill_events.505.changed_vote.tt
					desc = cotc_bill_events.505.changed_vote.desc
					left_icon = scope:voter
				}
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				add = {
					value = scope:speaker.diplomacy
					multiply = 5
				}
			}
			modifier = { add = "opinion(scope:speaker)" }
			modifier = { add = ai_rationality }
			modifier = {
				add = {
					value = 0
					subtract = ai_zeal
					multiply = 3
					max = 0
				}
				bill_is_ideologically_unacceptable = { BILL = scope:activity }
			}
		}
	}
	# I remain unconvinced.
	option = {
		name = cotc_bill_events.505.b
		if = {
			limit = {
				OR = {
					NOT = { has_variable = vote_bargained_for }
					ai_rationality >= ai_honor
				}
			}
			positive_rationality_stress_gain_effect = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				add = {
					value = var:vote_strength
					min = 0
				}
			}
		}
	}
}
