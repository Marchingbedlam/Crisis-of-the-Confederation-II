namespace = cotc_freelancer_events

# The group would be renamed to suit a new Charter or Posture. Fired from
# cotc_freelancer_rename_group_effect, players only, AI renames silently.
cotc_freelancer_events.0001 = {
	type = character_event
	title = cotc_freelancer_events.0001.t
	desc = cotc_freelancer_events.0001.desc
	theme = landless_adventurer

	left_portrait = {
		character = root
		animation = thinking
	}

	# The rename gate has already checked government, title and the lock, but the
	# event can arrive a tick later - re-check rather than rename a lapsed group.
	trigger = {
		government_has_flag = government_is_landless_adventurer
		exists = primary_title
	}

	option = {
		name = cotc_freelancer_events.0001.a
		cotc_freelancer_apply_group_name_effect = yes
	}

	option = {
		name = cotc_freelancer_events.0001.b
		custom_tooltip = cotc_freelancer_events.0001.b.tt
	}
}
