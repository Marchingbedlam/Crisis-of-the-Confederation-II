# STARTING PROJECTS
cotc_research_project_events.1 = { # Start a new research project
	immediate = {
		if = {
			limit = { any_character_war = { any_war_defender = { this = ROOT } } }
			ordered_character_war = {
				limit = { any_war_defender = { this = ROOT } }
				order_by = attacker_war_score
				primary_attacker = { save_scope_as = enemy }
			}
		}
		else_if = {
			limit = { any_character_war = { any_war_attacker = { this = ROOT } } }
			ordered_character_war = {
				limit = { any_war_attacker = { this = ROOT } }
				order_by = defender_war_score
				primary_defender = { save_scope_as = enemy }
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:theory } }
				desc = cotc_research_project_events.1.intro.theory
			}
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:engineering } }
				desc = cotc_research_project_events.1.intro.engineering
			}
			desc = cotc_research_project_events.1.intro.praxis
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					any_in_list = { list = research_project_flags this = flag:military }
					exists = scope:enemy
				}
				desc = cotc_research_project_events.1.desc.military_war
			}
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:military } }
				desc = cotc_research_project_events.1.desc.military_peace
			}
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:social } }
				desc = cotc_research_project_events.1.desc.social
			}
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:economic } }
				desc = cotc_research_project_events.1.desc.economic
			}
			triggered_desc = {
				trigger = { any_in_list = { list = research_project_flags this = flag:transhumanist } }
				desc = cotc_research_project_events.1.desc.transhumanist
			}
			desc = cotc_research_project_events.1.desc.other
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:enemy }
				desc = cotc_research_project_events.1.outro.war
			}
			desc = cotc_research_project_events.1.outro.peace
		}
	}
	theme = research
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	
	option = { # Start a public research project
		name = cotc_research_project_events.1.a
		custom_tooltip = {
			text = cotc_research_project_events.1.a.tt
			start_scheme = { type = research_project_public targets_nothing = yes }
		}
		random_scheme = {
			limit = { scheme_type = research_project_public }
			save_scope_as = scheme
			set_variable = { name = owner value = ROOT }
		}
		ai_chance = {
			base = 50
		}
	}
	
	option = { # Wait go back I don't want to do a research project after all
		name = cotc_research_project_events.1.c
		ai_chance = { base = 0 }
	}
	
	after = {
		remove_character_flag = starting_research_project
		scope:scheme ?= { # Assign research project data
			set_variable = { name = research_subject value = scope:research_subject }
			# Set all project flags
			every_in_list = {
				list = research_project_flags
				scope:scheme = { add_to_variable_list = { name = flags target = PREV } }
			}
			# Set agent type caps
			if = {
				limit = { exists = scope:max_economist_slots }
				set_variable = { name = max_economist_slots value = scope:max_economist_slots }
			}
			if = {
				limit = { exists = scope:max_sociologist_slots }
				set_variable = { name = max_sociologist_slots value = scope:max_sociologist_slots }
			}
			if = {
				limit = { exists = scope:max_officer_slots }
				set_variable = { name = max_officer_slots value = scope:max_officer_slots }
			}
			if = {
				limit = { exists = scope:max_doctor_slots }
				set_variable = { name = max_doctor_slots value = scope:max_doctor_slots }
			}
			if = {
				limit = { exists = scope:max_engineer_slots }
				set_variable = { name = max_engineer_slots value = scope:max_engineer_slots }
			}
			if = {
				limit = { exists = scope:max_physicist_slots }
				set_variable = { name = max_physicist_slots value = scope:max_physicist_slots }
			}
			set_research_project_difficulty_effect = yes
			
			# Set Codename
			generate_research_project_codename = yes
		}
		trigger_event = cotc_research_project_events.2
	}
}

# ENDING PROJECTS
cotc_research_project_events.10 = { # Successfully complete the research project
	title = cotc_research_project_events.title
	desc = {
		desc = cotc_research_project_events.10.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:first_ever_discovery
					scope:scheme = { any_in_list = { variable = researchers count >= 1 } }
				}
				desc = cotc_research_project_events.10.desc.first_ever_discovery_with_team
			}
			triggered_desc = {
				trigger = { exists = scope:first_ever_discovery }
				desc = cotc_research_project_events.10.desc.first_ever_discovery_no_team
			}
			triggered_desc = {
				trigger = {
					exists = scope:first_culture_discovery
					scope:scheme = { any_in_list = { variable = researchers count >= 1 } }
				}
				desc = cotc_research_project_events.10.desc.first_culture_discovery_with_team
			}
			triggered_desc = {
				trigger = { exists = scope:first_culture_discovery }
				desc = cotc_research_project_events.10.desc.first_culture_discovery_no_team
			}
			triggered_desc = {
				trigger = { scope:scheme = { any_in_list = { variable = researchers count < 1 } } }
				desc = cotc_research_project_events.10.desc.no_team
			}
		}
	}
	theme = research
	
	immediate = {
		scope:scheme = {
			if = {
				limit = { NOT = { is_target_in_global_variable_list = { name = all_discovered_innovations target = var:research_subject } } }
				save_scope_value_as = { name = first_ever_discovery value = flag:yes }
			}
			if = {
				limit = { NOT = { capital_county ?= { culture = { is_target_in_variable_list = { name = culture_discovered_innovations target = var:research_subject } } } } }
				save_scope_value_as = { name = first_culture_discovery value = flag:yes }
			}
			var:research_subject = { save_scope_as = discovered_innovation }
			# Add any current researchers as contributors
			every_scheme_agent_character = {
				limit = { NOT = { scope:scheme = { is_target_in_variable_list = { name = researchers target = PREV } } } }
				scope:scheme = { add_to_variable_list = { name = researchers target = PREV } }
			}
			if = {
				limit = { any_scheme_agent_character = { count >= 1 } }
				ordered_scheme_agent_character = {
					order_by = learning
					position = 1
					save_scope_as = first_agent
				}
				if = {
					limit = { any_scheme_agent_character = { count >= 2 } }
					ordered_scheme_agent_character = {
						order_by = learning
						position = 2
						save_scope_as = second_agent
					}
					if = {
						limit = { any_scheme_agent_character = { count >= 3 } }
						ordered_scheme_agent_character = {
							order_by = learning
							position = 3
							save_scope_as = third_agent
						}
						if = {
							limit = { any_scheme_agent_character = { count >= 4 } }
							ordered_scheme_agent_character = {
								order_by = learning
								position = 4
								save_scope_as = fourth_agent
							}
						}
					}
				}
				save_scope_value_as = { # Divide the remaining grant money among all agents
					name = agent_payout
					value = {
						value = var:grant_money
						divide = { every_in_list = { variable = researchers add = 1 } }
					}
				}
			}
		}
	}
	
	right_portrait = {
		character = ROOT
		animation = idle
	}
	left_portrait = {
		trigger = { exists = scope:first_agent }
		character = scope:first_agent
		animation = idle
	}
	lower_center_portrait = {
		trigger = { exists = scope:second_agent }
		character = scope:second_agent
	}
	lower_left_portrait = {
		trigger = { exists = scope:third_agent }
		character = scope:third_agent
	}
	lower_right_portrait = {
		trigger = { exists = scope:fourth_agent }
		character = scope:fourth_agent
	}
	
	option = {
		name = cotc_research_project_events.10.a
		capital_county = { add_tech_to_county = { TECH = scope:discovered_innovation } }
		if = { # You did this all by yourself, gain prestige and get back your investment
			limit = { scope:scheme = { NOT = { any_in_list = { variable = researchers count >= 1 } } } }
			add_prestige = research_project_prestige_reward
			add_treasury_or_gold = scope:scheme.var:grant_money
			add_research_project_secret_effect = yes
		}
		# Piety and stress effects
		if = {
			limit = { research_project_piety_reward != 0 }
			add_piety = research_project_piety_reward
		}
		add_research_project_stress_effects = yes
		scope:scheme = {
			every_in_list = { variable = researchers trigger_event = cotc_research_project_events.11 }
			end_scheme = yes
		}
	}
}

# PROJECT MANAGEMENT
cotc_research_project_events.100 = { # Research project in progress
	type = character_event
	window = scheme_preparations_event
	title = cotc_research_project_events.title
	desc = cotc_research_project_events.100.desc
	theme = research
	left_portrait = {
		character = scope:left_portrait
		scripted_animation = scheme_prep_pick_best_stance
	}
	center_portrait = {
		character = scope:center_portrait
		scripted_animation = scheme_prep_pick_best_stance
	}
	right_portrait = {
		character = scope:right_portrait
		scripted_animation = scheme_prep_pick_best_stance
	}
	
	trigger = {
		NOR = {
			has_character_flag = currently_in_scheme_preparations
			has_character_flag = currently_in_scheme_reminder
		}
	}

	immediate = {
		# Flag that we're in the event, so no doubles pls.
		add_character_flag = currently_in_scheme_preparations
		# Sort some aesthetic stuff for the player.
		if = {
			limit = { is_ai = no }
			save_scope_as = center_portrait
			scope:scheme = {
				if = {
					limit = { any_scheme_agent_character = { count >= 1 } }
					ordered_scheme_agent_character = {
						order_by = learning
						position = 1
						save_scope_as = right_portrait
					}
					if = {
						limit = { any_scheme_agent_character = { count >= 2 } }
						ordered_scheme_agent_character = {
							order_by = learning
							position = 2
							save_scope_as = left_portrait
						}
					}
				}
			}
		}
		# How much money do we need to upgrade to the next tier of funding?
		scope:scheme = {
			save_scope_value_as = {
				name = funding_required
				value = {
					if = {
						limit = { any_scheme_agent_slot = { count = 4 } }
						add = ROOT.major_research_project_cost
					}
					else_if = {
						limit = { any_scheme_agent_slot = { count = 3 } }
						add = ROOT.medium_research_project_cost
					}
					else = { add = ROOT.minor_research_project_cost }
					subtract = var:grant_money
				}
			}
		}
	}
	
	option = { # Increase funding
		show_as_unavailable = { always = yes }
		trigger = {
			trigger_if = {
				limit = { scope:scheme = { any_scheme_agent_slot = { count < 5 } } }
				treasury_or_gold >= scope:funding_required
			}
			trigger_else = {
				custom_tooltip = {
					text = project_slots_maxed.tt
					always = no
				}
			}
		}
		name = cotc_research_project_events.100.a
		custom_tooltip = {
			text = cotc_research_project_events.100.a.tt
			scope:scheme = {
				change_variable = { name = grant_money add = scope:funding_required }
				change_variable = { name = max_agent_slots add = 1 }
				add_research_project_agent_slots = yes
			}
		}
		remove_treasury_or_gold = scope:funding_required
		ai_chance = {
			base = 0
			modifier = { add = short_term_treasury_or_gold }
		}
	}
	option = { # Involve yourself personally
		name = cotc_research_project_events.100.b
		show_as_unavailable = { NOT = { has_character_modifier = researching_modifier } }
		trigger = {
			is_capable_adult = yes
			NOT = { has_character_modifier = researching_modifier }
		}
		add_research_project_personal_involvement = yes
		ai_chance = {
			base = 0
			modifier = {
				add = {
					value = ai_intellectualism # I like doing science!
					subtract = ai_sociability # I don't like being out of the spotlight
					subtract = { value = ai_greed min = 0 } # I don't like giving up control
				}
			}
			modifier = {
				add = { value = -50 subtract = ai_rationality max = 0 } # Our attention is needed elsewhere
				OR = {
					is_at_war = yes
					debt_level >= 0
				}
			}
			modifier = {
				add = { value = -30 subtract = ai_vengefulness max = 0 } # I don't trust any of my potential regents
				NOT = { any_vassal = { reverse_opinion = { target = ROOT value >= 0 } } }
			}
			modifier = { # I wouldn't be any use anyway
				factor = 0
				ai_rationality > 0 # Must not fall afoul of Dunning and Kruger
				learning <= 4
				trigger_if = {
					limit = { scope:scheme = { is_target_in_variable_list = { name = flags target = flag:military } } }
					martial <= 4
