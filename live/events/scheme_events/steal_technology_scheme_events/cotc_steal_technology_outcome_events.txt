namespace = cotc_steal_technology_outcome_events

cotc_steal_technology_outcome_events.1 = {
	hidden = yes

	immediate = {
		scope:scheme = {
			if = {
				limit = { scheme_type = steal_technology_character }
				scheme_target_character = {
					save_scope_as = st_target_character
					capital_county = { save_scope_as = st_target_county }
				}
			}
			scheme_owner = {
				save_scope_as = owner
				capital_county ?= { save_scope_as = st_home_county }
			}
			generic_scheme_process_ending_effect = {
				RESULTS_TOOLTIP = steal_technology_successful_roll_tt
				SUCCESS_EVENT_TYPE = on_action
				SUCCESS_EVENT_ID = steal_technology_success_on_action
				FAILURE_EVENT_TYPE = on_action
				FAILURE_EVENT_ID = steal_technology_failure_on_action
			}
		}
	}
}

	window = scheme_successful_event

	desc = {
		desc = cotc_steal_technology_outcome_events.100.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:scheme_discovered }
				desc = cotc_steal_technology_outcome_events.100.desc.discovered
			}
			desc = cotc_steal_technology_outcome_events.100.desc.undiscovered
		}
	}
	theme = generic_intrigue_scheme
	
	right_portrait = {
		character = ROOT
		animation = happy
	}
	left_portrait = {
		character = scope:st_target_character
		triggered_animation = {
			trigger = {
				exists = scope:scheme_discovered
				exists = scope:owner_to_reveal
			}
			animation = shock
		}
		animation = idle
	}

	widget = {
		gui = "event_window_widget_scheme"
		container = "custom_widgets_container"
	}
	
	option = {
		name = cotc_steal_technology_outcome_events.100.a
		if = {
			limit = {
				exists = scope:scheme_discovered
				exists = scope:owner_to_reveal
			}

		}

		scope:scheme = { end_scheme = yes }

			}
			animation = shock
		}
		animation = idle
	}
	lower_right_portrait = { character = scope:owner }
	
	immediate = {
		if = {
			limit = {
				exists = scope:scheme_discovered
				liege_or_court_owner = scope:st_target_character
			}
			reverse_add_opinion = { target = scope:st_target_character modifier = cotc_spy_opinion }
		}
	}
	
	option = { # If our involvement in the scheme was known and we're in the target's court, it might be time to get out of here!
		trigger = {
			exists = scope:scheme_discovered
			liege_or_court_owner = scope:st_target_character
			is_landed = no
			is_landless_ruler = no
		}
		name = cotc_steal_technology_outcome_events.111.a
		reverse_add_opinion = { target = scope:st_target_character modifier = cotc_spy_opinion }
		reverse_add_opinion = { target = scope:owner modifier = cotc_agent_opinion }
		show_as_tooltip = { scope:st_home_county = { add_tech_to_county = { TECH = scope:stolen_technology } } }
		move_to_pool_at = scope:st_home_county.title_province
		save_scope_value_as = { name = skipped_town value = flag:yes }
		scope:st_target_character = { trigger_event = cotc_steal_technology_outcome_events.112 }
		ai_chance = {
			base = 100
			modifier = { add = ai_rationality }
			modifier = {
				add = {
					subtract = ai_boldness
					min = 0
				}
			}
			opinion_modifier = {
				who = scope:st_target_character
				opinion_target = ROOT
				multiplier = -1
			}
		}
	}
	
	option = { # Otherwise we can stick around
		name = cotc_steal_technology_outcome_events.111.b
		if = {
			limit = {
				exists = scope:scheme_discovered
				liege_or_court_owner = scope:st_target_character
			}
			reverse_add_opinion = { target = scope:st_target_character modifier = cotc_spy_opinion }
			scope:st_target_character = { trigger_event = cotc_steal_technology_outcome_events.112 }
		}
		reverse_add_opinion = { target = scope:owner modifier = cotc_agent_opinion }
		show_as_tooltip = { scope:st_home_county = { add_tech_to_county = { TECH = scope:stolen_technology } } }
		ai_chance = {
			base = 0
			opinion_modifier = {
				who = ROOT
				opinion_target = scope:owner
				multiplier = -1
			}
		}
	}
}
