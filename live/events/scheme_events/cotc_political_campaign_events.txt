namespace = cotc_political_campaign_events
scripted_effect cotc_announce_political_campaign_progress = {
	send_interface_toast = {
		type = msg_political_campaign_good
		title = cotc_political_campaign_events.$EVENT_TYPE$.toast.title
		left_icon = scope:candidate
		custom_tooltip = cotc_political_campaign_events.$EVENT_TYPE$.toast.tt
	}
	scope:campaign = {
		every_in_list = {
			variable = rival_campaigns
			scheme_owner ?= {
				send_interface_message = {
					type = msg_political_campaign_bad
					title = cotc_political_campaign_events.$EVENT_TYPE$.notification.title
					desc = cotc_political_campaign_events.$EVENT_TYPE$.notification.desc
					left_icon = scope:candidate
				}
			}
		}
		scheme_target_title ?= {
			holder = {
				add_to_temporary_list = temp_interested_parties
				diarch ?= { add_to_temporary_list = temp_interested_parties }
			}
		}
		var:legislature ?= {
			var:legislature_title ?= {
				holder = {
					add_to_temporary_list = temp_interested_parties
					diarch ?= { add_to_temporary_list = temp_interested_parties }
				}
			}
		}
	}
	scope:location ?= {
		owner = {
			add_to_temporary_list = temp_interested_parties
			diarch ?= { add_to_temporary_list = temp_interested_parties }
		}
	}
	every_in_list = {
		name = temp_interested_parties
		send_interface_message = {
			type = msg_political_campaign_neutral
			title = cotc_political_campaign_events.$EVENT_TYPE$.notification.title
			desc = cotc_political_campaign_events.$EVENT_TYPE$.notification.desc
			left_icon = scope:candidate
		}
	}
}

scripted_effect cotc_hold_campaign_rally = {
	negative_sociability_stress_gain_effect = { SCALE = 1 }
	duel = {
		skill = diplomacy
		value = scope:rally_difficulty
		15 = { # Huge rally
			compare_modifier = {
				value = scope:duel_value
				multiplier = 3.5
				min = -49
			}
			min = 5
			desc = moderate_election_campaign_strength_gain_desc
			scope:campaign = { change_variable = { name = public_campaign_strength add = moderate_campaign_strength_change } }
			cotc_announce_political_campaign_progress = { EVENT_TYPE = huge_rally }
		}
		50 = { # Large rally
			desc = minor_election_campaign_strength_gain_desc
			scope:campaign = { change_variable = { name = public_campaign_strength add = minor_campaign_strength_change } }
			cotc_announce_political_campaign_progress = { EVENT_TYPE = large_rally }
		}
		5 = { # Modest rally
			compare_modifier = {
				value = scope:duel_value
				multiplier = -3.5
				min = -49
			}
			min = 5
			desc = miniscule_election_campaign_strength_gain_desc
			scope:campaign = { change_variable = { name = public_campaign_strength add = miniscule_campaign_strength_change } }
			cotc_announce_political_campaign_progress = { EVENT_TYPE = modest_rally }
		}
	}
}
cotc_political_campaign_events.100 = { # Hold a campaign rally
	type = character_event
	desc = {
		triggered_desc = {
			trigger = {
				exists = scope:legislature
				scope:candidate = { NOT = { knows_language_of_culture = scope:location.culture } }
			}
			desc = cotc_political_campaign_events.100.legislative.language_barrier
		}
		triggered_desc = {
			trigger = {
				exists = scope:legislature
				faith_hostility_level = { target = scope:candidate.faith value >= 3 }
			}
			desc = cotc_political_campaign_events.100.legislative.very_hostile_audience
		}
		triggered_desc = {
			trigger = {
				exists = scope:legislature
				faith_hostility_level = { target = scope:candidate.faith value >= 2 }
			}
			desc = cotc_political_campaign_events.100.legislative.hostile_audience
		}
		triggered_desc = {
			trigger = { exists = scope:legislature }
			desc = cotc_political_campaign_events.100.legislative
		}
		triggered_desc = {
			trigger = { scope:candidate = { NOT = { knows_language_of_culture = scope:location.culture } } }
			desc = cotc_political_campaign_events.100.executive.language_barrier
		}
		triggered_desc = {
			trigger = { faith_hostility_level = { target = scope:candidate.faith value >= 3 } }
			desc = cotc_political_campaign_events.100.executive.very_hostile_audience
		}
		triggered_desc = {
			trigger = { faith_hostility_level = { target = scope:candidate.faith value >= 2 } }
			desc = cotc_political_campaign_events.100.executive.hostile_audience
		}
		desc = cotc_political_campaign_events.100.executive
	}
	theme = realm
	
	right_portrait = {
		character = ROOT
		triggered_animation = {
			trigger = {
				ROOT = {
					ai_zeal > ai_honor
					ai_zeal > ai_greed
				}
			}
			animation = personality_zealous
		}
		triggered_animation = {
			trigger = { ROOT = { ai_honor > ai_greed } }
			animation = personality_honorable
		}
		animation = personality_greedy
	}
	
	trigger = { # Make sure we have all the required scopes
		exists = scope:campaign # The campaign this rally is supposed to advance
		exists = scope:location # A planet where we can hold the rally
		exists = scope:candidate # The candidate (should be ROOT)
		OR = {
			gold >= { value = 5 multiply = scope:campaign.political_campaign_cost_modifier }
			influence >= { value = 10 multiply = scope:campaign.political_campaign_cost_modifier }
		}
	}
	weight_multiplier = {
		base = 1
		modifier = { add = diplomacy }
		ai_value_scaled_factor_modifier = { CHARACTER = ROOT AI_VALUE = ai_sociability SCALE = 1 }
	}
	
	immediate = {
		save_scope_value_as = {
			name = rally_difficulty
			value = {
				value = 10
				if = {
					limit = { scope:candidate = { NOT = { knows_language_of_culture = scope:location.culture } } }
					add = 20
				}
				scope:location.faith = {
					if = {
						limit = { faith_hostility_level = { target = scope:candidate.faith value >= 3 } }
						add = 6
					}
					else_if = {
						limit = { faith_hostility_level = { target = scope:candidate.faith value >= 2 } }
						add = 4
					}
					else_if = {
						limit = { faith_hostility_level = { target = scope:candidate.faith value >= 1 } }
						add = 2
					}
				}
				min = 5
			}
		}
	}
	
	option = { # Spend gold to host the rally
		name = cotc_political_campaign_events.100.a
		trigger = { gold >= { value = 5 multiply = scope:campaign.political_campaign_cost_modifier } }
		remove_short_term_gold = { value = 5 multiply = scope:campaign.political_campaign_cost_modifier }
		cotc_hold_campaign_rally = yes
		ai_chance = {
			base = 0
			modifier = { add = diplomacy }
			modifier = {
				factor = 2
				short_term_gold > influence
			}
			modifier = {
				factor = 0
				NOT = { short_term_gold >= { value = 5 multiply = scope:campaign.political_campaign_cost_modifier } }
			}
		}
	}
	option = { # Spend influence to host the rally
		name = cotc_political_campaign_events.100.b
		trigger = { influence >= { value = 10 multiply = scope:campaign.political_campaign_cost_modifier } }
		change_influence = { value = -10 multiply = scope:campaign.political_campaign_cost_modifier }
		cotc_hold_campaign_rally = yes
		ai_chance = {
			base = 0
			modifier = { add = diplomacy }
			modifier = {
				factor = 2
				influence > short_term_gold
			}
		}
	}
	option = { # Do nothing
		name = cotc_political_campaign_events.100.c
		ai_chance = {
			base = 0
			modifier = { add = scope:rally_difficulty }
			ai_value_scaled_factor_modifier = { CHARACTER = ROOT AI_VALUE = ai_sociability SCALE = -1 }
		}
	}
}
scripted_effect = calculate_skill_political_importance = {
	save_scope_value_as = {
		name = military_importance
		value = {
			value = 1
			scope:relevant_title.holder ?= {
				if = {
					limit = { average_sub_realm_security_anxiety <= -20 }
					multiply = 0.5
				}
				else_if = {
					limit = { average_sub_realm_security_anxiety <= 0 }
					multiply = 0.8
				}
				else_if = {
					limit = { average_sub_realm_security_anxiety >= 40 }
					multiply = 2
				}
				else_if = {
					limit = { average_sub_realm_security_anxiety >= 20 }
					multiply = 1.25
				}
				if = {
					limit = { average_sub_realm_hawkishness < 0 }
					multiply = 0.7
				}
				else_if = {
					limit = { average_sub_realm_hawkishness > 1 }
					multiply = { value = 1 divide = 0.7 }
				}
			}
		}
	}
	save_scope_value_as = {
		name = economic_importance
		value = {
			value = 1
			scope:relevant_title.holder ?= {
				if = {
					limit = { average_sub_realm_economic_anxiety <= -20 }
					multiply = 0.5
				}
				else_if = {
					limit = { average_sub_realm_economic_anxiety <= 0 }
					multiply = 0.8
				}
				else_if = {
					limit = { average_sub_realm_economic_anxiety >= 40 }
					multiply = 2
				}
				else_if = {
					limit = { average_sub_realm_economic_anxiety >= 20 }
					multiply = 1.25
				}
			}
		}
	}
	save_scope_value_as = {
		name = technological_importance
		value = {
			value = 1
			scope:relevant_title.holder ?= {
				if = {
					limit = { average_sub_realm_intellectualism <= -1 }
					multiply = 0.2
				}
				else_if = {
					limit = { average_sub_realm_intellectualism = 0 }
					multiply = 0.5
				}
				else_if = {
					limit = { average_sub_realm_intellectualism = 2 }
					multiply = 1.25
				}
				else_if = {
					limit = { average_sub_realm_intellectualism >= 3 }
					multiply = 2
				}
			}
		}
	}
}

cotc_political_campaign_events.200 = { # Ad campaign
	type = character_event
	desc = {
		triggered_desc = {
			trigger = { exists = scope:legislature } # This is a legislative campaign
			desc = cotc_political_campaign_events.200.legislative
		}
		desc = cotc_political_campaign_events.200.executive
	}
	theme = realm
	
	right_portrait = {
		character = ROOT
		triggered_animation = {
			trigger = {
				ROOT = {
					ai_zeal > ai_honor
					ai_zeal > ai_greed
				}
			}
			animation = personality_zealous
		}
		triggered_animation = {
			trigger = { ROOT = { ai_honor > ai_greed } }
			animation = personality_honorable
		}
		animation = personality_greedy
	}
	
	trigger = { # Make sure we have all the required scopes
		exists = scope:campaign # The campaign this ad is supposed to advance
		exists = scope:relevant_title # The executive office or legislature title we're campaigning for
		exists = scope:candidate # The candidate (should be ROOT)
		OR = {
			gold >= { value = 10 multiply = scope:campaign.political_campaign_cost_modifier }
			influence >= { value = 10 multiply = scope:campaign.political_campaign_cost_modifier }
		}
	}
	weight_multiplier = {
		base = 1
		modifier = {
			add = {
				value = 0
				if = {
					limit = {
						diplomacy >= stewardship
						diplomacy >= martial
						diplomacy >= learning
					}
					add = diplomacy
				}
				else_if = {
					limit = {
						stewardship >= martial
						stewardship >= learning
					}
					add = stewardship
				}
				else_if = {
					limit = { martial >= learning }
					add = martial
				}
				else = { add = learning }
			}
		}
	}
	immediate = { calculate_skill_political_importance = yes }
	
	option = { # Spend gold to advertise your diplomacy
		trigger = { gold >= { value = 10 multiply = scope:campaign.political_campaign_cost_modifier } }
		name = cotc_political_campaign_events.200.a
		remove_short_term_gold = { value = 10 multiply = scope:campaign.political_campaign_cost_modifier }
		cotc_diplomatic_ad_campaign = yes
		ai_chance = {
			base = 0
			modifier = { add = diplomacy }
			modifier = {
				factor = 0
				short_term_gold < { value = -10 multiply = scope:campaign.political_campaign_cost_modifier }
			}
			modifier = {
				factor = 0
				short_term_gold < influence
			}
