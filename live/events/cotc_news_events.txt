# These are events intended to report major world events to the player, usually flavored
# as some sort of news broadcast. The goal is to keep the player relatively informed of
# what's going on in the world around them.

namespace = cotc_news

# Triggers on character death; report the news of this death to people who should care
cotc_news.0 = {
	type = character_event
	hidden = yes
	
	trigger = {
		# Is this character important enough for anyone to care about their death?
		OR = {
			is_ruler = yes
			is_diarch = yes
			is_councillor = yes
			prestige_level >= 3
			is_head_of_faith = yes
			any_heir_title = { tier >= tier_county }
		}
	}
	
	immediate = {
		ROOT = { save_scope_as = dead_character }
		every_player = { # The AI does not care about news
			limit = {
				in_diplomatic_range = ROOT
				# Why should the player care about this character?
				OR = {
					# We care about them personally
					has_relation_rival = ROOT
					has_relation_friend = ROOT
					has_relation_lover = ROOT
					has_relation_soulmate = ROOT
					is_close_family_of = ROOT
					is_spouse_of_even_if_dead = ROOT
					# They're our head of faith
					faith = { religious_head = ROOT }
					# They're our liege or suzerain
					target_is_liege_or_above = ROOT
					is_tributary_of = ROOT
					# They're on our council
					is_councillor_of = ROOT
					is_diarch_of_target = ROOT
					# We're they're direct liege or suzerain
					ROOT = { is_vassal_of = PREV }
					ROOT = { is_tributary_of = PREV }
					# They're an independent ruler we should care about...
					AND = {
						ROOT = {
							is_independent_ruler = yes
							OR = {
								# ...because they're just important in general
								primary_title = { tier >= tier_kingdom }
								# ...because they're a neighbor
								character_is_realm_neighbor = PREV
								# ...because we're at war with them
								is_at_war_with = PREV
								PREV = { any_liege_or_above = { is_at_war_with = ROOT } }
								# ...because we're allied with them
								is_allied_to = PREV
								PREV = { any_liege_or_above = { is_allied_to = ROOT } }
								# ...because we're on the same side in a war
								is_allied_in_war = PREV
								PREV = { any_liege_or_above = { is_allied_in_war = PREV } }
							}
						}
					}
				}
			}
			# Okay, this player should be informed, fire away!
			trigger_event = { id = cotc_news.1 }
		}
	}
}
# Inform a player that someone you should care about died
cotc_news.1 = {
	type = character_event
	desc = cotc_news.1.desc
	theme = death # Should probably make a theme specific for news broadcasts
	
	right_portrait = {
		character = scope:dead_character
		animation = dead
	}
	
	immediate = {
		# Save information about the dead character so we can use it in localization
		set_variable = { name = dead_character value = scope:dead_character }
		scope:dead_character = {
			if = {
				limit = { location ?= { is_sea_province = yes } }
				location = { ROOT = { set_variable = { name = death_location_deep_space value = PREV } } }
			}
			else_if = {
				limit = {
					OR = {
						is_commanding_army = yes
						is_travelling = yes
						is_landless_adventurer = yes
					}
				}
				location ?= { county ?= { ROOT = { set_variable = { name = death_system value = PREV } } } }
			}
			else = { location ?= { ROOT = { set_variable = { name = death_location value = PREV } } } }
			primary_title ?= {
				ROOT = { set_variable = { name = news_source_title value = PREV } }
				current_heir ?= { ROOT = { set_variable = { name = death_title_heir value = PREV } } }
			}
			if = {
				limit = {
					OR = {
						is_diarch = yes
						is_councillor = yes
						is_powerful_vassal = yes
					}
				}
				liege ?= { primary_title ?= { ROOT = { set_variable = { name = news_source_title value = PREV } } } }
			}
			ROOT = { set_variable = { name = dead_character_prestige value = PREV.prestige_level } }
		}
	}
	
	# Make a comment based on your feelings about the dead character
	option = {
		trigger = { opinion = { target = scope:dead_character value >= 80 } }
		name = cotc_news.1.a
	}
	option = {
		trigger = {
			opinion = { target = scope:dead_character value >= 20 }
			opinion = { target = scope:dead_character value < 80 }
		}
		name = cotc_news.1.b
	}
	option = {
		trigger = {
			opinion = { target = scope:dead_character value >= -19 }
			opinion = { target = scope:dead_character value < 20 }
		}
		name = cotc_news.1.c
	}
	option = {
		trigger = {
			opinion = { target = scope:dead_character value >= -79 }
			opinion = { target = scope:dead_character value < -19 }
		}
		name = cotc_news.1.d
	}
	option = {
		trigger = { opinion = { target = scope:dead_character value < -79 } }
		name = cotc_news.1.e
	}
	after = {
		# Get rid of all the extra variables
		remove_variable = dead_character
		remove_variable = death_location_deep_space
		remove_variable = death_system
		remove_variable = death_location
		remove_variable = death_title_heir
		remove_variable = death_liege_title
		remove_variable = dead_character_prestige
		remove_variable = news_source_title
	}
}
# Triggers on war declaration, lets players know there's a big war starting
cotc_news.10 = {
	hidden = yes
	
	immediate = {
		every_player = {
			limit = {
				# These characters will find out rather more directly
				NOR = {
					THIS = scope:attacker
					THIS = scope:defender
				}
				scope:attacker = { top_liege = { save_temp_scope_as = temp_attacker_realm } }
				scope:defender = { top_liege = { save_temp_scope_as = temp_defender_realm } }
				top_liege = { save_temp_scope_as = temp_player_realm }
				# Why is this war worth reporting to the player at all
				OR = {
					# We're involved indirectly
					scope:temp_player_realm = scope:temp_attacker_realm
					scope:temp_player_realm = scope:temp_defender_realm
					# We care about the attacker or defender
					has_relation_rival = scope:attacker
					has_relation_friend = scope:attacker
					has_relation_lover = scope:attacker
					has_relation_soulmate = scope:attacker
					is_close_family_of = scope:attacker
					is_spouse_of = scope:attacker
					has_relation_rival = scope:defender
					has_relation_friend = scope:defender
					has_relation_lover = scope:defender
					has_relation_soulmate = scope:defender
					is_close_family_of = scope:defender
					is_spouse_of = scope:defender
					# This conflict involves multiple great powers
					AND = {
						scope:attacker.primary_title = { tier >= tier_kingdom  }
						scope:defender.primary_title = { tier >= tier_kingdom }
					}
					# This is happening right next door to our realm
					any_sub_realm_county ?= {
						any_neighboring_county ?= {
							OR = {
								holder = scope:attacker
								holder = scope:defender
								holder = { target_is_liege_or_above = scope:attacker }
								holder = { target_is_liege_or_above = scope:defender }
								holder = { top_liege = { is_tributary_of = scope:attacker } }
								holder = { top_liege = { is_tributary_of = scope:defender } }
							}
						}
					}
				}
			}
			trigger_event = { id = cotc_news.11 }
		}
	}
}
cotc_news.11 = {
	type = character
	desc = cotc_news.11.desc
	theme = war # Should probably make a theme specific for news broadcasts
	
	left_portrait = {
		character = scope:attacker
		animation = war_attacker
	}
	right_portrait = {
		character = scope:defender
		animation = war_defender
	}
	
	immediate = {
		# Save attacker so they can be used as a news source
		set_variable = { name = news_source_title value scope:attacker.primary_title }
		set_variable = { name = news_source_character value = scope:attacker }
	}
	# No special opinion
	option = {
		name = cotc_news.11.a
	}
	# Hates both
	option = {
		trigger = {
			opinion = { target = scope:attacker value <= -20 }
			opinion = { target = scope:defender value <= -20 }
		}
		exclusive = yes
		name = cotc_news.11.b
	}
	# We're with the attackers, and we approve of this war
	option = {
		trigger = {
			is_liege_or_above = scope:attacker
			OR = {
				AND = {
					opinion = { target = scope:defender value <= -20 }
					ai_vengefulness >= ai_compassion
				}
				AND = {
					NOT = { faith = scope:defender }
					ai_zeal >= ai_compassion
				}
				AND = {
					ai_compassion <= -50
					ai_honor <= -50
					ai_rationality <= 0
				}
			}
			NOT = {
				opinion = { target = scope:defender value >= 20 }
				OR = {
					ai_compassion >= 0
					ai_honor >= 0
				}
			}
		}
		exclusive = yes
		name = cotc_news.11.c
	}
	# We're with the defender, and have a strong positive opinion of the attacker
	option = {
		trigger = {
			is_liege_or_above = scope:defender
			opinion = { target = scope:attacker value >= 60 }
			opinion = { target = scope:defender value > 0 }
		}
		exclusive = yes
		name = cotc_news.11.d
	}
	# We're with the defender, and have a strong positive opinion of the attacker and a strong negative opinion of the defender
	option = {
		trigger = {
			is_liege_or_above = scope:defender
			opinion = { target = scope:attacker value >= 60 }
			opinion = { target = scope:defender value <= 0 }
			ai_honor <= 0
		}
		exclusive = yes
		name = cotc_news.11.e
	}
	option = {
		trigger = {
			is_liege_or_above = scope:defender
			opinion = { target = scope:attacker value >= 60 }
			opinion = { target = scope:defender value <= 0 }
			ai_honor > 0
		}
		exclusive = yes
		name = cotc_news.11.f
	}
	# We're with the defender, and none of that stuff applies
	option = {
		trigger = {
			is_liege_or_above = scope:defender
			opinion = { target = scope:attacker value < 60 }
			ai_boldness > 0
		}
		exclusive = yes
		name = cotc_news_11.g
	}
	option = {
		trigger = {
			is_liege_or_above = scope:defender
			opinion = { target = scope:attacker value < 60 }
			ai_boldness <= 0
		}
		exclusive = yes
		name = cotc_news_11.h
	}
	# We're just don't like war in general
	option = {
		trigger = {
			ai_compassion >= 10
			ai_vengefulness <= 0
		}
		exclusive = yes
		name = cotc_news.11.i
	}
	
	after = {
		remove_variable = news_source_title
		remove_variable = news_source_character
	}
}
