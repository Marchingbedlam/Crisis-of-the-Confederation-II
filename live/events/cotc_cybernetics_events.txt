			positive_vengefulness_stress_gain_effect = { SCALE = 0.5 }

			}
			add = {
				value = "opinion(scope:physician)"
				divide = 2
			}
			subtract = {

				multiply = 2
			}
			if = { # If we're smart, be more forgiving of honest mistakes
				limit = { ai_rationality >= 0 }

					}
					else_if = {

						multiply = 3
					}
					else_if = {

					}
				}
			}
		}

		name = cotc_cybernetics_events.1030.c
		imprison_physician_effect = yes

			value = 50

			subtract = {
				value = ai_lawfulness
				min = 0
			}
		}

		execute_physician_effect = yes
