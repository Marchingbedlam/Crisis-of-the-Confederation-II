# Inform anyone relevant that the election has begun
legislature_events.1 = {
	desc = legislature_events.1.desc
	
	theme = news
	right_portrait = {
		character = scope:situation.var:legislature_title.holder
		animation = chancellor
	}
	
	immediate = {
		set_variable = { name = news_source_title value = scope:situation.var:legislature_title }
		set_variable = { name = news_source_character value = scope:situation.var:legislature_title.holder }
	}
	
	option = {
		name = legislature_events.1.a
		remove_variable = news_source_title
		remove_variable = news_source_character
	}
}

# You've been elected as a legislator
legislature_events.10 = {
	desc = legislature_events.10.desc
	
	theme = realm
	right_portrait = {
		character = ROOT
		animation = war_over_win
	}
	left_portrait = {
		character = scope:situation.var:legislature_title.holder
		animation = acknowledging
	}
	
	immediate = {
		save_scope_as = legislator_elect
		# Start counting our time in the legislature
		if = {
			limit = { NOT = { has_variable = years_in_legislature } }
			set_variable = { name = years_in_legislature value = 0 }
		}
		trigger_event = { id = legislature_events.20 years = 1 }
		# Inform our liege that we've been elected
		if = {
			limit = { is_pool_guest = yes }
			court_owner = {
				send_interface_toast = {
					type = msg_legislator_elected
					title = legislature_events.10.legislator_elected.tt
					desc = legislature_events.10.legislator_elected.desc
				}
			}
		}
		else = {
			liege = {
				send_interface_toast = {
					type = msg_legislator_elected
					title = legislature_events.10.legislator_elected.tt
					desc = legislature_events.10.legislator_elected.desc
				}
			}
		}
	}
	
	option = {
		name = legislature_events.10.a
		# If we're not already in the capital system, move there to be available
		# for senate meetings
		if = {
			limit = { is_pool_guest = yes }
			court_owner = { add_courtier = ROOT }
		}
		if = {
			limit = { NOT = { lives_in_capital_trigger = { TARGET = scope:situation.var:legislature_title.holder } } }
			visit_random_court_in_capital = { TARGET = scope:situation.var:legislature_title.holder }
		}
	}
}

				variable = legislators
				trigger_event = { id = legislature_events.301 }
			}
		}
	}
	
	option = {
		name = legislature_events.300.a
	}
}

# Build list of lobbyists for a legislator
legislature_events.1110 = {
	hidden = yes
	immediate = {
		# The ruler of your constituency, who may have had a hand in your election
		if = {
			limit = { exists = var:riding.holder }
			add_to_variable_list = { name = possible_lobbyists target = var:riding.holder }
		}
		# Friends and lovers
		every_relation = {
			type = friend
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = lover
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = soulmate
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		every_relation = {
			type = elder
			if = {
				limit = { NOT = { ROOT = { is_target_in_variable_list = { target = PREV name = possible_lobbyists } } } }
				ROOT = { add_to_variable_list = { name = possible_lobbyists target = PREV } }
			}
		}
		# The leader of this legislator's ideology in the legislature
		scope:activity = {
			ordered_attending_character = {
				limit = {
					save_temporary_scope_as = evaluated_character
					is_character_in_legislature_trigger = { LEGISLATURE = scope:activity.var:legislature CHARACTER = scope:evaluated_character }
					faith = ROOT.faith
				}
				order_by = legislative_seniority
				ROOT = {
					if = {
						limit = {
							NOR = {
								ROOT = PREV
								is_target_in_variable_list = { target = PREV name = possible_lobbyists }
							}
						}
						add_to_variable_list = { name = possible_lobbyists target = PREV }
					}
				}
			}
		}
	}
}

legislature_events.2000 = {
	desc = legislature_events.2000.desc
	
	theme = realm
	override_background = council_chamber
	
	right_portrait = {
		character = ROOT
		animation = thinking
	}
	left_portrait = {
		character = scope:host
		animation = throne_room_ruler
	}
	
	immediate = { set_variable = event_did_fire }
	
	# Intend to support the bill
	option = {
		name = legislature_events.2000.a
		scope:activity = {
			add_to_variable_list = {
				name = bill_supporters
				target = ROOT
			}
		}
		vote_for_budget_strength_effect = yes
		ai_chance = {
			base = 50
			# I just want to block any bills, period
			modifier = {
				factor = 0
				has_activity_intent = block_bill_intent
			}
			# I want to balance the budget, and this bill is overspendingg
			modifier = {
				factor = 0
				OR = {
					has_activity_intent = balance_budget_intent
					has_activity_intent = balance_budget_maintain_military_intent
					has_activity_intent = balance_budget_maintain_ministry_intent
					has_activity_intent = balance_budget_maintain_salary_intent
				}
				scope:activity.proposed_total_budget > 100
			}
			# This budget makes cuts I don't like
			modifier = {
				factor = 0
				has_activity_intent = balance_budget_maintain_military_intent
				scope:activity.var:proposed_military_budget < scope:activity.var:ruler.current_military_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = balance_budget_maintain_ministry_intent
				scope:activity.var:proposed_ministry_budget < scope:activity.var:ruler.current_ministry_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = balance_budget_maintain_salary_intent
				scope:activity.var:proposed_salary_budget < scope:activity.var:ruler.current_salary_budget
			}
			# I want spending increases this budget doesn't provide
			modifier = {
				factor = 0
				has_activity_intent = increase_military_budget_intent
				scope:activity.var:proposed_military_budget <= scope:activity.var:ruler.current_military_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = increase_ministry_budget_intent
				scope:activity.var:proposed_ministry_budget <= scope:activity.var:ruler.current_ministry_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = increase_salary_budget_intent
				scope:activity.var:proposed_salary_budget <= scope:activity.var:ruler.current_salary_budget
			}
		}
	}
	# Intend to oppose the bill
	option = {
		name = legislature_events.2000.b
		scope:activity = {
			add_to_variable_list = {
				name = bill_opponents
				target = ROOT
			}
		}
		vote_against_budget_strength_effect = yes
		ai_chance = {
			base = 50
			# I am a rubber stamp
			modifier = {
				factor = 0
				has_activity_intent = support_bill_intent
			}
			# This is a balanced budget
			modifier = {
				factor = 0
				OR = {
					has_activity_intent = balance_budget_intent
					has_activity_intent = balance_budget_maintain_military_intent
					has_activity_intent = balance_budget_maintain_ministry_intent
					has_activity_intent = balance_budget_maintain_salary_intent
				}
				NOR = {
					has_activity_intent = balance_budget_maintain_military_intent
					scope:activity.var:proposed_military_budget < scope:activity.var:ruler.current_military_budget
				}
				NOR = {
					has_activity_intent = balance_budget_maintain_ministry_intent
					scope:activity.var:proposed_ministry_budget < scope:activity.var:ruler.current_ministry_budget
				}
				NOR = {
					has_activity_intent = balance_budget_maintain_salary_intent
					scope:activity.var:proposed_salary_budget < scope:activity.var:ruler.current_salary_budget
				}
				scope:activity.proposed_total_budget <= 100
			}
			# This budget increases spending in a way I like
			modifier = {
				factor = 0
				has_activity_intent = increase_military_budget_intent
				scope:activity.var:proposed_military_budget > scope:activity.var:ruler.current_military_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = increase_ministry_budget_intent
				scope:activity.var:proposed_ministry_budget > scope:activity.var:ruler.current_ministry_budget
			}
			modifier = {
				factor = 0
				has_activity_intent = increase_salary_budget_intent
				scope:activity.var:proposed_salary_budget > scope:activity.var:ruler.current_salary_budget
			}
		}
	}
}

	desc = legislature_events.2200.desc
	
	theme = realm
	override_background = council_chamber
	
	immediate = {
		ordered_in_list = {
			variable = supporters_gained
			order_by = legislative_seniority
			save_scope_as = representative_opponent
		}
	}

		character = scope:representative_opponent
		animation = debate
	}
	left_portrait = {
		character = ROOT
		animation = thinking
	}
	
	# Agree to the amendment
	option = {
		name = legislature_events.2200.a
		custom_tooltip = amend_bill_desc
		custom_tooltip = gain_supporters_desc
		custom_tooltip = lose_supporters_desc
		hidden_effect = {
			scope:activity = {
				set_variable = { name = proposed_military_budget value = ROOT.var:amended_military_budget }
				set_variable = { name = proposed_ministry_budget value = ROOT.var:amended_ministry_budget }
				set_variable = { name = proposed_salary_budget value = ROOT.var:amended_salary_budget }
			}
			every_in_list = {
				variable = supporters_gained
				trigger_event = { id = legislature_events.1300 }
			}
			every_in_list = {
				variable = supporters_lost
				trigger_event = { id = legislature_events.1310 }
			}
		}
		ai_chance = {
			base = 90
			modifier = {
				add = ai_rationality
			}
			# Don't do this if you would lose more supporters than you gain
			modifier = {
				factor = 0
				total_supporters_lost >= total_supporters_gained
			}
		}
	}
	
	# Do not agree to the amendment
	option = {
		name = legislature_events.2200.b
		hidden_effect = {
			every_in_list = {
				variable = supporters_gained
				send_interface_toast = {
					type = msg_vote_change_bad
					title = legislature_events.2200.refused_amendment.tt
					desc = legislature_events.2200.refused_amendment.desc
				}
			}
			every_in_list = {
				variable = supporters_lost
				send_interface_toast = {
					type = msg_vote_change_good
					title = legislature_events.2200.refused_amendment.tt
					desc = legislature_events.2200.refused_amendment.desc
				}
			}
		}
		ai_chance = {
			base = 10
			modifier = { add = ai_vengefulness }
			modifier = { add = ai_greed }
		}
	}
	
	# Clean up all the now-unnecessary variables
	after = {
		remove_variable = amended_military_budget
		remove_variable = amended_ministry_budget
		remove_variable = amended_salary_budget
		clear_variable_list = supporters_gained
		clear_variable_list = supporters_lost
	}
