# Inform anyone relevant that the election has begun
legislature_events.1 = {
	desc = legislature_events.1.desc
	
	theme = news
	right_portrait = {
		character = scope:situation.var:legislature_title.holder
		animation = chancellor
	}
	
	immediate = {
		set_variable = { name = news_source_title value = scope:situation.var:legislature_title }
		set_variable = { name = news_source_character value = scope:situation.var:legislature_title.holder }
	}
	
	option = {
		name = legislature_events.1.a
		remove_variable = news_source_title
		remove_variable = news_source_character
	}
}

# You've been elected as a legislator
legislature_events.10 = {
	desc = legislature_events.10.desc
	
	theme = realm
	right_portrait = {
		character = ROOT
		animation = war_over_win
	}
	left_portrait = {
		character = scope:situation.var:legislature_title.holder
		animation = acknowledging
	}
	
	immediate = {
		save_scope_as = legislator_elect
		# Start counting our time in the legislature
		if = {
			limit = { NOT = { has_variable = years_in_legislature } }
			set_variable = { name = years_in_legislature value = 0 }
		}
		trigger_event = { id = legislature_events.20 years = 1 }
		# Inform our liege that we've been elected
		if = {
			limit = { is_pool_guest = yes }
			court_owner = {
				send_interface_message = {
					type = msg_legislator_elected
					title = legislature_events.10.legislator_elected.tt
					desc = legislature_events.10.legislator_elected.desc
					left_icon = scope:legislator_elect
				}
			}
		}
		else = {
			liege = {
				send_interface_message = {
					type = msg_legislator_elected
					title = legislature_events.10.legislator_elected.tt
					desc = legislature_events.10.legislator_elected.desc
					left_icon = scope:legislator_elect
				}
			}
		}
	}
	
	option = {
		name = legislature_events.10.a
		# If we're not already in the capital system, move there to be available
		# for senate meetings
		if = {
			limit = { is_pool_guest = yes }
			court_owner = { add_courtier = ROOT }
		}
		if = {
			limit = { NOT = { lives_in_capital_trigger = { TARGET = scope:situation.var:legislature_title.holder } } }
			visit_random_court_in_capital = { TARGET = scope:situation.var:legislature_title.holder }
		}
	}
}

# Legislature abolished
legislature_events.100 = {
	desc = legislature_events.100.desc
	
	theme = realm
	right_portrait = {
		character = ROOT
		animation = throne_room_ruler
	}
	
	immediate = {
		save_scope_as = ruler
		scope:situation = {
			situation_participant_group:vassals = { every_situation_group_participant = { trigger_event = { id = legislature_events.101 } } }
		}
	}
	
	option = {
		name = legislature_events.100.a
		scope:situation = {
			hidden_effect = { var:legislature_title = { remove_variable = legislature } }
			end_situation = yes
		}
	}
}

# Legislature promoted
legislature_events.200 = {
	desc = legislature_events.200.desc
	
	theme = realm
	right_portrait = {
		character = ROOT
		animation = throne_room_ruler
	}
	
	immediate = {
		save_scope_as = ruler
		scope:ruler.primary_title = { set_variable = { name = legislature value = scope:situation } }
		scope:situation = {
			var:legislature_title = {
				remove_variable = legislature
				save_scope_as = old_title
			}
			set_variable = { name = legislature_title value = scope:ruler.primary_title }
			situation_participant_group:vassals = { every_situation_group_participant = { trigger_event = { id = legislature_events.201 } } }
			every_in_list = {
				list = legislators
				trigger_event = { id = legislature_events.201 }
			}
		}
	}
	
	option = { name = legislature_events.200.a }
}

# Legislatures merged
legislature_events.300 = {
	desc = legislature_events.300.desc
	
	theme = realm
	right_portrait = {
		character = ROOT
		animation = throne_room_ruler
	}
	
	# Run this only once, for the primary title
	trigger = { scope:situation.var:legislature_title = { this = ROOT.primary_title } }
	
	immediate = {
		save_scope_as = ruler
		every_held_title = {
			limit = {
				NOT = { this = ROOT.primary_title }
				has_variable = legislature
			}
			var:legislature = {
				every_in_list = {
					variable = legislators
					add_to_legislature_effect = { LEGISLATURE = scope:situation RIDING = var:riding }
				}
				situation_participant_group:vassals = {
					every_situation_group_participant = {
						recalculate_participant_group = scope:situation
					}
				}
				end_situation = yes
			}
		}
		scope:situation = {
			situation_participant_group:vassals = {
				every_situation_group_participant = {
					trigger_event = { id = legislature_events.301 }
				}
			}
			every_in_list = {
				variable = legislators
				trigger_event = { id = legislature_events.301 }
			}
		}
	}
	
	option = {
		name = legislature_events.300.a
	}
}
